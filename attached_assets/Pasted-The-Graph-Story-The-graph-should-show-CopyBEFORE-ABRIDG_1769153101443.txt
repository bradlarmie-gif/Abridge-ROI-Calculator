function calculateFullScaleValue(currentValue, currentProviders, currentUtilization, targetProviders, targetUtilization) {
  // Scale multiplier
  const providerMultiplier = targetProviders / currentProviders;
  
  // Utilization multiplier
  const utilizationMultiplier = targetUtilization / currentUtilization;
  
  // Maturity multiplier (compounding effects)
  const maturityMultiplier = 1.15; // 15% bonus from full maturity
  
  return currentValue * providerMultiplier * utilizationMultiplier * maturityMultiplier;
}

function generateJourneyData(currentState, hardValue, expansion) {

  const { targetProviders, targetUtilization } = expansion;

  const valuePerProvider = hardValue / providers;

  const fullScaleValue = calculateFullScaleValue(

      value: 0,
      providers: providers,
      utilization: 0,
      isActual: true,

function calculateFullScaleValue(currentValue, currentProviders, currentUtil, targetProviders, targetUtil) {
