// Inputs (user-editable)
const providers = 40;                      // Total providers
const annualEncounters = 80000;            // Total outpatient encounters
const utilizationRate = 0.65;              // % of encounters using Abridge
const timeSavedPerEncounter = 2.5;         // Minutes saved per encounter
const accessAllocation = 0.33;             // % of saved time → access potential
const conversionRate = 0.60;               // % of access time → actual visits
const timePerVisit = 30;                   // Minutes per visit
const revenuePerVisit = 200;               // Blended reimbursement per visit

// Calculations

// Step 1: Time Returned
const eligibleEncounters = annualEncounters * utilizationRate;
// 80,000 × 0.65 = 52,000 encounters

const minutesReturned = eligibleEncounters * timeSavedPerEncounter;
// 52,000 × 2.5 = 130,000 minutes

const hoursReturned = minutesReturned / 60;
// 130,000 ÷ 60 = 2,167 hrs

// Step 2: Time Allocated to Access
const accessHours = hoursReturned * accessAllocation;
// 2,167 × 0.33 = 715 hrs

// Step 3: Conversion to Visits
const usableHours = accessHours * conversionRate;
// 715 × 0.60 = 429 hrs

// Step 4: New Visits
const usableMinutes = usableHours * 60;
// 429 × 60 = 25,740 minutes

const additionalVisits = usableMinutes / timePerVisit;
// 25,740 ÷ 30 = 858 visits

// Step 5: Revenue Impact
const annualAccessRevenue = additionalVisits * revenuePerVisit;
// 858 × $200 = $171,600
