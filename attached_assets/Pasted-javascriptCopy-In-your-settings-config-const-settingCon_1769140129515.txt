      'survey-compliance',
      'care-coordination'
    ]
  },

    hasQualitativeSection: true,

      'patient-experience'
    ]
  },
  inpatient: {
    hasQualitativeSection: true,
    qualitativeStyle: 'inline', // Simple one-liner
    qualitativeDrivers: [
      'care-coordination',

    ]
  }
};
jsxCopy// In your ROI Results page

function QualitativeBenefits({ setting, selectedQualitativeDrivers }) {
  const config = settingConfig[setting];
  
  // No qualitative for outpatient
  if (!config.hasQualitativeSection) {
    return null;
  }
  
  // Filter to only show drivers the user actually selected
  const driversToShow = config.qualitativeDrivers.filter(
    d => selectedQualitativeDrivers.includes(d)
  );
  
  if (driversToShow.length === 0) {
    return null;
  }
  
  // Full section for Nursing
  if (config.qualitativeStyle === 'full') {
    return <TheFullPictureSection drivers={driversToShow} />;
  }
  
  // Inline callout for ED and Inpatient
  if (config.qualitativeStyle === 'inline') {
    return <InlineQualitativeCallout drivers={driversToShow} />;
  }
  
  return null;
}
jsxCopy// The inline version for ED/Inpatient

function InlineQualitativeCallout({ drivers }) {
  const driverContent = {
    'patient-experience': {
      icon: '😊',
      title: 'Patient Experience',
      description: 'Less waiting room frustration, better communication during high-stress encounters'
    },
    'care-coordination': {
      icon: '🔗',
      title: 'Care Coordination',
      description: 'Complete documentation enables better handoffs between care teams'
    },
    'survey-compliance': {
      icon: '✓',
      title: 'Survey Readiness',
      description: 'Charts that are always complete and audit-ready'
    }
  };
  
  return (
    <div className="inline-qualitative-callout">
      <div className="callout-header">
        <span className="callout-icon">✨</span>
        <span className="callout-label">Also improves (not quantified):</span>
      </div>
      <div className="callout-drivers">
        {drivers.map(driverId => {
          const driver = driverContent[driverId];
          return (
            <div key={driverId} className="callout-driver">
              <span className="driver-icon">{driver.icon}</span>
              <span className="driver-title">{driver.title}</span>
              <span className="driver-description">— {driver.description}</span>
            </div>
          );
        })}
      </div>
    </div>
  );
}
