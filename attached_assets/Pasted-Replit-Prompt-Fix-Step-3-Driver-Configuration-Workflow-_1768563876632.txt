PART 4: Main Step 3 Component (State Management)
tsxCopyfunction GapAnalysisStep({ competitor, baselineInputs, onContinue }) {
  // State
  const [activeDriverId, setActiveDriverId] = useState<string | null>(null);
  const [configuredDrivers, setConfiguredDrivers] = useState<Record<string, DriverCalculation>>({});
  
  // All 6 drivers
  const ALL_DRIVERS = [
    { id: 'patient_access', label: 'Patient Access', subtitle: '...', icon: '📈' },
    { id: 'overtime', label: 'Overtime Savings', subtitle: '...', icon: '⏰' },
    { id: 'level_of_service', label: 'Level of Service', subtitle: '...', icon: '📊' },
    { id: 'retention', label: 'Clinician Retention', subtitle: '...', icon: '👤' },
    { id: 'hcc', label: 'HCC Capture', subtitle: '...', icon: '🏥' },
    { id: 'denials', label: 'Documentation Denials', subtitle: '...', icon: '⚠️' },
  ];
  
  // Handle calculate
  const handleCalculate = (driverId: string, assumptions: Record<string, number>) => {
    const calculation = calculateDriver(driverId, assumptions, baselineInputs, competitor);
    setConfiguredDrivers(prev => ({
      ...prev,
      [driverId]: calculation
    }));
    setActiveDriverId(null); // Close modal
  };
  
  // Handle edit (re-opens modal)
  const handleEdit = (driverId: string) => {
    setActiveDriverId(driverId);
  };
  
  // Handle remove
  const handleRemove = (driverId: string) => {
    setConfiguredDrivers(prev => {
      const next = {...prev};
      delete next[driverId];
      return next;
    });
  };
  
  const selectedDrivers = Object.entries(configuredDrivers).map(([id, calc]) => ({
    ...ALL_DRIVERS.find(d => d.id === id)!,
    ...calc
  }));
  
  const totalGap = selectedDrivers.reduce((sum, d) => sum + d.gap, 0);
  
  return (
    <div className="space-y-8">
      {/* Gap at a Glance - keep as is */}
      <GapAtAGlance {...} />
      
      {/* Driver Selection */}
      <section>
        <h2 className="text-xl font-semibold text-gray-900 mb-2">
          How the Gap Translates to Value
        </h2>
        <p className="text-gray-500 mb-6">
          Select the value drivers that matter to your organization
        </p>
        
        {/* 3x2 Grid of Mini Cards */}
        <div className="grid grid-cols-3 gap-4 mb-6">
          {ALL_DRIVERS.map(driver => (
            <DriverMiniCard
              key={driver.id}
              driver={driver}
              isConfigured={!!configuredDrivers[driver.id]}
              gapValue={configuredDrivers[driver.id]?.gap}
              onClick={() => setActiveDriverId(driver.id)}
            />
          ))}
        </div>
        
        {/* Expanded Cards (configured drivers) */}
        <div className="space-y-4">
          {selectedDrivers.map(driver => (
            <DriverExpandedCard
              key={driver.id}
              driver={driver}
              calculations={configuredDrivers[driver.id]}
              competitor={competitor}
              onEdit={() => handleEdit(driver.id)}
              onRemove={() => handleRemove(driver.id)}
            />
          ))}
        </div>
      </section>
      
      {/* Summary */}
      {selectedDrivers.length > 0 && (
        <DriverSummary 
          selectedDrivers={selectedDrivers}
          totalGap={totalGap}
          onContinue={() => onContinue(selectedDrivers)}
        />
      )}
      
      {/* Configuration Modal */}
      {activeDriverId && (
        <DriverConfigModal
          driver={ALL_DRIVERS.find(d => d.id === activeDriverId)!}
          competitor={competitor}
          onCalculate={handleCalculate}
          onClose={() => setActiveDriverId(null)}
        />
      )}
    </div>
  );
}

Summary of Workflow:

Click mini card → Modal opens
Enter values in modal form
Click "Calculate Value" → Modal closes, expanded card appears
Click "Edit assumptions" on expanded card → Modal re-opens with current values
Click "Remove" → Expanded card disappears, mini card returns to "Click to configure" state
