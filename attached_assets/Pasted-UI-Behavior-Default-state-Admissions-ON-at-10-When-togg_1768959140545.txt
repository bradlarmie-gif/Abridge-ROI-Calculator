javascriptCopy// ============ ED PATIENT THROUGHPUT (LWBS) ============

// Inputs (user-editable)
const annualEdVisits = 45000;              // Annual ED volume
const lwbsRate = 0.035;                    // 3.5% LWBS rate, adjustable
const improvementRate = 0.20;              // 20% LWBS reduction
const avgEdVisitRevenue = 600;             // Avg ED visit revenue, adjustable

// Admission toggle
const includeAdmissions = true;            // Toggle ON/OFF
const admissionPercent = 0.10;             // 10% default, adjustable when ON
const avgAdmissionRevenue = 15000;         // Avg admission revenue, adjustable

// Calculations

// Step 1: Current LWBS
const patientsLeaving = annualEdVisits * lwbsRate;
// 45,000 × 0.035 = 1,575 patients

// Step 2: Patients Retained
const patientsRetained = patientsLeaving * improvementRate;
// 1,575 × 0.20 = 315 patients

// Step 3: Revenue Mix
let edVisitPatients, admissionPatients, edVisitRevenue, admissionRevenue;

if (includeAdmissions) {
  // Split between ED visits and admissions
  edVisitPatients = patientsRetained * (1 - admissionPercent);
  admissionPatients = patientsRetained * admissionPercent;
  
  edVisitRevenue = edVisitPatients * avgEdVisitRevenue;
  admissionRevenue = admissionPatients * avgAdmissionRevenue;
} else {
  // All ED visits, no admissions
  edVisitPatients = patientsRetained;
  admissionPatients = 0;
  
  edVisitRevenue = edVisitPatients * avgEdVisitRevenue;
  admissionRevenue = 0;
}

// Step 4: Total Value
const totalLwbsRevenue = edVisitRevenue + admissionRevenue;

// With admissions ON (default):
// ED: 284 × $600 = $170,400
// Admissions: 31 × $15,000 = $465,000
// Total: $635,400

// With admissions OFF:
// ED: 315 × $600 = $189,000
// Total: $189,000
