export function calculateOvertimeReduction(eligibleEncounters, inputs) {
  const minutesSaved = inputs.minutesSaved || inputs.patientAccess.minutesSaved;
  const afterHoursReduction = inputs.afterHoursReduction || 0.20;
  const premiumRate = inputs.premiumRate || 145;
  
  const totalHours = (minutesSaved × eligibleEncounters) / 60;
  const premiumHours = totalHours × afterHoursReduction;
  const value = premiumHours × premiumRate;
  
  return Math.round(value);
}
