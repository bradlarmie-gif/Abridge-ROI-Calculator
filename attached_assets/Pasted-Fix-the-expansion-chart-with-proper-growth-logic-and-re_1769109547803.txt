// This creates the curved line between milestones
function generateSmoothCurve(milestones) {
  const points = [];
  
  for (let i = 0; i < milestones.length - 1; i++) {
    const current = milestones[i];
    const next = milestones[i + 1];
    
    // Create 10 interpolated points between each milestone
    for (let j = 0; j <= 10; j++) {
      const t = j / 10;
      
      // Use ease-in-out curve (S-curve) instead of linear
      const eased = t < 0.5 
        ? 4 * t * t * t // Ease in (slow start)
        : 1 - Math.pow(-2 * t + 2, 3) / 2; // Ease out (slow end)
      
      points.push({
        x: current.providers + (next.providers - current.providers) * eased,
        y: current.value + (next.value - next.value) * eased
      });
    }
  }
  
  return points;
}
