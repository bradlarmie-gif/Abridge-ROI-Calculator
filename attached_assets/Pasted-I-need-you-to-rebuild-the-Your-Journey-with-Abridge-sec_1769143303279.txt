## CALCULATION LOGIC

Here's how to calculate the values:

```javascript
function calculateScenario(pilot, fullScaleBeds, targetUtilization, pace) {
  // Pilot values (locked from their model)
  const pilotValue = pilot.annualValue;
  const pilotBeds = pilot.beds;
  const pilotUtilization = pilot.utilization;
  const pilotInvestment = pilot.annualInvestment;
  
  // Scale multiplier
  const bedMultiplier = fullScaleBeds / pilotBeds;
  
  // Linear projection (just scaling)
  const linearFullScale = pilotValue * bedMultiplier;
  
  // Utilization boost
  const utilizationBoost = targetUtilization / pilotUtilization;
  
  // Base full scale (beds + utilization)
  const baseFullScale = linearFullScale * utilizationBoost;
  
  // Pace affects maturity bonus
  const paceConfig = {
    measured: { months: 36, maturityMultiplier: 1.15 },
    steady: { months: 24, maturityMultiplier: 1.20 },
    aggressive: { months: 18, maturityMultiplier: 1.25 }
  };
  
  const { months, maturityMultiplier } = paceConfig[pace];
  
  // Final full scale value with maturity effects
  const fullScaleValue = baseFullScale * maturityMultiplier;
  
  // Compounding bonus (everything above linear)
  const compoundingBonus = fullScaleValue - linearFullScale;
  
  // Full scale investment
  const pricePerBed = pilotInvestment / pilotBeds / 12; // monthly price per bed
  const fullScaleInvestment = fullScaleBeds * pricePerBed * 12;
  
  // ROI calculations
  const pilotROI = pilotValue / pilotInvestment;
  const fullScaleROI = fullScaleValue / fullScaleInvestment;
  
  return {
    pilot: {
      value: pilotValue,
      roi: pilotROI
    },
    fullScale: {
      value: fullScaleValue,
      roi: fullScaleROI
    },
    compounding: {
      value: compoundingBonus,
      months: months
    },
    // Data points for the graph
    graphData: generateGraphData(pilot, fullScaleBeds, targetUtilization, months)
  };
}

function generateGraphData(pilot, fullScaleBeds, targetUtilization, totalMonths) {
  const points = [];
  const milestones = [0, 6, 12, 18, 24, 36].filter(m => m <= totalMonths || m === 0);
  
  // Make sure we include the final point
  if (!milestones.includes(totalMonths)) {
    milestones.push(totalMonths);
  }
  
  milestones.forEach(month => {
    const progress = month / totalMonths; // 0 to 1
    
    // Beds scale linearly over time
    const beds = Math.round(pilot.beds + (fullScaleBeds - pilot.beds) * progress);
    
    // Utilization ramps up (slightly curved)
    const utilizationProgress = Math.pow(progress, 0.8); // Faster early gains
    const utilization = pilot.utilization + (targetUtilization - pilot.utilization) * utilizationProgress;
    
    // Linear value (just bed scaling)
    const linearValue = pilot.annualValue * (beds / pilot.beds);
    
    // Actual value (beds + utilization + maturity)
    const utilizationBoost = utilization / pilot.utilization;
    const maturityBoost = 1 + (0.2 * Math.pow(progress, 1.5)); // Compounds over time
    const actualValue = linearValue * utilizationBoost * maturityBoost;
    
    points.push({
      month,
      label: month === 0 ? 'Today' : month === totalMonths ? 'Full Scale' : `${month} mo`,
      beds: Math.round(beds),
      utilization: Math.round(utilization),
      linearValue: Math.round(linearValue),
      actualValue: Math.round(actualValue),
      phase: getPhase(progress)
    });
  });
  
  return points;
}

function getPhase(progress) {
  if (progress === 0) return 'Pilot';
  if (progress < 0.3) return 'Early Expansion';
  if (progress < 0.6) return 'Expansion';
  if (progress < 0.9) return 'Scale';
  return 'Full Scale';
}
