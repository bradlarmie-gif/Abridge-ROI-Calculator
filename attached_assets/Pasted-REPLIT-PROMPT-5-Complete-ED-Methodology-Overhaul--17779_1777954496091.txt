        </div>                                                                                                                                                                                                 
        <div className="space-y-4">                                                                                                                                                                            
          {edDomainMethodology.map((domain) => {         
            const detailData = domainDetails.find((d) => d.domain === domain.domain);
            const isExpanded = expandedDomain === domain.domain;

              />                                         
            );
          })}
        </div>
      </div>
    );
