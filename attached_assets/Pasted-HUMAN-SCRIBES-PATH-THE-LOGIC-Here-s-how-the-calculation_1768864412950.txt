// Inputs
const totalProviders = 50;
const providersWithScribes = 20;
const scribeHourlyRate = 25;
const scribeHoursPerWeek = 40;
const weeksPerYear = 50;

// Calculated
const providersWithoutScribes = totalProviders - providersWithScribes;
const coveragePercent = (providersWithScribes / totalProviders) * 100;

// Current scribe cost
const annualScribeCost = providersWithScribes * scribeHourlyRate * scribeHoursPerWeek * weeksPerYear;

// Abridge cost (placeholder - make this configurable)
const abridgeCostPerProvider = 0; // Set to 0 or placeholder for now
const annualAbridgeCost = totalProviders * abridgeCostPerProvider;

// Direct savings
const directCostSavings = annualScribeCost - annualAbridgeCost;

// Turnover costs
const turnoverRate = 0.35;
const costPerReplacement = 4000;
const annualTurnoverCost = providersWithScribes * turnoverRate * costPerReplacement;

// Management overhead (estimate)
const managementOverhead = providersWithScribes > 10 ? 20000 : 10000;

// Hidden costs total
const hiddenCosts = annualTurnoverCost + managementOverhead;

// Expanded coverage value (optional - time savings for newly covered providers)

const hoursSavedNewProviders = (providersWithoutScribes * encountersPerProvider * timeSavedPerEncounter) / 60;

// Total value
const totalAnnualValue = directCostSavings + hiddenCosts + expandedCoverageValue;
