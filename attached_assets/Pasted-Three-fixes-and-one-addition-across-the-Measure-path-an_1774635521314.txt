  Fix 2 — clipboard.ts (iOS clipboard broken)                                                                                                                                                                  
                                                                                                                                                                                                               
  The current copyToClipboard tries execCopy first (fails silently on iOS with hidden textarea), then navigator.clipboard.writeText, then calls navigator.clipboard.readText() to verify — which throws on iOS
  because readText requires separate permission. This causes the function to always return false on iOS, showing the fallback textarea every time.                                                             
                                                                                                          
  Replace the entire copyToClipboard function with:
                                                                                                                                                                                                               
  export async function copyToClipboard(text: string): Promise<boolean> {
    // Modern Clipboard API — works on iOS 13.4+, Chrome, Firefox                                                                                                                                              
    if (navigator.clipboard?.writeText) {                                                                 
      try {                                   
        await navigator.clipboard.writeText(text);                                                                                                                                                             
        return true;                                                                                                                                                                                           
      } catch {                                                                                                                                                                                                
        // fall through to execCommand                                                                                                                                                                         
      }                                                                                                                                                                                                        
    }                                                                                                     

    // Legacy fallback — older desktop browsers                                                                                                                                                                
    try {
      const textarea = document.createElement('textarea');                                                                                                                                                     
      textarea.value = text;                                                                              
      // Must be in viewport for iOS execCommand to work
      textarea.style.cssText = 'position:fixed;top:0;left:0;width:1px;height:1px;opacity:0.01;border:none;padding:0;';
      document.body.appendChild(textarea);                                                                                                                                                                     
      textarea.focus();                       
      textarea.select();                                                                                                                                                                                       
      textarea.setSelectionRange(0, text.length);                                                                                                                                                              
      const ok = document.execCommand('copy');                                                                                                                                                                 
      document.body.removeChild(textarea);                                                                                                                                                                     
      return ok;                                                                                                                                                                                               
    } catch {                                                                                             
      return false;                                                                                                                                                                                            
    }
  }
