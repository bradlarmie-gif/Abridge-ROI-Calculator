export const calculateExpansionModel = (baseline, inputs) => {
  const newProviders = inputs.targetProviders - baseline.providers;
  
  // Calculate costs
  const baselineCost = baseline.annualCost || (baseline.providers * baseline.costPerProviderMonth * 12);
  const scenarioCost = calculateScenarioCost(inputs, baseline);
  
  // Maturity curves for different drivers
  const maturityCurves = {
    access: { year1: 0.50, year2: 0.80, year3: 1.00 },
    retention: { year1: 0.20, year2: 0.70, year3: 1.00 }, // Lag effect
    los: { year1: 0.50, year2: 0.80, year3: 1.00 },
    locum: { year1: 0.50, year2: 0.80, year3: 0.95 }, // Efficiency gains
    denials: { year1: 0.50, year2: 0.80, year3: 1.00 },
    hcc: { year1: 0.50, year2: 0.80, year3: 1.00 }
  };

  // Calculate benefits by year
  const calculateYearBenefits = (year) => {
    const benefitsByDriver = {};
    let totalBenefit = 0;

    Object.entries(baseline.benefits || {}).forEach(([driver, baseValue]) => {
      if (baseValue > 0) {
        const valuePerProvider = baseValue / baseline.providers;
        const curve = maturityCurves[driver] || { year1: 0.5, year2: 0.8, year3: 1.0 };
        const maturityFactor = curve[`year${year}`];
        
        // Apply validation adjustments
        const adjustedValue = applyDriverValidation(
          driver, 
          valuePerProvider, 
          newProviders, 
          maturityFactor,
          inputs.driverValidations?.[driver],
          baseline
        );
        
        benefitsByDriver[driver] = adjustedValue;
        totalBenefit += adjustedValue;
      }
    });

    return { benefitsByDriver, totalBenefit };
  };

  const year1 = calculateYearBenefits(1);
  const year2 = calculateYearBenefits(2);
  const year3 = calculateYearBenefits(3);

  // Build the model
  return {
    baseline: {
      providers: baseline.providers,
      cost: baselineCost,
      benefit: baseline.totalBenefit,
      netGain: baseline.totalBenefit - baselineCost,
      roi: baseline.totalBenefit / baselineCost
    },
    
    year1: {
      avgUtilization: 0.50,
      totalCost: scenarioCost,
      incrementalCost: scenarioCost - baselineCost,
      benefitsByDriver: year1.benefitsByDriver,
      totalBenefit: baseline.totalBenefit + year1.totalBenefit,
      incrementalBenefit: year1.totalBenefit,
      totalNetGain: (baseline.totalBenefit + year1.totalBenefit) - scenarioCost,
      incrementalNetGain: year1.totalBenefit - (scenarioCost - baselineCost),
      totalROI: (baseline.totalBenefit + year1.totalBenefit) / scenarioCost,
      incrementalROI: year1.totalBenefit / (scenarioCost - baselineCost)
    },
    
    year2: {
      avgUtilization: 0.80,
      totalCost: scenarioCost,
      incrementalCost: scenarioCost - baselineCost,
      benefitsByDriver: year2.benefitsByDriver,
      totalBenefit: baseline.totalBenefit + year2.totalBenefit,
      incrementalBenefit: year2.totalBenefit,
      totalNetGain: (baseline.totalBenefit + year2.totalBenefit) - scenarioCost,
      incrementalNetGain: year2.totalBenefit - (scenarioCost - baselineCost),
      totalROI: (baseline.totalBenefit + year2.totalBenefit) / scenarioCost,
      incrementalROI: year2.totalBenefit / (scenarioCost - baselineCost)
    },
    
    year3: {
      avgUtilization: 0.90,
      totalCost: scenarioCost,
      incrementalCost: scenarioCost - baselineCost,
      benefitsByDriver: year3.benefitsByDriver,
      totalBenefit: baseline.totalBenefit + year3.totalBenefit,
      incrementalBenefit: year3.totalBenefit,
      totalNetGain: (baseline.totalBenefit + year3.totalBenefit) - scenarioCost,
      incrementalNetGain: year3.totalBenefit - (scenarioCost - baselineCost),
      totalROI: (baseline.totalBenefit + year3.totalBenefit) / scenarioCost,
      incrementalROI: year3.totalBenefit / (scenarioCost - baselineCost)
    },
    
    threeYear: {
      totalIncrementalCost: (scenarioCost - baselineCost) * 3,
      totalIncrementalBenefit: year1.totalBenefit + year2.totalBenefit + year3.totalBenefit,
      totalIncrementalNetGain: (year1.totalBenefit + year2.totalBenefit + year3.totalBenefit) - ((scenarioCost - baselineCost) * 3),
      blendedROI: (year1.totalBenefit + year2.totalBenefit + year3.totalBenefit) / ((scenarioCost - baselineCost) * 3)
    },
    
    incremental: {
      providers: newProviders
    },
    
    hasRetention: baseline.benefits?.retention > 0,
    hasAccess: baseline.benefits?.access > 0
  };
};

function calculateScenarioCost(inputs, baseline) {
  const { targetProviders, pricingModel, enterpriseCost } = inputs;
  
  if (pricingModel === 'per-provider') {
    return targetProviders * baseline.costPerProviderMonth * 12;
  }
  
  if (enterpriseCost && enterpriseCost > 0) {
    return enterpriseCost;
  }
  
  // Apply volume discount
  const discount = getVolumeDiscount(targetProviders);
  return targetProviders * baseline.costPerProviderMonth * 12 * (1 - discount);
}

function getVolumeDiscount(providers) {
  if (providers >= 250) return 0.30;
  if (providers >= 150) return 0.25;
  if (providers >= 100) return 0.20;
  if (providers >= 75) return 0.15;
  return 0;
}

function applyDriverValidation(driver, valuePerProvider, newProviders, maturityFactor, validation, baseline) {
  let adjustedValue = valuePerProvider * newProviders * maturityFactor;
  
  // Apply driver-specific adjustments
  if (validation) {
    switch(driver) {
      case 'access':
        if (!validation.hasCapacity) {
          adjustedValue *= 0.5; // Reduce if no capacity constraint
        }
        if (!validation.sameDemand) {
          const ratio = validation.additionalVisitsPerWeek / (baseline.benefits.access / baseline.providers / 48 / (baseline.revenuePerVisit || 200));
          adjustedValue *= ratio;
        }
        break;
      
      case 'retention':
        // Already handled by maturity curve lag
        break;
      
      case 'los':
        if (!validation.sameCaseMix) {
          const baselineWRVU = baseline.benefits.los / baseline.providers / (baseline.revenuePerWRVU || 40);
          const ratio = validation.wrvuUplift / baselineWRVU;
          adjustedValue *= ratio;
        }
        break;
    }
  }
  
  return Math.round(adjustedValue);
}

import { LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, Area, AreaChart } from 'recharts';

const MaturityCurveViz = ({ model }) => {
  // Generate data points for the curve
  const data = [
    { month: 0, utilization: 0, label: 'Start' },
    { month: 1, utilization: 40, label: 'M1' },
    { month: 3, utilization: 50, label: 'M3' },
    { month: 6, utilization: 58, label: 'M6', stage: 'Ramp-Up' },
    { month: 9, utilization: 68, label: 'M9' },
    { month: 12, utilization: 75, label: 'M12', stage: 'Maturing' },
    { month: 15, utilization: 80, label: 'M15' },
    { month: 18, utilization: 85, label: 'M18' },
    { month: 21, utilization: 88, label: 'M21', stage: 'Mature' },
    { month: 24, utilization: 90, label: 'M24' }
  ];

  const CustomTooltip = ({ active, payload }) => {
    if (active && payload && payload.length) {
      return (
        <div className="maturity-tooltip">
          <p className="tooltip-label">{payload[0].payload.label}</p>
          <p className="tooltip-value">{payload[0].value}% utilization</p>
          {payload[0].payload.stage && (
            <p className="tooltip-stage">{payload[0].payload.stage}</p>
          )}
        </div>
      );
    }
    return null;
  };

  return (
    <div className="maturity-curve-container">
      <ResponsiveContainer width="100%" height={300}>
        <AreaChart data={data} margin={{ top: 10, right: 30, left: 0, bottom: 0 }}>
          <defs>
            <linearGradient id="utilizationGradient" x1="0" y1="0" x2="0" y2="1">
              <stop offset="5%" stopColor="#ef4444" stopOpacity={0.3}/>
              <stop offset="95%" stopColor="#ef4444" stopOpacity={0}/>
            </linearGradient>
          </defs>
          <CartesianGrid strokeDasharray="3 3" stroke="#e5e7eb" />
          <XAxis 
            dataKey="month" 
            label={{ value: 'Months', position: 'insideBottom', offset: -5 }}
            stroke="#6b7280"
          />
          <YAxis 
            label={{ value: 'Utilization %', angle: -90, position: 'insideLeft' }}
            stroke="#6b7280"
            domain={[0, 100]}
          />
          <Tooltip content={<CustomTooltip />} />
          <Area 
            type="monotone" 
            dataKey="utilization" 
            stroke="#ef4444" 
            strokeWidth={3}
            fillOpacity={1} 
            fill="url(#utilizationGradient)" 
          />
        </AreaChart>
      </ResponsiveContainer>
    </div>
  );
};

export default MaturityCurveViz;
