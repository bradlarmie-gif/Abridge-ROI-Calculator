2. Step 4 State Management
typescriptCopyfunction OpportunityStep({ step3Data }) {
  // Scale controls
  const [providers, setProviders] = useState(step3Data.baselineProviders);
  const [encounters, setEncounters] = useState(step3Data.baselineEncounters);
  
  // Driver assumptions (can be edited in Step 4)
  const [driverAssumptions, setDriverAssumptions] = useState(
    step3Data.selectedDrivers.reduce((acc, d) => ({
      ...acc,
      [d.id]: d.assumptions
    }), {})
  );
  
  // Calculate scale factor
  const scaleFactor = encounters / step3Data.baselineEncounters;
  
  // Recalculate each driver when assumptions or scale change
  const scaledDrivers = useMemo(() => {
    return step3Data.selectedDrivers.map(driver => {
      const currentAssumptions = driverAssumptions[driver.id];
      const recalculated = calculateDriver(driver.id, currentAssumptions, step3Data);
      
      return {
        ...driver,
        assumptions: currentAssumptions,
        competitorValue: recalculated.competitorValue * scaleFactor,
        abridgeValue: recalculated.abridgeValue * scaleFactor,
        gap: recalculated.gap * scaleFactor,
      };
    });
  }, [driverAssumptions, scaleFactor]);
  
  // Totals
  const totalCompetitorValue = scaledDrivers.reduce((sum, d) => sum + d.competitorValue, 0);
  const totalAbridgeValue = scaledDrivers.reduce((sum, d) => sum + d.abridgeValue, 0);
  const totalGap = totalAbridgeValue - totalCompetitorValue;
  
  // Multi-year projections (Abridge ramps up, competitor stays flat)
  const chartData = [
    { year: 'Today', competitor: 0, abridge: 0 },
    { year: 'Year 1', competitor: totalCompetitorValue, abridge: totalAbridgeValue * 0.4 },
    { year: 'Year 2', competitor: totalCompetitorValue * 2, abridge: totalAbridgeValue * 0.4 + totalAbridgeValue * 0.7 },
    { year: 'Year 3', competitor: totalCompetitorValue * 3, abridge: totalAbridgeValue * 0.4 + totalAbridgeValue * 0.7 + totalAbridgeValue },
  ];
  
  const threeYearGap = chartData[3].abridge - chartData[3].competitor;

}
