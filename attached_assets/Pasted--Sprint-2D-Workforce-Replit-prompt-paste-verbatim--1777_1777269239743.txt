  import { useMemo } from "react";                           
  import { FormattedNumberInput } from "@/components/FormattedNumberInput";                                                                                                                                    
  import type { ExploreCalcComponentProps } from "@/lib/exploreDrivers";
                                                                                                                                                                                                               
  type Props = ExploreCalcComponentProps;                 
                                                                                                                                                                                                               
  export default function PhysicianLocumAgencyCalc({ state, updateTimeDriverInputs }: Props) {                                                                                                                 
    const { timeDriverInputs } = state;                                                                                                                                                                        
                                                                                                                                                                                                               
    const retentionScenarios: Record<string, number> = { conservative: 5, typical: 10, optimistic: 15 };
                                                                                                                                                                                                               
    const calc = useMemo(() => {                             
      const turnoverRate = timeDriverInputs.annualTurnoverRate / 100;                                                                                                                                          
      const burnoutRate = timeDriverInputs.burnoutRelatedTurnover / 100;
      const impactRate = retentionScenarios[timeDriverInputs.retentionImpactScenario] / 100;                                                                                                                   
      const providersLeaving = state.numberOfProviders * turnoverRate;
      const burnoutRelated = providersLeaving * burnoutRate;                                                                                                                                                   
      const retained = burnoutRelated * impactRate;                                                                                                                                                            
      const weeksOfCoverage = timeDriverInputs.physicianAgencyWeeksPerVacancy || 16;                                                                                                                           
      const weeklyPremium = timeDriverInputs.physicianAgencyWeeklyPremium || 5000;                                                                                                                             
      const value = Math.round(retained * weeksOfCoverage * weeklyPremium);
      return { retained, weeksOfCoverage, weeklyPremium, value };                                                                                                                                              
    }, [state.numberOfProviders, timeDriverInputs.annualTurnoverRate, timeDriverInputs.burnoutRelatedTurnover, timeDriverInputs.retentionImpactScenario, timeDriverInputs.physicianAgencyWeeksPerVacancy,
  timeDriverInputs.physicianAgencyWeeklyPremium]);                                                                                                                                                             
                                                             
    const formatCurrency = (n: number) => '$' + Math.round(n).toLocaleString();                                                                                                                                
                                                             
    return (                                                                                                                                                                                                   
      <div>                                                                                                                                                                                                    
        {!timeDriverInputs.wellbeingEnabled && (                                                                                                                                                               
          <div className="mb-3 px-3 py-2 bg-amber-50 border border-amber-200 rounded-lg text-xs text-amber-800">                                                                                               
            Locum/Agency Cost Avoidance depends on Provider Wellbeing's retention calculation. Enable Provider Wellbeing above and turn on "Calculate retention value" to see this value.                      
          </div>                                          
        )}                                                                                                                                                                                                     
        {timeDriverInputs.wellbeingEnabled && !timeDriverInputs.calculateRetentionValue && (
          <div className="mb-3 px-3 py-2 bg-amber-50 border border-amber-200 rounded-lg text-xs text-amber-800">                                                                                               
            Turn on "Calculate retention value" inside Provider Wellbeing to compute Locum/Agency avoidance.
          </div>                                          
        )}                                                                                                                                                                                                     
                                                                                                                                                                                                               
        <p className="text-sm text-black mb-4">                                                                                                                                                                
          When retention improves, contracted locum and agency physician coverage tends to decline. This models the avoided premium-labor spend tied to retained providers.                                    
        </p>                                                                                                                                                                                                   
                                                                                                                                                                                                               
        <p className="text-xs font-medium text-[#888888] uppercase tracking-[1.5px] mb-3">Your Locum / Agency Profile</p>                                                                                      
                                                                                                                                                                                                               
        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 sm:gap-6 mb-6">                                                                                                                                  
          <div className="space-y-2.5">                                                                                                                                                                        
            <label className="text-sm text-[#888888]">Weeks of agency coverage per vacancy</label>                                                                                                             
            <FormattedNumberInput                            
              value={timeDriverInputs.physicianAgencyWeeksPerVacancy}                                                                                                                                          
              onChange={(v: number) => updateTimeDriverInputs({ physicianAgencyWeeksPerVacancy: v })}
              className="h-12 bg-white"                                                                                                                                                                        
            />                                               
            <p className="text-xs text-[#888888]">Typical: 12-20 weeks to fill a physician vacancy.</p>                                                                                                        
          </div>                                             
          <div className="space-y-2.5">                                                                                                                                                                        
            <label className="text-sm text-[#888888]">Weekly premium above base cost</label>                                                                                                                   
            <div className="relative">                       
              <span className="absolute left-3 top-1/2 -translate-y-1/2 text-sm text-[#888888]">$</span>                                                                                                       
              <FormattedNumberInput                          
                value={timeDriverInputs.physicianAgencyWeeklyPremium}                                                                                                                                          
                onChange={(v: number) => updateTimeDriverInputs({ physicianAgencyWeeklyPremium: v })}
                className="h-12 bg-white pl-7"                                                                                                                                                                 
              />                                                                                                                                                                                               
            </div>                                                                                                                                                                                             
            <p className="text-xs text-[#888888]">Premium paid above the equivalent base salary for the role.</p>                                                                                              
          </div>                                             
        </div>                                                                                                                                                                                                 
                                                                                                                                                                                                               
        <p className="text-xs font-medium text-[#888888] uppercase tracking-[1.5px] mb-3">Calculation</p>                                                                                                      
                                                                                                                                                                                                               
        <div className="bg-[#F5F0EB] rounded-lg p-4">     
          <div className="space-y-2 text-sm">                                                                                                                                                                  
            <div className="flex justify-between gap-2">     
              <span className="text-[#666666]">Providers retained (from Wellbeing)</span>                                                                                                                      
              <span className="font-semibold text-black">{calc.retained.toFixed(2)}</span>
            </div>                                                                                                                                                                                             
            <div className="flex justify-between gap-2">                                                                                                                                                       
              <span className="text-[#666666]">× Weeks of coverage avoided per retained provider</span>
              <span className="font-semibold text-black">{calc.weeksOfCoverage}</span>                                                                                                                         
            </div>                                           
            <div className="flex justify-between gap-2">                                                                                                                                                       
              <span className="text-[#666666]">× Weekly premium</span>
              <span className="font-semibold text-black">{formatCurrency(calc.weeklyPremium)}</span>                                                                                                           
            </div>                                                                                                                                                                                             
            <div className="h-px bg-[#333333] my-2" />                                                                                                                                                         
            <div className="flex justify-between gap-2">                                                                                                                                                       
              <span className="font-semibold text-black">Locum/Agency Avoidance</span>                                                                                                                         
              <span className="font-bold text-[#EA2C00] flex-shrink-0">{formatCurrency(calc.value)}</span>
            </div>                                                                                                                                                                                             
          </div>                                             
        </div>                                                                                                                                                                                                 
      </div>                                                                                                                                                                                                   
    );                                                       
  }
