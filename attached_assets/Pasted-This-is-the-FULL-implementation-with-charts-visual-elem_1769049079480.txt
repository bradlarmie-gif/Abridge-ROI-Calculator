  async generate() {
    const { jsPDF } = window.jspdf;
    this.pdf = new jsPDF({
      orientation: 'portrait',
      unit: 'mm',
      format: 'a4'
    });

    // Generate all pages
    this.addCoverPage();           // Page 1

  generateFilename() {
    const date = new Date().toISOString().split('T')[0];
    const setting = this.data.setting.replace(/\s+/g, '_');
    return `Abridge_ROI_${setting}_${this.data.providers}P_${date}.pdf`;
  }

  drawBox(x, y, width, height, fillColor = this.colors.lightGray, strokeColor = null) {
    this.pdf.setFillColor(...fillColor);
    if (strokeColor) {
      this.pdf.setDrawColor(...strokeColor);
      this.pdf.roundedRect(x, y, width, height, 3, 3, 'FD');
    } else {
      this.pdf.roundedRect(x, y, width, height, 3, 3, 'F');
    }
  }

  drawCircle(x, y, radius, fillColor) {
    this.pdf.setFillColor(...fillColor);
    this.pdf.circle(x, y, radius, 'F');
  }

  // Draw a simple bar chart
  drawBarChart(x, y, width, height, data) {
    const barWidth = width / data.length;
    const maxValue = Math.max(...data.map(d => d.value));
    
    data.forEach((item, i) => {
      const barHeight = (item.value / maxValue) * height;
      const barX = x + (i * barWidth) + 2;
      const barY = y + height - barHeight;
      
      // Bar
      this.pdf.setFillColor(...item.color);
      this.pdf.rect(barX, barY, barWidth - 4, barHeight, 'F');
      
      // Label
      this.pdf.setFontSize(8);
      this.pdf.setTextColor(...this.colors.darkGray);
      this.pdf.text(
        item.label,
        barX + (barWidth - 4) / 2,
        y + height + 5,
        { align: 'center' }
      );
      
      // Value
      this.pdf.setFontSize(9);
      this.pdf.setFont('helvetica', 'bold');
      this.pdf.text(
        item.displayValue,
        barX + (barWidth - 4) / 2,
        barY - 2,
        { align: 'center' }
      );
    });
  }

      `Generated: ${new Date().toLocaleString('en-US', { 
        dateStyle: 'long',
        timeStyle: 'short'
      })}`,
      `Model ID: ${this.data.modelId}`,
      `Version: 2.1`,
      `Methodology: Based on 200+ health system deployments`
    ];

    details.forEach(detail => {
      this.pdf.setFontSize(8);
      this.pdf.setFont('helvetica', 'normal');
      this.pdf.setTextColor(...this.colors.mediumGray);
      this.pdf.text(detail, this.margin, y);
      y += 5;
    });

    y += 8;

    // Disclaimer
    this.pdf.setFontSize(7);
    this.pdf.setFont('helvetica', 'italic');
    this.pdf.setTextColor(...this.colors.mediumGray);
    const disclaimer = 'This model is provided for evaluation purposes. Actual results may vary based on deployment specifics, adoption rates, and organizational factors. All financial projections are estimates based on industry data and should be validated with your finance team.';
    const disclaimerLines = this.pdf.splitTextToSize(disclaimer, this.contentWidth);
    this.pdf.text(disclaimerLines, this.margin, y);

    this.addPageNumber();
  }

  // ============================================
  // UTILITY METHODS
  // ============================================

  formatNumber(num) {
    if (num >= 1000000) {
      return (num / 1000000).toFixed(1) + 'M';
    } else if (num >= 1000) {
      return (num / 1000).toFixed(0) + 'K';
    }
    return num.toLocaleString();
  }
}

  const generator = new AbridgeROIPDF(modelData);
  generator.generate();
}
