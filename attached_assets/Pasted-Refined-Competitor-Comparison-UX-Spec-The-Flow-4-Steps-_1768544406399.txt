STEP 2: Current Usage Details
Key insight: Different inputs for AI solutions vs Human Scribe. Show calculated summaries in real-time.
tsxCopy// CurrentUsageForm.tsx

interface UsageInputs {
  // For AI solutions
  providers: number;
  monthlyEncounters: number;
  monthlyCostPerProvider: number;
  utilizationRate: number; // 0-100
  timeSavedPerEncounter: number; // minutes
  
  // Optional refinements
  baselineWrvu?: number;
  wrvuChange?: number;
}

interface ScribeInputs {
  numberOfScribes: number;
  hourlyRate: number;
  hoursPerWeek: number;
  providersCovered: number;
  encountersCovered: number;
  coveragePercent: number;
  reductionPercent: number;
}

export function CurrentUsageForm({ competitor, onComplete }: Props) {
  const [inputs, setInputs] = useState<UsageInputs>(getDefaults(competitor));
  
  // Real-time calculations
  const annualCost = inputs.providers * inputs.monthlyCostPerProvider * 12;
  const annualEncounters = inputs.monthlyEncounters * 12;
  const documentedEncounters = annualEncounters * (inputs.utilizationRate / 100);
  const hoursReturned = (inputs.timeSavedPerEncounter * documentedEncounters) / 60;
  
  return (
    <div className="space-y-8">
      <div>
        <h2 className="text-xl font-semibold text-gray-900">
          Tell us about your {competitor.name} deployment
        </h2>
      </div>
      
      {/* Core Metrics Section */}
      <div className="space-y-6">
        <div className="text-sm font-medium text-gray-500 uppercase tracking-wide">
          Core Metrics
        </div>
        
        <div className="grid grid-cols-2 gap-6">
          <InputField
            label="Total Providers Using"
            value={inputs.providers}
            onChange={(v) => setInputs({ ...inputs, providers: v })}
            suffix="providers"
          />
          
          <InputField
            label="Monthly Cost per Provider"
            value={inputs.monthlyCostPerProvider}
            onChange={(v) => setInputs({ ...inputs, monthlyCostPerProvider: v })}
            prefix="$"
            suffix="/mo"
          />
        </div>
        
        <InputField
          label="Monthly Outpatient Encounters"
          value={inputs.monthlyEncounters}
          onChange={(v) => setInputs({ ...inputs, monthlyEncounters: v })}
          hint={`~${Math.round(inputs.monthlyEncounters / inputs.providers)} encounters/provider/month`}
        />
      </div>
      
      {/* Performance Section */}
      <div className="space-y-6">
        <div className="text-sm font-medium text-gray-500 uppercase tracking-wide">
          Current Performance
        </div>
        
        <SliderField
          label="Current Utilization Rate"
          description="What % of providers actively use this daily?"
          value={inputs.utilizationRate}
          onChange={(v) => setInputs({ ...inputs, utilizationRate: v })}
          min={0}
          max={100}
          hint={`Industry average for ${competitor.name}: ${competitor.typicalUtilization.min}-${competitor.typicalUtilization.max}%`}
        />
        
        <SliderField
          label="Time Saved per Encounter"
          description="How much documentation time is saved per visit?"
          value={inputs.timeSavedPerEncounter}
          onChange={(v) => setInputs({ ...inputs, timeSavedPerEncounter: v })}
          min={0}
          max={10}
          step={0.5}
          unit="min"
          hint={`${competitor.name} customers typically report ${competitor.typicalTimeSaved.min}-${competitor.typicalTimeSaved.max} min`}
        />
      </div>
      
      {/* Calculated Summary */}
      <div className="bg-gray-50 rounded-xl p-6">
        <div className="text-sm font-medium text-gray-500 uppercase tracking-wide mb-4">
          Calculated Summary
        </div>
        
        <div className="grid grid-cols-2 gap-6">
          <div>
            <div className="text-sm text-gray-500">Annual {competitor.name} Cost</div>
            <div className="text-2xl font-semibold text-gray-900">
              ${annualCost.toLocaleString()}
            </div>
            <div className="text-sm text-gray-400">
              ${inputs.monthlyCostPerProvider}/provider × {inputs.providers} × 12
            </div>
          </div>
          
          <div>
            <div className="text-sm text-gray-500">Annual Time Saved</div>
            <div className="text-2xl font-semibold text-gray-900">
              {hoursReturned.toLocaleString()} hours
            </div>
            <div className="text-sm text-gray-400">
              @ {inputs.utilizationRate}% utilization
            </div>
          </div>
        </div>
      </div>
      
      <div className="flex justify-end">
        <Button onClick={() => onComplete(inputs)}>
          Analyze the Gap →
        </Button>
      </div>
    </div>
  );
}

interface GapAnalysisProps {

}

export function GapAnalysis({ competitor, competitorInputs, selectedDrivers }: Props) {

            <div className="flex items-center justify-center mb-6">
              <img src={competitor.logo} className="h-10" alt={competitor.name} />
            </div>

          </div>

              value: `${competitorInputs.utilizationRate}%`,
              subtext: 'Your utilization'

            abridge={{
              value: '65%',

export function useGapCalculations(
  competitor: Competitor,
  inputs: UsageInputs,
  selectedDrivers: string[]
) {
  const competitorBenchmarks = COMPETITOR_ESTIMATES[competitor.id];
  const abridgeBenchmarks = ABRIDGE_BENCHMARKS;
  
  // Base calculations
  const annualEncounters = inputs.monthlyEncounters * 12;
  const competitorAnnualCost = inputs.providers * inputs.monthlyCostPerProvider * 12;
  const abridgeAnnualCost = inputs.providers * 180 * 12; // $180/provider/mo
  const costDelta = competitorAnnualCost - abridgeAnnualCost;
  
  // Driver calculations
  const patientAccess = useMemo(() => {
    if (!selectedDrivers.includes('patient_access')) return null;
    return calculatePatientAccess(
      annualEncounters,
      inputs.utilizationRate,
      inputs.timeSavedPerEncounter,
      abridgeBenchmarks.utilization.typical,
      abridgeBenchmarks.timeSavedPerEncounter.typical
    );
  }, [annualEncounters, inputs, selectedDrivers]);
  
  const overtime = useMemo(() => {
    if (!selectedDrivers.includes('overtime')) return null;
    return calculateOvertime(
      annualEncounters,
      inputs.utilizationRate,
      inputs.timeSavedPerEncounter,
      abridgeBenchmarks.utilization.typical,
      abridgeBenchmarks.timeSavedPerEncounter.typical
    );
  }, [annualEncounters, inputs, selectedDrivers]);
  
  // ... similar for other drivers
  
  // Totals
  const totalCompetitorValue = [
    patientAccess?.competitorValue,
    overtime?.competitorValue,
    levelOfService?.competitorValue,
    hcc?.competitorValue,
    denials?.competitorValue,
    retention?.competitorValue,
  ].filter(Boolean).reduce((sum, v) => sum + v, 0);
  
  const totalAbridgeValue = [
    patientAccess?.abridgeValue,
    overtime?.abridgeValue,
    levelOfService?.abridgeValue,
    hcc?.abridgeValue,
    denials?.abridgeValue,
    retention?.abridgeValue,
  ].filter(Boolean).reduce((sum, v) => sum + v, 0);
  
  const competitorNetGain = totalCompetitorValue - competitorAnnualCost;
  const abridgeNetGain = totalAbridgeValue - abridgeAnnualCost;
  
  const competitorROI = totalCompetitorValue / competitorAnnualCost;
  const abridgeROI = totalAbridgeValue / abridgeAnnualCost;
  
  // 3-year with ramp
  const threeYearAbridgeValue = (abridgeNetGain * 0.4) + (abridgeNetGain * 0.7) + abridgeNetGain;
  const threeYearCompetitorValue = competitorNetGain * 3;
  const threeYearGap = threeYearAbridgeValue - threeYearCompetitorValue;
  
  return {
    competitorAnnualCost,
    abridgeAnnualCost,
    costDelta,
    totalCompetitorValue,
    totalAbridgeValue,
    competitorNetGain,
    abridgeNetGain,
    competitorROI,
    abridgeROI,
    threeYearGap,
    driverBreakdown: [
      patientAccess && { 
        id: 'patient_access', 
        label: 'Patient Access', 
        icon: <UserPlusIcon />,
        ...patientAccess 
      },
      overtime && { 
        id: 'overtime', 
        label: 'Overtime & Locum Savings', 
        icon: <ClockIcon />,
        ...overtime 
      },
      // ... etc
    ].filter(Boolean),
  };
}
