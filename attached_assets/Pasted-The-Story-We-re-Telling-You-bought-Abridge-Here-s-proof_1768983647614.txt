Level of Service Input (Full E/M Table)
jsxCopy// ============ LEVEL OF SERVICE INPUT ============

import { useMemo } from 'react';

export default function LevelOfServiceInput({ data, setData }) {
  
  const emCodes = ['99215', '99214', '99213', '99212', '99211'];
  
  // Calculate totals
  const beforeTotal = useMemo(() => {
    return emCodes.reduce((sum, code) => sum + (data.before?.[code] || 0), 0);
  }, [data.before]);
  
  const afterTotal = useMemo(() => {
    return emCodes.reduce((sum, code) => sum + (data.after?.[code] || 0), 0);
  }, [data.after]);
  
  // Calculate weighted average (99211=1, 99212=2, etc.)
  const calculateAverage = (distribution) => {
    if (!distribution) return null;
    const weights = { '99211': 1, '99212': 2, '99213': 3, '99214': 4, '99215': 5 };
    let weighted = 0;
    let total = 0;
    emCodes.forEach(code => {
      const pct = distribution[code] || 0;
      weighted += pct * weights[code];
      total += pct;
    });
    return total > 0 ? (weighted / total).toFixed(2) : null;
  };
  
  const beforeAvg = calculateAverage(data.before);
  const afterAvg = calculateAverage(data.after);
  const avgChange = beforeAvg && afterAvg ? (((afterAvg - beforeAvg) / beforeAvg) * 100).toFixed(1) : null;
  
  const updateBefore = (code, value) => {
    setData({
      ...data,
      before: { ...data.before, [code]: value ? Number(value) : 0 }
    });
  };
  
  const updateAfter = (code, value) => {
    setData({
      ...data,
      after: { ...data.after, [code]: value ? Number(value) : 0 }
    });
  };
  
  return (
    <section className="metric-input-section">
      
      <div className="metric-input-header">
        <div className="metric-icon-large">📋</div>
        <div className="metric-title-group">
          <h2>Level of Service</h2>
          <p>E/M code distribution — Established patients</p>
        </div>
      </div>
      
      {/* E/M Input Table */}
      <div className="em-input-table">
        
        <div className="em-table-header">
          <div className="em-col-code">CODE</div>
          <div className="em-col-before">BEFORE ABRIDGE</div>
          <div className="em-col-after">AFTER ABRIDGE</div>
          <div className="em-col-change">CHANGE</div>
        </div>
        
        {emCodes.map(code => {
          const beforeVal = data.before?.[code] || 0;
          const afterVal = data.after?.[code] || 0;
          const change = afterVal - beforeVal;
          
          return (
            <div key={code} className="em-table-row">
              <div className="em-col-code">
                <span className="em-code-label">{code}</span>
              </div>
              
              <div className="em-col-before">
                <div className="em-input-wrapper">
                  <input
                    type="number"
                    min="0"
                    max="100"
                    placeholder="0"
                    value={beforeVal || ''}
                    onChange={(e) => updateBefore(code, e.target.value)}
                    className="em-input"
                  />
                  <span className="em-input-suffix">%</span>
                </div>
                <div 
                  className="em-bar before-bar"
                  style={{ width: `${beforeVal}%` }}
                ></div>
              </div>
              
              <div className="em-col-after">
                <div className="em-input-wrapper">
                  <input
                    type="number"
                    min="0"
                    max="100"
                    placeholder="0"
                    value={afterVal || ''}
                    onChange={(e) => updateAfter(code, e.target.value)}
                    className="em-input"
                  />
                  <span className="em-input-suffix">%</span>
                </div>
                <div 
                  className="em-bar after-bar"
                  style={{ width: `${afterVal}%` }}
                ></div>
              </div>
              
              <div className="em-col-change">
                {change !== 0 && (
                  <span className={`em-change ${change > 0 ? 'positive' : 'negative'}`}>
                    {change > 0 ? '↑' : '↓'} {Math.abs(change)}pp
                  </span>
                )}
                {change === 0 && <span className="em-change neutral">—</span>}
              </div>
            </div>
          );
        })}
        
        {/* Totals Row */}
        <div className="em-table-row totals-row">
          <div className="em-col-code">
            <span className="em-code-label">TOTAL</span>
          </div>
          <div className="em-col-before">
            <span className={`em-total ${beforeTotal === 100 ? 'valid' : 'invalid'}`}>
              {beforeTotal}%
              {beforeTotal !== 100 && <span className="total-hint"> (must = 100%)</span>}
            </span>
          </div>
          <div className="em-col-after">
            <span className={`em-total ${afterTotal === 100 ? 'valid' : 'invalid'}`}>
              {afterTotal}%
              {afterTotal !== 100 && <span className="total-hint"> (must = 100%)</span>}
            </span>
          </div>
          <div className="em-col-change"></div>
        </div>
        
      </div>
      
      {/* Average Level Calculation */}
      {beforeAvg && afterAvg && beforeTotal === 100 && afterTotal === 100 && (
        <div className="average-level-card">
          <h3>AVERAGE LEVEL OF SERVICE</h3>
          <div className="average-level-content">
            <div className="avg-item">
              <span className="avg-label">Before</span>
              <span className="avg-value">{beforeAvg}</span>
            </div>
            <div className="avg-arrow">→</div>
            <div className="avg-item">
              <span className="avg-label">After</span>
              <span className="avg-value">{afterAvg}</span>
            </div>
            <div className="avg-change">
              <span className="avg-change-value success">+{avgChange}%</span>
            </div>
          </div>
          <p className="average-level-note">
            Weighted average where 99211=1, 99212=2, 99213=3, 99214=4, 99215=5
          </p>
        </div>
      )}
      
      {/* Insight */}
      {data.before?.['99215'] && data.after?.['99215'] && (
        <div className="insight-callout">
          <span className="insight-icon">💡</span>
          <div className="insight-content">
            <strong>+{(data.after['99215'] - data.before['99215'])} percentage point shift to Level 5</strong>
            <p>
              This suggests more complete documentation of complex visits. 
              Not upcoding — capturing work already being done.
            </p>
          </div>
        </div>
      )}
      
    </section>
  );
}

import { useMemo } from 'react';

export default function ChartClosureInput({ data, setData }) {
  
  const buckets = [
    { id: 'within24', label: 'Within 24 hours', icon: '⚡' },
    { id: '24to48', label: '24-48 hours', icon: '📅' },
    { id: '48to72', label: '48-72 hours', icon: '📆' },
    { id: 'over72', label: '72+ hours', icon: '⏳' }
  ];
  
  // Calculate totals
  const beforeTotal = useMemo(() => {
    return buckets.reduce((sum, b) => sum + (data.before?.[b.id] || 0), 0);
  }, [data.before]);
  
  const afterTotal = useMemo(() => {
    return buckets.reduce((sum, b) => sum + (data.after?.[b.id] || 0), 0);
  }, [data.after]);
  
  const updateBefore = (bucketId, value) => {
    setData({
      ...data,
      before: { ...data.before, [bucketId]: value ? Number(value) : 0 }
    });
  };
  
  const updateAfter = (bucketId, value) => {
    setData({

  deploymentData,
  metricsData,
  calculatedROI,
  onBack
}) {
  
  const [expansionTarget, setExpansionTarget] = useState({
    providers: deploymentData.providers * 4,
    utilization: 80
  });
  
  // Calculate value per provider (from their actual data)
  const valuePerProvider = calculatedROI.totalValue / deploymentData.providers;
  
  // Generate curve data points
  const chartData = useMemo(() => {
    const points = [];
    const currentProviders = deploymentData.providers;
    const targetProviders = expansionTarget.providers;
    const steps = 20;
    
    for (let i = 0; i <= steps; i++) {
      const progress = i / steps;
      const providers = Math.round(currentProviders + (targetProviders - currentProviders) * progress);
      
      // Utilization improves as adoption matures
      const utilization = (
        deploymentData.utilizationRate + 
        (expansionTarget.utilization - deploymentData.utilizationRate) * progress
      ) / 100;
      
      // Value scales with providers and utilization improvement
      const utilizationMultiplier = utilization / (deploymentData.utilizationRate / 100);
      const value = valuePerProvider * providers * utilizationMultiplier;
      
      points.push({
        providers,
        value: Math.round(value),
        isCurrent: i === 0,
        isTarget: i === steps
      });
    }
    
    return points;
  }, [deploymentData, expansionTarget, valuePerProvider]);
