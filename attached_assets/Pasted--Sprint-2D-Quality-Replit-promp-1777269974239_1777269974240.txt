  The page was copied from ExploreWorkforce, which reads toggle state from state.timeDriverInputs. But all Revenue drivers (wrvuEnabled, hccEnabled, denialsEnabled, ipDrgEnabled, etc.) live in
  state.docQualityInputs. The toggles never read true.                                                                                                                                                         
                                                            
  REPLACE the existing isEnabled, isExpanded, toggleEnabled, toggleExpand helpers with versions that read state.docQualityInputs:                                                                              
                                                             
  const updateDocQualityInputs = (updates: Partial<typeof state.docQualityInputs>) => {                                                                                                                        
    updateState({ docQualityInputs: { ...state.docQualityInputs, ...updates } });                                                                                                                              
  };                                                        
                                                                                                                                                                                                               
  const isEnabled = (driver: ExploreDriver): boolean => {   
    return Boolean((state.docQualityInputs as any)[driver.enabledStateKey]);                                                                                                                                   
  };                                                                                                                                                                                                           
                                                             
  const isExpanded = (driver: ExploreDriver): boolean => {                                                                                                                                                     
    return driver.expandedStateKey ? Boolean((state.docQualityInputs as any)[driver.expandedStateKey]) : false;
  };                                                                                                                                                                                                           
                                                             
  const toggleEnabled = (driver: ExploreDriver) => {                                                                                                                                                           
    const current = isEnabled(driver);                       
    const updates: any = { [driver.enabledStateKey]: !current };                                                                                                                                               
    if (!current && driver.expandedStateKey) {          
      updates[driver.expandedStateKey] = true;                                                                                                                                                                 
    }                                                     
    updateDocQualityInputs(updates);                                                                                                                                                                           
  };                                                        
                                                                                                                                                                                                               
  const toggleExpand = (driver: ExploreDriver) => {         
    if (!driver.expandedStateKey) return;                                                                                                                                                                      
    updateDocQualityInputs({ [driver.expandedStateKey]: !isExpanded(driver) } as any);                                                                                                                         
  };

  const isEnabled = (driver: ExploreDriver): boolean => {                                                                                                                                                      
    const td = state.timeDriverInputs as any;
