The Math:
javascriptCopy// ============ HCC & CHRONIC CONDITION CAPTURE ============

// Inputs (user-editable)
const eligibleEncounters = 52000;          // From earlier in flow
const riskContractPercent = 0.25;          // Default 25%, adjustable
const conditionsPerVisit = 2.0;            // Default 2.0, adjustable
const documentationGap = 0.20;             // 20% discussed but not documented
const hccEligiblePercent = 0.35;           // 35% of chronic conditions are HCC-mapped
const abridgeCaptureRate = 0.40;           // 40% — must be assessed, not just mentioned
const avgHccValue = 800;                   // Blended average HCC value
const auditFactor = 0.60;                  // 60% survives RADV/validation

// Intermediate calculation (hidden from user, shown in Step 2)
const missedHccsPerEncounter = conditionsPerVisit * documentationGap * hccEligiblePercent;
// 2.0 × 0.20 × 0.35 = 0.14 missed HCCs per encounter

// Calculations

// Step 1: Risk-Based Encounters
const riskEncounters = eligibleEncounters * riskContractPercent;
// 52,000 × 0.25 = 13,000 risk encounters

// Step 2: Missed HCC Opportunities
const missedHccOpportunities = riskEncounters * missedHccsPerEncounter;
// 13,000 × 0.14 = 1,820 missed HCCs

// Step 3: Abridge Capture
const hccsCaptured = missedHccOpportunities * abridgeCaptureRate;
// 1,820 × 0.40 = 728 HCCs captured

// Step 4: Revenue Impact
const annualHccRevenue = hccsCaptured * avgHccValue * auditFactor;
// 728 × $800 × 0.60 = $349,440
