Calculations for Scribe Path
javascriptCopyfunction calculateScribeAnalysis(inputs) {
  const {
    scribeCount,
    scribeCost,
    costType,
    scribeHours,
    providersWithScribes,
    totalProviders,
    encounters,
  } = inputs;
  
  // Normalize scribe cost to annual
  const scribeSalary = costType === 'hourly' 
    ? scribeCost * scribeHours * 52 
    : scribeCost;
  
  // Current scribe program cost
  const totalScribeCost = scribeCount * scribeSalary;
  
  // Coverage
  const coveragePercent = Math.round((providersWithScribes / totalProviders) * 100);
  const unsupportedProviders = totalProviders - providersWithScribes;
  
  // Scribe ratio
  const scribeRatio = Math.round(providersWithScribes / scribeCount * 10) / 10; // e.g., 1.5:1
  
  // Cost to scale scribes to full coverage
  const scribesNeededForFull = Math.ceil(totalProviders / scribeRatio);
  const fullScribeCost = scribesNeededForFull * scribeSalary;
  
  // Abridge costs
  const abridgeCostPerProvider = 250 * 12; // $250/month
  const abridgeCostForAll = totalProviders * abridgeCostPerProvider;
  const abridgeCostForUnsupported = unsupportedProviders * abridgeCostPerProvider;
  
  // Scribes for unsupported (comparison)
  const scribesForUnsupported = Math.ceil(unsupportedProviders / scribeRatio) * scribeSalary;
  
  // Unsupported provider impact
  const encountersPerProvider = encounters / totalProviders;
  const unsupportedEncounters = unsupportedProviders * encountersPerProvider;
  const docTimePerEncounter = 12; // minutes, without help
  const unsupportedDocTime = Math.round((unsupportedEncounters * docTimePerEncounter) / 60);
  const pajamaTimeHours = Math.round(unsupportedDocTime * 0.4); // 40% done after hours
  const opportunityCost = Math.round(unsupportedDocTime * 150 * 0.15); // $150/hr, 15% could convert
  
  return {
    scribeSalary,
    totalScribeCost,
    coveragePercent,
    unsupportedProviders,
    scribeRatio,
    scribesNeededForFull,
    fullScribeCost,
    abridgeCostForAll,
    abridgeCostForUnsupported,
    scribesForUnsupported,
    unsupportedDocTime,
    pajamaTimeHours,
    opportunityCost,
  };
}
