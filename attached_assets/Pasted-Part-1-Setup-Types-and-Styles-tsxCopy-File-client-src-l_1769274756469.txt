const formatCurrency = (value: number): string => {
  if (Math.abs(value) >= 1000000) {
    return `$${(value / 1000000).toFixed(2)}M`;
  }
  if (Math.abs(value) >= 1000) {
    return `$${Math.round(value / 1000)}K`;
  }
  return `$${value.toLocaleString()}`;
};

const formatNumber = (value: number): string => {
  return value.toLocaleString();
};

const formatPercent = (value: number): string => {
  return `${value}%`;
};

const driverImplications: Record<string, (value: number, data: OutpatientPDFData) => string> = {

    return `Level-of-service improvements contribute ${formatCurrency(value)} annually—${pctOfInvestment}% of your investment. If your payer mix skews commercial (where conversion factors run $45-65 vs. Medicare's $33), actual results could be 30-50% higher.`;

  hcc_capture: (value, data) => {
    const conditions = Math.round(value / 200); // Approximate HCCs captured
    return `HCC capture adds ${formatCurrency(value)} in risk-adjusted revenue annually. This reflects ${formatNumber(conditions)} conditions that were discussed but not documented—now captured. Organizations with higher MA populations will see proportionally higher value.`;
  },
  
  denials: (value, data) => {
    const claimsRecovered = Math.round(value / 250);
    return `Denial prevention recovers ${formatCurrency(value)} annually—approximately ${formatNumber(claimsRecovered)} claims that would otherwise be written off. This value flows directly to the bottom line without additional clinical or administrative effort.`;
  },
};
