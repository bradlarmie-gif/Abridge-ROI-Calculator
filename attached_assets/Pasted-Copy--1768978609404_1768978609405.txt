Code for Replit — Updated Journey Section:
jsxCopy// ============ JOURNEY SECTION WITH AREA CHART ============

import { useState, useMemo } from 'react';
import { 
  AreaChart, 
  Area, 
  XAxis, 
  YAxis, 
  Tooltip, 
  ResponsiveContainer,
  ReferenceDot,
  ReferenceLine
} from 'recharts';

export default function JourneySection({
  initialProviders,
  initialEncountersPerProvider,
  initialUtilization,
  valuePerEncounter,
  pricePerProvider
}) {
  
  // ============ EDITABLE STATE ============
  
  // Pilot (current)
  const [pilotProviders, setPilotProviders] = useState(initialProviders);
  const [encountersPerProvider, setEncountersPerProvider] = useState(initialEncountersPerProvider);
  const [pilotUtilization, setPilotUtilization] = useState(initialUtilization);
  
  // Full Scale (opportunity)
  const [fullScaleProviders, setFullScaleProviders] = useState(initialProviders * 4);
  const [fullScaleUtilization, setFullScaleUtilization] = useState(75);
  
  // ============ CALCULATIONS ============
  
  // Generate curve data points
  const chartData = useMemo(() => {
    const points = [];
    const steps = 20; // Smooth curve with 20 points
    
    for (let i = 0; i <= steps; i++) {
      const progress = i / steps;
      
      // Providers scale linearly
      const providers = Math.round(
        pilotProviders + (fullScaleProviders - pilotProviders) * progress
      );
      
      // Utilization increases with adoption (starts at pilot, ends at full scale)
      const utilization = (
        pilotUtilization + (fullScaleUtilization - pilotUtilization) * progress
      ) / 100;
      
      // Calculate value
      const encounters = providers * encountersPerProvider * utilization;
      const value = encounters * valuePerEncounter;
      const investment = providers * pricePerProvider * 12;
      const net = value - investment;
      const roi = value / investment;
      
      points.push({
        providers,
        value: Math.round(value),
        net: Math.round(net),
        roi: roi.toFixed(1),
        utilization: Math.round(utilization * 100),
        isPilot: i === 0,
        isFullScale: i === steps
      });
    }
    
    return points;
  }, [
    pilotProviders, 
    fullScaleProviders, 
    pilotUtilization, 
    fullScaleUtilization,
    encountersPerProvider,
    valuePerEncounter,
    pricePerProvider
  ]);
  
  // Summary calculations
  const pilot = chartData[0];
  const fullScale = chartData[chartData.length - 1];
  const expansionPotential = fullScale.net - pilot.net;
  const valueMultiple = (fullScale.value / pilot.value).toFixed(1);
  
  // ============ RENDER ============
  
  return (
    <section className="journey-section">
      
      {/* Header */}
      <div className="section-header">
        <h2>Your Journey with Abridge</h2>
        <p>Start with a pilot. Prove the value. Scale across your organization.</p>
      </div>
      
      {/* Area Chart */}
      <div className="journey-chart">
        <ResponsiveContainer width="100%" height={400}>
          <AreaChart data={chartData} margin={{ top: 20, right: 30, left: 20, bottom: 20 }}>
            <defs>
              <linearGradient id="valueGradient" x1="0" y1="0" x2="1" y2="0">
                <stop offset="0%" stopColor="#f97316" stopOpacity={0.8} />
                <stop offset="50%" stopColor="#94a3b8" stopOpacity={0.6} />
                <stop offset="100%" stopColor="#059669" stopOpacity={0.8} />
              </linearGradient>
              <linearGradient id="areaFill" x1="0" y1="0" x2="0" y2="1">
                <stop offset="0%" stopColor="#059669" stopOpacity={0.3} />
                <stop offset="100%" stopColor="#059669" stopOpacity={0.05} />
              </linearGradient>
            </defs>
            
            <XAxis 
              dataKey="providers" 
              axisLine={false}
              tickLine={false}
              tick={{ fill: '#64748b', fontSize: 12 }}
              tickFormatter={(v) => `${v}`}
              label={{ 
                value: 'PROVIDERS', 
                position: 'bottom', 
                fill: '#94a3b8',
                fontSize: 11,
                fontWeight: 500
              }}
            />
            
            <YAxis 
              axisLine={false}
              tickLine={false}
              tick={{ fill: '#64748b', fontSize: 12 }}
              tickFormatter={(v) => `$${(v/1000000).toFixed(1)}M`}
              width={60}
            />
            
            <Tooltip 
              content={({ active, payload }) => {
                if (active && payload && payload.length) {
                  const data = payload[0].payload;
                  return (
                    <div className="custom-tooltip">
                      <p className="tooltip-providers">{data.providers} providers</p>
                      <p className="tooltip-utilization">{data.utilization}% utilization</p>
                      <p className="tooltip-value">${data.value.toLocaleString()} value</p>
                      <p className="tooltip-roi">{data.roi}x ROI</p>
                    </div>
                  );
                }
                return null;
              }}
            />
            
            <Area 
              type="monotone" 
              dataKey="value" 
              stroke="url(#valueGradient)"
              strokeWidth={3}
              fill="url(#areaFill)"
            />
            
            {/* Pilot marker */}
            <ReferenceDot 
              x={pilot.providers} 
              y={pilot.value} 
              r={8} 
              fill="#f97316" 
              stroke="white"
              strokeWidth={3}
            />
            
            {/* Full Scale marker */}
            <ReferenceDot 
              x={fullScale.providers} 
              y={fullScale.value} 
              r={8} 
              fill="#059669" 
              stroke="white"
              strokeWidth={3}
            />
            
          </AreaChart>
        </ResponsiveContainer>
        
        {/* Chart Labels */}
        <div className="chart-annotations">
          <div className="annotation pilot">
            <span className="annotation-dot pilot-dot"></span>
            <span className="annotation-label">You are here</span>
          </div>
          <div className="annotation fullscale">
            <span className="annotation-dot fullscale-dot"></span>
            <span className="annotation-label">Your opportunity</span>
          </div>
        </div>
      </div>
      
      {/* Model Your Scenario Panel */}
      <div className="scenario-panel">
        <h3>🎯 MODEL YOUR SCENARIO</h3>
        
        <div className="scenario-columns">
          
          {/* Pilot (Current) Column */}
          <div className="scenario-column pilot">
            <div className="column-header">
              <span className="column-icon">📍</span>
              <div className="column-titles">
                <h4>YOUR PILOT</h4>
                <p>Current state</p>
              </div>
            </div>
            
            <div className="input-group">
              <label>Providers</label>
              <input 
                type="number" 
                value={pilotProviders}
                onChange={(e) => setPilotProviders(Number(e.target.value))}
                min={1}
                max={500}
              />
            </div>
            
            <div className="input-group">
              <label>Encounters per provider</label>
              <input 
                type="number" 
                value={encountersPerProvider}
                onChange={(e) => setEncountersPerProvider(Number(e.target.value))}
                min={100}
                max={5000}
              />
            </div>
            
            <div className="input-group">
              <label>Current utilization</label>
              <div className="input-with-suffix">
                <input 
                  type="number" 
                  value={pilotUtilization}
                  onChange={(e) => setPilotUtilization(Number(e.target.value))}
                  min={10}
                  max={90}
                />
                <span className="suffix">%</span>
              </div>
            </div>
          </div>
          
          {/* Divider with Arrow */}
          <div className="scenario-divider">
            <div className="divider-line"></div>
            <div className="divider-arrow">→</div>
            <div className="divider-line"></div>
          </div>
          
          {/* Full Scale (Opportunity) Column */}
          <div className="scenario-column fullscale">
            <div className="column-header">
              <span className="column-icon">🚀</span>
              <div className="column-titles">
                <h4>YOUR OPPORTUNITY</h4>
                <p>Full scale potential</p>
              </div>
            </div>
            
            <div className="input-group">
              <label>Total providers in organization</label>
              <input 
                type="number" 
                value={fullScaleProviders}
                onChange={(e) => setFullScaleProviders(Number(e.target.value))}
                min={pilotProviders}
                max={1000}
              />
            </div>
            
            <div className="input-group">
              <label>Target utilization at full adoption</label>
              <div className="input-with-suffix">
                <input 
                  type="number" 
                  value={fullScaleUtilization}
                  onChange={(e) => setFullScaleUtilization(Number(e.target.value))}
                  min={50}
                  max={95}
                />
                <span className="suffix">%</span>
              </div>
            </div>
            
            <div className="input-note">
              <span>💡</span>
              <p>Utilization typically increases as adoption matures. Most organizations reach 70-80% at full scale.</p>
            </div>
          </div>
          
        </div>
        
        {/* Results Summary */}
        <div className="scenario-results">
          
          <div className="result-card pilot">
            <div className="result-label">PILOT (Today)</div>
            <div className="result-value">${pilot.value.toLocaleString()}</div>
            <div className="result-sublabel">annual value</div>
            <div className="result-roi">{pilot.roi}x ROI</div>
          </div>
          
          <div className="result-card fullscale">
            <div className="result-label">FULL SCALE</div>
            <div className="result-value">${fullScale.value.toLocaleString()}</div>
            <div className="result-sublabel">annual value</div>
            <div className="result-roi">{fullScale.roi}x ROI</div>
          </div>
          
          <div className="result-card expansion">
            <div className="result-label">EXPANSION POTENTIAL</div>
            <div className="result-value highlight">+${expansionPotential.toLocaleString()}</div>
            <div className="result-sublabel">additional per year</div>
            <div className="result-roi">{valueMultiple}x more value</div>
          </div>
          
        </div>
        
      </div>
      
      {/* Key Insight */}
      <div className="key-insight">
        <div className="insight-icon">💡</div>
        <div className="insight-content">
          <h4>KEY INSIGHT</h4>
          <p>
            ROI improves as you scale. Higher utilization + more providers = more value 
            per dollar invested. The pilot proves it works. Expansion captures the full opportunity.
          </p>
        </div>
      </div>
      
    </section>
  );
}
