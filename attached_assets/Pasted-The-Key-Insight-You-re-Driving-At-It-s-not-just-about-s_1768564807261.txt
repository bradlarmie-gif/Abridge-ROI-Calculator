THE CALCULATION LOGIC
typescriptCopy// State
const [showExpansion, setShowExpansion] = useState(false);
const [providers, setProviders] = useState(baselineProviders);
const [encounters, setEncounters] = useState(baselineEncounters);
const [chartView, setChartView] = useState<'current' | 'expansion'>('current');

// Scale factor
const scaleFactor = encounters / baselineEncounters;

// Current state (baseline)
const currentStateGap = selectedDrivers.reduce((sum, d) => sum + d.gap, 0);
const currentStateDax = selectedDrivers.reduce((sum, d) => sum + d.competitorValue, 0);
const currentStateAbridge = selectedDrivers.reduce((sum, d) => sum + d.abridgeValue, 0);

// Scaled state
const scaledDrivers = selectedDrivers.map(d => ({
  ...d,
  competitorValue: d.competitorValue * scaleFactor,
  abridgeValue: d.abridgeValue * scaleFactor,
  gap: d.gap * scaleFactor,
}));
const scaledGap = scaledDrivers.reduce((sum, d) => sum + d.gap, 0);

// Multi-year projections (Abridge ramps up: 20%, 50%, 85% of full value)
const currentChartData = [
  { year: 'Today', daxCumulative: 0, abridgeCumulative: 0 },
  { year: 'Year 1', daxCumulative: currentStateDax, abridgeCumulative: currentStateAbridge * 0.2 },
  { year: 'Year 2', daxCumulative: currentStateDax * 2, abridgeCumulative: (currentStateAbridge * 0.2) + (currentStateAbridge * 0.5) },
  { year: 'Year 3', daxCumulative: currentStateDax * 3, abridgeCumulative: (currentStateAbridge * 0.2) + (currentStateAbridge * 0.5) + (currentStateAbridge * 0.85) },
];

const expansionChartData = [
  { year: 'Today', daxCumulative: 0, abridgeCumulative: 0 },
  { year: 'Year 1', daxCumulative: scaledDrivers.reduce((s,d) => s + d.competitorValue, 0), abridgeCumulative: scaledDrivers.reduce((s,d) => s + d.abridgeValue, 0) * 0.2 },
  // ... similar for Y2, Y3
];

const currentThreeYearGap = currentChartData[3].abridgeCumulative - currentChartData[3].daxCumulative;
const expansionThreeYearGap = expansionChartData[3].abridgeCumulative - expansionChartData[3].daxCumulative;

Summary of the Flow

They land on Step 4 → See their current state gap immediately
Click "See what this looks like at full scale" → Expansion section appears with sliders
Adjust sliders → See the scaled gap update in real-time
Multi-year chart has toggle → Can switch between "Current Scale" and "Expansion Scale" views

This creates the "oh shit" moment when they see the expansion numbers, while keeping the current state as the anchor point.
