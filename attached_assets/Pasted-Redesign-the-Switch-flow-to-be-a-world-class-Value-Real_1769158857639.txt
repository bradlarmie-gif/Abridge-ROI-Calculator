Redesign the Switch flow to be a world-class "Value Realization Assessment" that combines gap analysis with the positioning of Abridge as a guide to help organizations realize more value. This isn't just a feature comparison — it's a diagnostic that shows where they are and what's possible.

## THE FRAME

"Every organization using ambient AI is somewhere on the value realization spectrum. Most are capturing 30-50% of potential. Top performers capture 80%+. Where do you stand, and what would it take to reach your potential?"

## THE FOUR DIMENSIONS

1. **Utilization** (Adoption)
   - Question: "What % of encounters are documented with your solution?"
   - Abridge benchmark: 75%
   - Typical competitor range: 30-50%
   - Gap = wasted investment

2. **Efficiency** (Time Saved)
   - Question: "How much time does your solution save per encounter?"
   - Abridge benchmark: 4 min (range: 3-5 min)
   - Typical competitor range: 1-2 min
   - Gap = unreturned time

3. **Quality** (Revenue Capture)
   - Question: "What wRVU lift are you seeing?"
   - Abridge benchmark: +5% (range: 3-7%)
   - Typical competitor: minimal lift
   - Gap = unrealized revenue

4. **Satisfaction** (Sustainability)
   - Question: "Would your providers recommend this solution?"
   - Abridge benchmark: 85%+ would recommend
   - Low satisfaction = future utilization drop
   - Gap = adoption risk

CALCULATIONS
javascriptCopyconst BENCHMARKS = {
  utilization: 75,
  efficiency: 4,
  wrvuLift: 5,
  satisfaction: 85,
};

function calculateRealizationScore(utilization, timeSaved, wrvuLift, satisfaction) {
  const utilizationScore = Math.min(100, Math.round((utilization / BENCHMARKS.utilization) * 100));
  const efficiencyScore = Math.min(100, Math.round((timeSaved / BENCHMARKS.efficiency) * 100));
  const qualityScore = Math.min(100, Math.round((wrvuLift / BENCHMARKS.wrvuLift) * 100));
  const satisfactionScore = Math.min(100, Math.round((satisfaction / BENCHMARKS.satisfaction) * 100));
  
  // Multiplicative
  const realizationScore = Math.round(
    (utilizationScore / 100) * 
    (efficiencyScore / 100) * 
    (qualityScore / 100) * 
    (satisfactionScore / 100) * 100
  );
  
  // Maturity level
  let maturityLevel;
  if (realizationScore < 30) maturityLevel = 'Early Stage';
  else if (realizationScore < 60) maturityLevel = 'Developing';
  else if (realizationScore < 85) maturityLevel = 'Optimized';
  else maturityLevel = 'Transformed';
  
  return {
    utilizationScore,
    efficiencyScore,
    qualityScore,
    satisfactionScore,
    realizationScore,
    maturityLevel,
  };
}

function calculateGaps(providers, encounters, utilization, timeSaved, wrvuLift) {
  // Utilization gap
  const utilizationGapPP = Math.max(0, BENCHMARKS.utilization - utilization);
  const utilizationGapEncounters = Math.round(encounters * (utilizationGapPP / 100));
  const utilizationGapValue = utilizationGapEncounters * 4;
  
  // Efficiency gap
  const efficiencyGapMin = Math.max(0, BENCHMARKS.efficiency - timeSaved);
  const encountersAtBenchmark = encounters * (BENCHMARKS.utilization / 100);
  const efficiencyGapHours = Math.round((encountersAtBenchmark * efficiencyGapMin) / 60);
  const efficiencyGapValue = Math.round(efficiencyGapHours * 150 * 0.20);
  
  // Quality gap
  const wrvuGapPercent = Math.max(0, BENCHMARKS.wrvuLift - wrvuLift);
  const avgWRVU = 1.5;
  const wrvuGapPerEnc = avgWRVU * (wrvuGapPercent / 100);
  const wrvuGapValue = Math.round(wrvuGapPerEnc * encountersAtBenchmark * 33 * 0.5);
  
  // Totals
  const annualGap = utilizationGapValue + efficiencyGapValue + wrvuGapValue;
  const monthlyGap = Math.round(annualGap / 12);
  const threeYearGap = Math.round(annualGap * 2.75);
  
  return {
    utilizationGapEncounters,
    utilizationGapValue,
    efficiencyGapMin,
    efficiencyGapHours,
    efficiencyGapValue,
    wrvuGapPercent,
    wrvuGapValue,
    annualGap,
    monthlyGap,
    threeYearGap,
    encountersAtBenchmark,
  };
}
