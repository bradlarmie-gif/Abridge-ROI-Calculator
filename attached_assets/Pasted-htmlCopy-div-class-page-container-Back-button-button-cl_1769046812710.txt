JAVASCRIPT - LIVE CALCULATION
javascriptCopyconst providersInput = document.getElementById('providers-input');
const encountersInput = document.getElementById('encounters-input');
const utilizationInputs = document.querySelectorAll('input[name="utilization"]');
const continueBtn = document.getElementById('continue-baseline');

function calculateBaseline() {
  const providers = parseInt(providersInput.value) || 0;
  const encounters = parseInt(encountersInput.value) || 0;
  const utilization = parseFloat(document.querySelector('input[name="utilization"]:checked').value) || 0.65;
  
  const eligible = Math.round(providers * encounters * utilization);
  
  // Animate the number
  animateNumber(document.getElementById('eligible-encounters'), eligible);
  
  // Update formula
  document.getElementById('formula-providers').textContent = providers.toLocaleString();
  document.getElementById('formula-encounters').textContent = encounters.toLocaleString();
  document.getElementById('formula-utilization').textContent = (utilization * 100) + '%';
  
  // Enable continue if valid
  if (providers > 0 && encounters > 0) {
    continueBtn.disabled = false;
  } else {
    continueBtn.disabled = true;
  }
}

function animateNumber(element, target) {
  const duration = 800;
  const start = parseInt(element.textContent.replace(/,/g, '')) || 0;
  const range = target - start;
  const startTime = performance.now();
  
  function update(currentTime) {
    const elapsed = currentTime - startTime;
    const progress = Math.min(elapsed / duration, 1);
    const easeOut = 1 - Math.pow(1 - progress, 3);
    const current = Math.round(start + (range * easeOut));
    
    element.textContent = current.toLocaleString();
    
    if (progress < 1) {
      requestAnimationFrame(update);
    }
  }
  
  requestAnimationFrame(update);
}

// Listen to changes
providersInput.addEventListener('input', calculateBaseline);
encountersInput.addEventListener('input', calculateBaseline);
utilizationInputs.forEach(input => {
  input.addEventListener('change', calculateBaseline);
});

// Initial calculation
calculateBaseline();
