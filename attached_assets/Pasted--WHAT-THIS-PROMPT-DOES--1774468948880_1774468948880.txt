  PART 3 — Multi-setting metric resolution logic            
                                                                                                                                                                                                               
  In measureCareSettings.ts, add a resolver function:       
                                                                                                                                                                                                               
  export function getMetricsForSettings(settings: CareSetting[]): {                                                                                                                                            
    metric: MetricDefinition;                 
    settings: CareSetting[];  // which settings this metric applies to                                                                                                                                         
    isOrgWide: boolean;       // true for burnout, likelihoodToStay
  }[] {                                                                                                                                                                                                        
    const ORG_WIDE_METRIC_IDS = ['burnoutAssessment', 'likelihoodToStay'];                                                                                                                                     
                                                                                                                                                                                                               
    const allMetrics = settings.flatMap(setting => {                                                                                                                                                           
      const metricsForSetting = getMetricsForSetting(setting); // existing function
      return metricsForSetting.map(m => ({ metric: m, setting }));                                                                                                                                             
    });                                                     
                                                                                                                                                                                                               
    // Group by metric id                                                                                                                                                                                      
    const grouped = new Map<string, { metric: MetricDefinition; settings: CareSetting[] }>();                                                                                                                  
    for (const { metric, setting } of allMetrics) {                                                                                                                                                            
      if (!grouped.has(metric.id)) {                                                                                                                                                                           
        grouped.set(metric.id, { metric, settings: [] });   
      }                                                                                                                                                                                                        
      grouped.get(metric.id)!.settings.push(setting);
    }                                                                                                                                                                                                          
                                                            
    return Array.from(grouped.values()).map(({ metric, settings }) => ({                                                                                                                                       
      metric,                                               
      settings,
      isOrgWide: ORG_WIDE_METRIC_IDS.includes(metric.id),                                                                                                                                                      
    }));
  }                                                                                                                                                                                                            
                                                            
  For org-wide metrics, always render a single input regardless of how many settings are active.
  For per-setting metrics present in multiple settings, render one input row per setting.
  For setting-specific metrics (LWBS, LOS, etc.), render once with the setting tagged.
