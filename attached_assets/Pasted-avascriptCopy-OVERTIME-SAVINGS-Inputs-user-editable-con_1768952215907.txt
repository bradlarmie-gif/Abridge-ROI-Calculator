const otProviders = 40;                    // Total providers
const otPercentWithOT = 0.60;              // % who regularly work OT
const otHoursPerWeek = 4;                  // Avg OT hours per week
const otWeeksPerYear = 50;                 // Working weeks
const otReductionRate = 0.70;              // Abridge impact on documentation time
const otConversionRate = 0.33;             // % of saved time → actual OT reduction
const physicianHourlyRate = 150;           // Fully-loaded physician labor cost

// Calculations
const annualOTHours = otProviders * otPercentWithOT * otHoursPerWeek * otWeeksPerYear;
// 40 × 0.60 × 4 × 50 = 4,800 hrs

const otHoursEliminated = annualOTHours * otReductionRate;
// 4,800 × 0.70 = 3,360 hrs

const otHoursConvertedToSavings = otHoursEliminated * otConversionRate;
// 3,360 × 0.33 = 1,109 hrs

const annualOTSavings = otHoursConvertedToSavings * physicianHourlyRate;
// 1,109 × $150 = $166,320
