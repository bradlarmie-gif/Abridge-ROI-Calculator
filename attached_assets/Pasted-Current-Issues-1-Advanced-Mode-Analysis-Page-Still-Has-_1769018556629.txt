jsxCopy// ============ ADVANCED MODE: METRIC COMPARISON PAGE ============
// File: SwitchAdvancedComparison.tsx

import { ArrowLeft, ArrowRight, Sparkles, TrendingUp, TrendingDown } from 'lucide-react';
import { formatNumber } from '../utils/formatting';

export default function SwitchAdvancedComparison({
  inputs,
  benchmarks,
  currentStep,
  totalSteps,
  onNext,
  onBack
}) {
  
  // Build comparison data for each selected metric
  const comparisons = [];
  
  // Time Savings
  if (inputs.selectedMetrics?.includes('timeSavings') && inputs.metricValues?.timeSavings) {
    const current = inputs.metricValues.timeSavings;
    const benchmark = 4; // Abridge benchmark
    const gap = benchmark - current;
    const percentOfBenchmark = Math.round((current / benchmark) * 100);
    
    comparisons.push({
      id: 'timeSavings',
      name: 'Time Savings',
      description: 'Minutes saved per encounter',
      current: `${current} min`,
      benchmark: `${benchmark} min`,
      gap: `+${gap.toFixed(1)} min`,
      percentOfBenchmark,
      direction: 'up', // Higher is better
      impact: `+${formatNumber(Math.round(gap * inputs.totalEncounters * 0.65 / 60))} hours/year`
    });
  }
  
  // Work Outside of Work
  if (inputs.selectedMetrics?.includes('workOutsideWork') && inputs.metricValues?.workOutsideWork) {
    const current = inputs.metricValues.workOutsideWork;
    const benchmark = 2; // Abridge benchmark (lower is better)
    const gap = current - benchmark;
    const percentOfBenchmark = Math.round((benchmark / current) * 100);
    
    comparisons.push({
      id: 'workOutsideWork',
      name: 'Work Outside of Work',
      description: 'Hours worked after scheduled time',
      current: `${current} hrs/week`,
      benchmark: `${benchmark} hrs/week`,
      gap: `-${gap.toFixed(1)} hrs/week`,
      percentOfBenchmark,
      direction: 'down', // Lower is better
      impact: `-${formatNumber(Math.round(gap * inputs.providers * 48))} hours/year`
    });
  }
  
  // wRVU Capture
  if (inputs.selectedMetrics?.includes('wrvuLift') && inputs.metricValues?.wrvuLift !== undefined) {
    const current = inputs.metricValues.wrvuLift;
    const benchmark = 5; // Abridge benchmark
    const gap = benchmark - current;
    const percentOfBenchmark = Math.round((current / benchmark) * 100);
    
    comparisons.push({
      id: 'wrvuLift',
      name: 'wRVU Capture',
      description: '% improvement in wRVU capture',
      current: `${current}% lift`,
      benchmark: `${benchmark}% lift`,
      gap: `+${gap}pp`,
      percentOfBenchmark,
      direction: 'up',
      impact: `+${formatNumber(Math.round(inputs.totalEncounters * 0.65 * 1.5 * (gap / 100)))} wRVUs/year`
    });
  }
  
  // Chart Closure
  if (inputs.selectedMetrics?.includes('chartClosure24h') && inputs.metricValues?.chartClosure24h) {
    const current = inputs.metricValues.chartClosure24h;
    const benchmark = 78; // Abridge benchmark
    const gap = benchmark - current;
    const percentOfBenchmark = Math.round((current / benchmark) * 100);
    
    comparisons.push({
      id: 'chartClosure24h',
      name: 'Chart Closure (24h)',
      description: '% of charts closed within 24 hours',
      current: `${current}%`,
      benchmark: `${benchmark}%`,
      gap: `+${gap}pp`,
      percentOfBenchmark,
      direction: 'up',
      impact: 'Faster billing cycles'
    });
  }
  
  // Calculate overall capture rate
  const avgCaptureRate = comparisons.length > 0
    ? Math.round(comparisons.reduce((sum, c) => sum + c.percentOfBenchmark, 0) / comparisons.length)
    : 0;
  
  return (
    <div className="switch-page">
      
      {/* Header */}
      <header className="page-header">
        <button className="back-button" onClick={onBack}>
          <ArrowLeft size={16} />
          <span>Back</span>
        </button>
        <div className="header-brand">
          <span className="brand-mark">A</span>
          <span className="brand-label">SWITCH</span>
        </div>
        <div className="progress-indicator">
          {Array.from({ length: totalSteps }, (_, i) => (
            <span 
              key={i} 
              className={`progress-dot ${i < currentStep ? 'complete' : ''} ${i === currentStep - 1 ? 'active' : ''}`}
            />
          ))}
        </div>
      </header>
      
      <main className="page-content wide">
        
        <div className="page-title">
          <h1>How you compare</h1>
          <p className="page-subtitle">
            Your current metrics vs. Abridge customer benchmarks
          </p>
        </div>
        
        {/* Overall Summary */}
        <section className="overall-summary">
          <div className="summary-card">
            <span className="summary-label">OVERALL BENCHMARK ACHIEVEMENT</span>
            <div className="summary-gauge">
              <div className="gauge-track">
                <div 
                  className="gauge-fill"
                  style={{ width: `${avgCaptureRate}%` }}
                />
                <div 
                  className="gauge-marker"
                  style={{ left: `${avgCaptureRate}%` }}
                >
                  <span className="marker-value">{avgCaptureRate}%</span>
                </div>
              </div>
              <div className="gauge-labels">
                <span>0%</span>
                <span>Abridge Benchmark</span>
                <span>100%</span>
              </div>
            </div>
            <p className="summary-insight">
              Across the metrics you selected, you're achieving <strong>{avgCaptureRate}%</strong> of 
              Abridge customer benchmarks.
            </p>
          </div>
        </section>
        
        {/* Metric Comparisons */}
        <section className="comparisons-section">
          <h2 className="section-label">METRIC-BY-METRIC COMPARISON</h2>
          
          <div className="comparisons-list">
            {comparisons.map((comparison) => (
              <div key={comparison.id} className="comparison-card">
                
                <div className="comparison-header">
                  <div className="comparison-title">
                    <h3>{comparison.name}</h3>
                    <p>{comparison.description}</p>
                  </div>
                  <div className={`comparison-badge ${comparison.percentOfBenchmark >= 80 ? 'good' : comparison.percentOfBenchmark >= 50 ? 'moderate' : 'needs-improvement'}`}>
                    {comparison.percentOfBenchmark >= 80 ? 'Near benchmark' : comparison.percentOfBenchmark >= 50 ? 'Room to improve' : 'Significant gap'}
                  </div>
                </div>
                
                <div className="comparison-values">
                  
                  <div className="value-column current">
                    <span className="value-label">YOUR CURRENT</span>
                    <span className="value-number">{comparison.current}</span>
                  </div>
                  
                  <div className="value-arrow">
                    <span className="arrow-line"></span>
                    <div className={`arrow-gap ${comparison.direction}`}>
                      {comparison.direction === 'up' ? <TrendingUp size={14} /> : <TrendingDown size={14} />}
                      <span>{comparison.gap}</span>
                    </div>
                    <span className="arrow-head">→</span>
                  </div>
                  
                  <div className="value-column benchmark">
                    <span className="value-label">
                      <Sparkles size={12} />
                      ABRIDGE BENCHMARK
                    </span>
                    <span className="value-number">{comparison.benchmark}</span>
                  </div>
                  
                </div>
                
                <div className="comparison-progress">
                  <div className="progress-track">
                    <div 
                      className="progress-fill"
                      style={{ width: `${comparison.percentOfBenchmark}%` }}
                    />
                  </div>
                  <span className="progress-label">
                    {comparison.percentOfBenchmark}% of benchmark
                  </span>
                </div>
                
                <div className="comparison-impact">
                  <span className="impact-label">Closing this gap means:</span>
                  <span className="impact-value">{comparison.impact}</span>
                </div>
                
              </div>
            ))}
          </div>
        </section>
        
        {/* What This Means */}
        <section className="meaning-section">
          <div className="meaning-card">
            <p>
              These gaps represent <strong>unrealized value</strong> — not because your current solution 
              is bad, but because small differences across multiple dimensions compound into significant 
              overall impact.
            </p>
            <p>
              Let's quantify what this means in dollars.
            </p>
          </div>
        </section>
        
      </main>
      
      {/* Footer */}
      <footer className="page-footer">
        <button className="continue-button enabled" onClick={onNext}>
          <span>See the value breakdown</span>
          <ArrowRight size={16} />
        </button>
      </footer>
      
    </div>
  );
}

Updated Advanced Mode Flow:
jsxCopy// ============ ADVANCED MODE FLOW ============

export default function SwitchAdvancedFlow({ onBack }) {
  const [currentStep, setCurrentStep] = useState(1);
  const [inputs, setInputs] = useState({ /* ... */ });
  
  const totalSteps = 4; // Setup → Comparison → Analysis → Conclusion
  
  return (
    <>
      {currentStep === 1 && (
        <SwitchAdvancedSetup 
          inputs={inputs}
          setInputs={setInputs}
          onNext={() => setCurrentStep(2)}
          onBack={onBack}
          currentStep={1}
          totalSteps={totalSteps}
        />
      )}
      
      {currentStep === 2 && (
        <SwitchAdvancedComparison 
          inputs={inputs}
          benchmarks={benchmarks}
          onNext={() => setCurrentStep(3)}
          onBack={() => setCurrentStep(1)}
          currentStep={2}
          totalSteps={totalSteps}
        />
      )}
      
      {currentStep === 3 && (
        <SwitchAdvancedAnalysis 
          inputs={inputs}
          calculations={calculations}
          onNext={() => setCurrentStep(4)}
          onBack={() => setCurrentStep(2)}
          currentStep={3}
          totalSteps={totalSteps}
        />
      )}
      
      {currentStep === 4 && (
        <SwitchAmbientConclusion 
          inputs={inputs}
          calculations={calculations}
          benchmarks={benchmarks}
          mode="advanced"
          onBack={() => setCurrentStep(3)}
          currentStep={4}
          totalSteps={totalSteps}
        />
      )}
    </>
  );
}
