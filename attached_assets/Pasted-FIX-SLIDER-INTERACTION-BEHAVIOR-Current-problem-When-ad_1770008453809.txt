Implementation hint:
```javascript
function redistributeSliders(changedIndex, newValue, currentValues) {
  const values = [...currentValues]
  const oldValue = values[changedIndex]
  const delta = newValue - oldValue
  
  // Get indices of other sliders
  const otherIndices = values.map((_, i) => i).filter(i => i !== changedIndex)
  
  // Calculate total of other sliders
  const otherTotal = otherIndices.reduce((sum, i) => sum + values[i], 0)
  
  // Redistribute proportionally
  otherIndices.forEach(i => {
    const proportion = values[i] / otherTotal
    values[i] = Math.max(0, values[i] - (delta * proportion))
  })
  
  values[changedIndex] = newValue
  
  // Normalize to exactly 100%
  const sum = values.reduce((a, b) => a + b, 0)
  return values.map(v => (v / sum) * 100)
}
