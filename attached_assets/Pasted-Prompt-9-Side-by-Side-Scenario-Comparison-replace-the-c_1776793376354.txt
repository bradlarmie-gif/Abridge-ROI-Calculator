  KPI ROWS COMPONENT — the heart of the comparison:                                                                                                                                                            
                                                                                                                                                                                                               
  function KPIRows({
    result,                                                                                                                                                                                                    
    baseline,                                           
  }: {
    result: ForecastResult;
    baseline: ForecastResult | null; // null = this IS the baseline (current)
  }) {                                                                                                                                                                                                         
    const rows = [
      {                                                                                                                                                                                                        
        label: 'Total Cost',                                                                                                                                                                                   
        current: result.kpis.totalContractCost,
        fmt: fmtCurrencyShort,                                                                                                                                                                                 
        lowerIsBetter: true,                                                                                                                                                                                   
      },
      {                                                                                                                                                                                                        
        label: 'Net Value',                             
        current: result.kpis.netContractValue,
        fmt: fmtCurrencyShort,                                                                                                                                                                                 
        lowerIsBetter: false,
      },                                                                                                                                                                                                       
      {                                                 
        label: 'ROI',
        current: result.kpis.roiMultiple,
        fmt: (n: number) => `${n.toFixed(2)}x`,                                                                                                                                                                
        lowerIsBetter: false,                                                                                                                                                                                  
      },                                                                                                                                                                                                       
      {                                                                                                                                                                                                        
        label: 'Break-Even',                            
        current: result.kpis.fullBreakEvenMonth,
        fmt: (n: number | null) => n ? `Month ${n}` : 'Not reached',                                                                                                                                           
        lowerIsBetter: true, // earlier is better                                                                                                                                                              
        isMonth: true,                                                                                                                                                                                         
      },                                                                                                                                                                                                       
                                                                                                                                                                                                               
    return (
      <div className="space-y-3 border-t border-white/10 pt-4">                                                                                                                                                
        {rows.map((row) => {                                   
          const val = row.current;
          const baseVal = baseline ? (row.label === 'Total Cost' ? baseline.kpis.totalContractCost : row.label === 'Net Value' ? baseline.kpis.netContractValue : row.label === 'ROI' ?                        
  baseline.kpis.roiMultiple : baseline.kpis.fullBreakEvenMonth) : null;                                                                                                                
                                                                                                                                                                                                               
          let delta: number | null = null;              
          let deltaStr = '';                                                                                                                                                                                   
          let deltaColor = 'text-neutral-400';          
                                                                                                                                                                                                               
          if (baseline !== null && baseVal !== null && val !== null && typeof val === 'number' && typeof baseVal === 'number') {                                                                               
            delta = val - baseVal;                                                                                                                                                                             
            const isGood = row.lowerIsBetter ? delta < 0 : delta > 0;                                                                                                                                          
            const isBad = row.lowerIsBetter ? delta > 0 : delta < 0;                                                                                                                                           
                                                                                                                                                                                                               
            if (row.isMonth) {                                                                                                                                                                                 
              deltaStr = delta === 0 ? '=' : delta < 0 ? `${Math.abs(delta)}mo earlier` : `${delta}mo later`;                                                                                                  
            } else if (row.label === 'ROI') {                                                                                                                                                                  
              deltaStr = delta === 0 ? '=' : `${delta > 0 ? '+' : ''}${delta.toFixed(2)}x`;                                                                                                                    
            } else {                                                                                                                                                                                           
              deltaStr = delta === 0 ? '=' : `${delta > 0 ? '+' : ''}${fmtCurrencyShort(Math.abs(delta))}`;                                                                                                    
            }                                                                                                                                                                                                  
                                                                                                                                                                                                               
            deltaColor = isGood ? 'text-emerald-600' : isBad ? 'text-red-500' : 'text-neutral-400';
          }                                                                                                                                                                                                    
                                                        
          return (                                                                                                                                                                                             
            <div key={row.label} className="flex items-center justify-between">                                                                                                                                
              <span className={`text-[10px] uppercase tracking-wide ${baseline === null ? 'text-white/50' : 'text-neutral-400'} font-semibold`}>
                {row.label}                                                                                                                                                                                    
              </span>                                                                                                                                                                                          
              <div className="flex items-center gap-2">
                {deltaStr && (                                                                                                                                                                                 
                  <span className={`text-[11px] font-semibold ${deltaColor}`}>                                                                                                                                 
                    {deltaStr}                                                
                  </span>                                                                                                                                                                                      
                )}                                                                                                                                                                                             
                <span className={`text-sm font-bold font-abridge ${baseline === null ? 'text-white' : 'text-[#1A1A1A]'}`}>
                  {val === null || val === undefined                                                                                                                                                           
                    ? '—'                                                                                                                                                                                      
                    : row.fmt(val as any)}                                                                                                                                                                     
                </span>                                                                                                                                                                                        
              </div>                                                                                                                                                                                           
            </div>                                      
          );      
        })} 
      </div>
    );                                                                                                                                                                                                         
  }   
                                                                                                                                                                                                               
  ALTERNATIVE CALCULATION: For each alternative, derive a modified ForecastState and run calculateForecast():
  function buildAltState(base: ForecastState, alt: ComparisonColumn): ForecastState {                                                                                                                          
    const adoptionCurve = {                                                          
      ...base.adoptionCurve,                                                                                                                                                                                   
      startPct: Math.min(95, base.adoptionCurve.startPct * alt.adoptionMultiplier),                                                                                                                            
      endPct: Math.min(98, base.adoptionCurve.endPct * alt.adoptionMultiplier),                                                                                                                                
    };                                                                                                                                                                                                         
    return {                                                                                                                                                                                                   
      ...base,                                                                                                                                                                                                 
      currentPricing: alt.pricing,                                                                                                                                                                             
      adoptionCurve,                                    
    };                                                                                                                                                                                                         
  }
