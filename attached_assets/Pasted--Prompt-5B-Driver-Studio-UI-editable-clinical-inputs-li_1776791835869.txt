  ```typescript                                                                                                                                                                                                
  export function recalculateProjectedDelta(driver: ForecastValueDriver): number {
    const ci = driver.clinicalInputs;                                                                                                                                                                          
    if (!ci || !ci.formulaType) return driver.projectedDelta;
                                                                                                                                                                                                               
    const delta = (ci.metricAfter ?? 0) - (ci.metricBefore ?? 0);
    const absDelta = Math.abs(delta);                                                                                                                                                                          
    const allocPct = (ci.allocationPct ?? 100) / 100;                                                                                                                                                          
  
    switch (ci.formulaType) {                                                                                                                                                                                  
      case 'timeSavingsWorkforce': {                    
        // (minutesDelta / 60) * hourlyRate * allocationPct                                                                                                                                                    
        return (absDelta / 60) * (ci.factor1Value ?? 150) * allocPct;                                                                                                                                          
      }                                                                                                                                                                                                        
      case 'timeSavingsCapacity': {                                                                                                                                                                            
        // (minutesDelta / 60 / minutesPerVisit) * revenuePerVisit * allocationPct                                                                                                                             
        const minPerVisit = Math.max(ci.factor1Value ?? 30, 1);                                                                                                                                                
        const revPerVisit = ci.factor2Value ?? 200;
        return (absDelta / 60 / minPerVisit) * revPerVisit * allocPct;                                                                                                                                         
      }                                                 
      case 'wrvuLift': {                                                                                                                                                                                       
        // delta * conversionFactor                                                                                                                                                                            
        return absDelta * (ci.factor1Value ?? 33);
      }                                                                                                                                                                                                        
      case 'emLevelLift': {                             
        // delta * valuePerLevel                                                                                                                                                                               
        return absDelta * (ci.factor1Value ?? 15);
      }                                                                                                                                                                                                        
      case 'cmiLift': {                                 
        // delta * valuePerPoint
        return absDelta * (ci.factor1Value ?? 1500);                                                                                                                                                           
      }
      case 'retentionLift': {                                                                                                                                                                                  
        // fixed formula: 3% * replacementCost / 12 (monthly per user)
        return 0.03 * (ci.factor1Value ?? 150000) / 12;                                                                                                                                                        
      }                                                                                                                                                                                                        
      case 'hccCapture': {                                                                                                                                                                                     
        // captureRateDelta * rafValue                                                                                                                                                                         
        return (absDelta / 100) * (ci.factor1Value ?? 1200);                                                                                                                                                   
      }
      case 'denialReduction': {                                                                                                                                                                                
        // denialRateDelta * avgClaimValue                                                                                                                                                                     
        return (absDelta / 100) * (ci.factor1Value ?? 350);
      }                                                                                                                                                                                                        
      case 'nursingOvertimeReduction': {                
        // hoursReduced * hourlyRate * 1.5                                                                                                                                                                     
        return absDelta * (ci.factor1Value ?? 50) * 1.5;
      }                                                                                                                                                                                                        
      case 'losReduction': {                            
        // daysReduced * dailyCost                                                                                                                                                                             
        return absDelta * (ci.factor1Value ?? 2800);    
      }
      default:
        return driver.projectedDelta;
    }
  }
