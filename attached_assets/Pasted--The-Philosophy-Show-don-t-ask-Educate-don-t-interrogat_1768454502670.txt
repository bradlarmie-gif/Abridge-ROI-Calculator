const NewCareSettingFlow = ({ baseline, onBack, onSave }) => {
  const [selectedSetting, setSelectedSetting] = useState(null);
  const [settingConfig, setSettingConfig] = useState({
    providers: 20,
    encounters: null, // null = auto-calculate
    customEncounters: false,
    utilization: baseline.utilization || 0.65
  });
  
  const [model, setModel] = useState(null);
  
  // Recalculate instantly when anything changes
  useEffect(() => {
    if (selectedSetting) {
      const calculated = calculateCombinedDeployment(baseline, {
        type: selectedSetting,
        ...settingConfig
      });
      setModel(calculated);
    }
  }, [selectedSetting, settingConfig, baseline]);
  
  if (!selectedSetting) {
    return (
      <CareSettingExplorer 
        baseline={baseline}
        onSelect={setSelectedSetting} 
        onBack={onBack} 
      />
    );
  }
  
  return (
    <CombinedPreview 
      baseline={baseline}
      settingType={selectedSetting}
      config={settingConfig}
      onConfigChange={setSettingConfig}
      model={model}
      onBack={() => setSelectedSetting(null)}
      onSave={onSave}
    />
  );
};

javascriptCopyexport const calculateCombinedDeployment = (baseline, newSetting) => {
  const settingDefaults = getSettingDefaults(newSetting.type);
  
  // NEW SETTING CALCULATIONS
  const newProviders = newSetting.providers;
  const newEncounters = newSetting.encounters || (newProviders * settingDefaults.encountersPerProvider);
  const newUtilization = newSetting.utilization || baseline.utilization || 0.65;
  const newCost = newProviders * (baseline.costPerProviderMonth || 120) * 12;
  
  // Calculate benefits for new setting
  const newBenefits = {};
  settingDefaults.primaryDrivers.forEach(driver => {
    newBenefits[driver] = calculateDriverValue(
      driver,
      newProviders,
      newEncounters,
      newUtilization,
      settingDefaults.metrics,
      baseline
    );
  });
  
  const newTotalBenefit = Object.values(newBenefits).reduce((sum, val) => sum + val, 0);
  const newNetGain = newTotalBenefit - newCost;
  const newROI = newTotalBenefit / newCost;
  
  // COMBINED CALCULATIONS
  const combinedProviders = baseline.providers + newProviders;
  const combinedEncounters = baseline.encounters + newEncounters;
  const combinedCost = (baseline.annualCost || 0) + newCost;
  const combinedBenefit = baseline.totalBenefit + newTotalBenefit;
  
  // Merge benefit breakdowns
  const combinedBenefitsByDriver = { ...baseline.benefits };
  Object.entries(newBenefits).forEach(([driver, value]) => {
    combinedBenefitsByDriver[driver] = (combinedBenefitsByDriver[driver] || 0) + value;
  });
  
  const combinedNetGain = combinedBenefit - combinedCost;
  const combinedROI = combinedBenefit / combinedCost;
  
  // Generate insight
  const insight = generateInsight(baseline, newSetting, newROI, combinedNetGain, settingDefaults);
  
  return {
    baseline: {
      providers: baseline.providers,
      encounters: baseline.encounters,
      cost: baseline.annualCost || 0,
      benefit: baseline.totalBenefit,
      netGain: baseline.netGain || (baseline.totalBenefit - (baseline.annualCost || 0)),
      roi: baseline.roi
    },
    newSetting: {
      type: newSetting.type,
      providers: newProviders,
      encounters: newEncounters,
      utilization: newUtilization,
      cost: newCost,
      benefits: newBenefits,
      totalBenefit: newTotalBenefit,
      netGain: newNetGain,
      roi: newROI
    },
    combined: {
      providers: combinedProviders,
      encounters: combinedEncounters,
      cost: combinedCost,
      benefit: combinedBenefit,
      benefitsByDriver: combinedBenefitsByDriver,
      netGain: combinedNetGain,
      roi: combinedROI,
      threeYearValue: combinedNetGain * 3
    },
    deltas: {
      netGainIncrease: (((combinedNetGain - baseline.netGain) / baseline.netGain) * 100).toFixed(0)
    },
    insight
  };
};

function getSettingDefaults(settingType) {
  const defaults = {
    ed: {
      encountersPerProvider: 5000,
      avgWRVUPerEncounter: 2.8,
      primaryDrivers: ['throughput', 'los', 'denials'],
      metrics: {
        revenuePerVisit: 200,
        revenuePerWRVU: 45,
        denialRate: 0.08,
        avgDenialValue: 500
      }
    },
    nursing: {
      encountersPerProvider: 2000,
      primaryDrivers: ['timeSavings', 'overtime', 'quality'],
      metrics: {}
    },
    inpatient: {
      encountersPerProvider: 800,
      avgWRVUPerEncounter: 3.5,
      primaryDrivers: ['hcc', 'los', 'denials'],
      metrics: {
        revenuePerWRVU: 50,
        maPopulation: 0.25,
        denialRate: 0.06
      }
    }
  };
  
  return defaults[settingType];
}

function calculateDriverValue(driver, providers, encounters, utilization, metrics, baseline) {
  const eligibleEncounters = Math.round(encounters * utilization);
  
  switch(driver) {
    case 'throughput':
    case 'access':
      // Additional visits per week × weeks × revenue
      const visitsPerWeek = 2.5;
      const revenuePerVisit = metrics.revenuePerVisit || baseline.revenuePerVisit || 200;
      return Math.round(providers * visitsPerWeek * 48 * revenuePerVisit);
    
    case 'los':
      // wRVU uplift × wRVU rate
      const wrvuUplift = metrics.avgWRVUPerEncounter ? eligibleEncounters * 0.05 : 0; // 5% uplift
      const wrvuRate = metrics.revenuePerWRVU || baseline.revenuePerWRVU || 40;
      return Math.round(wrvuUplift * wrvuRate);
    
    case 'denials':
      // Encounters × denial rate × reduction % × avg value
      const denialRate = metrics.denialRate || 0.05;
      const reductionPercent = 0.20;
      const avgDenialValue = metrics.avgDenialValue || 500;
      return Math.round(eligibleEncounters * denialRate * reductionPercent * avgDenialValue);
    
    case 'hcc':
      // MA patients × conditions captured × RAF value
      const maPopulation = metrics.maPopulation || 0.15;
      const maPatients = Math.round((eligibleEncounters / 2.5) * maPopulation);
      const conditionsPerPatient = 0.15;
      const rafValue = 3000;
      return Math.round(maPatients * conditionsPerPatient * rafValue);
    
    default:
      return 0;
  }
}

function generateInsight(baseline, newSetting, newROI, combinedNetGain, settingDefaults) {
  const settingNames = {
    ed: 'ED',
    nursing: 'Nursing',
    inpatient: 'Inpatient'
  };
  
  const name = settingNames[newSetting.type];
  const gainIncrease = (((combinedNetGain - baseline.netGain) / baseline.netGain) * 100).toFixed(0);
  
  if (newROI > baseline.roi) {
    return `Adding ${name} increases your net gain by ${gainIncrease}% while achieving higher ROI (${newROI.toFixed(1)}x vs ${baseline.roi.toFixed(1)}x). ${name}'s higher ${settingDefaults.avgWRVUPerEncounter ? 'wRVU density' : 'efficiency gains'} drives superior value.`;
  } else {
    return `Adding ${name} increases your net gain by ${gainIncrease}% with solid ${newROI.toFixed(1)}x ROI. This expands your deployment while maintaining strong returns.`;
  }
}
