  Change 1 — Add emEligibilityRate to the wRVU billing capture calculation
                                                                                                                                                                                                               
  In measureCalculator.ts, find the wRVU billing capture formula:                                         
  billingCaptureLow += wrvuDelta * adoptedEncounters * conversionFactor * attributionLow * realizationLow;                                                                                                     
  billingCaptureHigh += wrvuDelta * adoptedEncounters * conversionFactor * attributionHigh * realizationHigh;                                                                                                  
                                                                                                             
  Add emEligibilityRate as a multiplier:                                                                                                                                                                       
  const emEligibilityRate = inputs.emEligibilityRate ?? 0.80; // default 80%                              
  const emEligibleEncounters = Math.round(adoptedEncounters * emEligibilityRate);                                                                                                                              
                                                                                                          
  billingCaptureLow += wrvuDelta * emEligibleEncounters * conversionFactor * attributionLow * realizationLow;                                                                                                  
  billingCaptureHigh += wrvuDelta * emEligibleEncounters * conversionFactor * attributionHigh * realizationHigh;
                                                                                                                                                                                                               
  Apply the same emEligibilityRate to the E/M level improvement calculation:                                                                                                                                   
  billingCaptureLow += emDelta * emEligibleEncounters * undercaptureRate * avgEmValue * attributionLow * realizationLow;

  const emEligibilityDefaults: Record<string, number> = {
