FORMULAS:
═══════════════════════════════════════════════════════════

// SETUP
patientDaysPerYear = staffedBeds × occupancyRate × 365
totalShiftsPerYear = nurseFTEs × 156
eligibleShifts = totalShiftsPerYear × utilizationRate
totalHoursSaved = (eligibleShifts × minutesPerShift) / 60

// STAFFING EFFICIENCY
hoursToEfficiency = totalHoursSaved × efficiencyAllocation
currentAnnualOT = nurseFTEs × otHoursPerWeek × 50
docDrivenOT = currentAnnualOT × docDrivenPercent
hoursEliminated = Math.min(hoursToEfficiency, docDrivenOT)
efficiencyValue = hoursEliminated × otHourlyRate

// NURSE WELLBEING
hoursToWellbeing = totalHoursSaved × wellbeingAllocation
hoursPerNursePerYear = hoursToWellbeing / nurseFTEs

// Threshold logic
if (hoursPerNursePerYear < 25) retentionLift = 0.05
else if (hoursPerNursePerYear < 50) retentionLift = 0.10
else retentionLift = 0.17

annualDepartures = nurseFTEs × turnoverRate
burnoutDepartures = annualDepartures × burnoutPercent
nursesRetained = burnoutDepartures × retentionLift
retentionValue = nursesRetained × replacementCost

agencyFTEs = staffedBeds × agencyRatio
agencyPremium = agencyCost - staffSalary
agencyReduction = agencyFTEs × 0.10 × retentionLift × 2
agencyValue = agencyReduction × agencyPremium

wellbeingValue = retentionValue + agencyValue

// CARE QUALITY
hoursToQuality = totalHoursSaved × qualityAllocation

annualFalls = (patientDaysPerYear / 1000) × fallsRate
preventableFalls = annualFalls × fallsPreventionPercent
fallsValue = preventableFalls × costPerFall

annualHAPIs = (patientDaysPerYear / 1000) × hapiRate
preventableHAPIs = annualHAPIs × hapiPreventionPercent
hapiValue = preventableHAPIs × costPerHAPI

qualityGross = fallsValue + hapiValue
qualityValue = qualityGross × realizationRate

// TOTAL
directValue = efficiencyValue + wellbeingValue
potentialValue = qualityValue
totalValue = directValue + potentialValue
