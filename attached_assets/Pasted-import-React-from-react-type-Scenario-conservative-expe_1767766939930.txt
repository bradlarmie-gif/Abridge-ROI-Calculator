import React from "react";

type Scenario = "conservative" | "expected" | "optimistic" | "custom";

function Pill({
  label,
  selected,
  onClick,
}: {
  label: string;
  selected?: boolean;
  onClick?: () => void;
}) {
  return (
    <button
      type="button"
      onClick={onClick}
      className={[
        "inline-flex items-center justify-center rounded-full px-3 py-1.5 text-sm font-medium transition",
        "border",
        selected
          ? "border-neutral-900 bg-neutral-900 text-white"
          : "border-neutral-200 bg-white text-neutral-700 hover:bg-neutral-50",
      ].join(" ")}
    >
      {label}
    </button>
  );
}

function StatRow({
  label,
  value,
}: {
  label: string;
  value: React.ReactNode;
}) {
  return (
    <div className="flex items-center justify-between gap-4 py-2">
      <div className="text-sm text-neutral-600">{label}</div>
      <div className="text-sm font-semibold text-neutral-900 tabular-nums">
        {value}
      </div>
    </div>
  );
}
