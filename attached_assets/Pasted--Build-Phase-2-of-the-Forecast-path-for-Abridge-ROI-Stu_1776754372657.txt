  ▎ 3. Calculation Engine (client/src/lib/forecastCalculator.ts)                                                                                                                                               
  ▎                                                         
  ▎ Pure function: calculateForecast(state: ForecastState): ForecastResult.                                                                                                                                    
  ▎                                                         
  ▎ Monthly projection loop — for each month from 1 to contractTermYears × 12:                                                                                                                                 
  ▎                                                         
  ▎ // 1. Adoption → active users                                                                                                                                                                              
  ▎ const adoptionPct = computeAdoptionCurve(state.adoptionCurve, month);
  ▎ const utilizationPct = state.utilizationCurve.values[quarterIdx(month)] / 100;                                                                                                                             
  ▎ const activeUsers = state.provisionedSeats * adoptionPct * utilizationPct;                                                                                                                                 
  ▎                                                                                                                                                                                                            
  ▎ // 2. Encounters on Abridge                                                                                                                                                                                
  ▎ const monthlyOrgEncounters =                                                                                                                                                                               
  ▎   (state.totalOrgEncountersLTM / 12) * Math.pow(1 + monthlyGrowthRate, month - 1);                                                                                                                         
  ▎ const abridgeSharePct = state.encounterShareCurve.values[quarterIdx(month)] / 100;                                                                                                                         
  ▎ const monthlyAbridgeEncounters = monthlyOrgEncounters * abridgeSharePct;                                                                                                                                   
  ▎                                                                                                                                                                                                            
  ▎ // 3. Cost (current pricing)                                                                                                                                                                               
  ▎ const cost = computeCost(                                                                                                                                                                                  
  ▎   state.currentPricing, month, activeUsers,                                                                                                                                                                
  ▎   monthlyAbridgeEncounters, state.nursingStaffedBeds, cumulativeEncounters                                                                                                                                 
  ▎ );                                                                                                                                                                                                         
  ▎                                                                                                                                                                                                            
  ▎ // 4. Value (sum over drivers with onset ramp)                                                                                                                                                             
  ▎ let totalValue = 0;                                     
  ▎ const valueByDomain: Record<ValueDomain, number> = { ... };                                                                                                                                                
  ▎ for (const driver of state.valueDrivers) {                                                                                                                                                                 
  ▎   const ramp = rampFactor(driver.onset, month);                                                                                                                                                            
  ▎   const basis = scalingBasis(driver.scalingUnit, activeUsers,                                                                                                                                              
  ▎                              monthlyAbridgeEncounters, state.nursingStaffedBeds);                                                                                                                          
  ▎   const v = driver.projectedDelta * basis * ramp        
  ▎             * (driver.confidence / 100) * (driver.realizationPct / 100);                                                                                                                                   
  ▎   totalValue += v;                                                                                                                                                                                         
  ▎   valueByDomain[driver.domain] += v;                                                                                                                                                                       
  ▎ }                                                                                                                                                                                                          
  ▎                                                                                                                                                                                                            
  ▎ const netValue = totalValue - cost;                     
  ▎ cumulativeNet += netValue;
  ▎ cumulativeNetExcludingLongTerm += (immediate/delayed/phased drivers only) - cost;                                                                                                                          
  ▎ cumulativeEncounters += monthlyAbridgeEncounters;                                                                                                                                                          
  ▎                                                                                                                                                                                                            
  ▎ Cost function (inside forecastCalculator.ts):                                                                                                                                                              
  ▎                                                                                                                                                                                                            
  ▎ function computeCost(p: PricingConfig, month, activeUsers, monthlyEncounters,                                                                                                                              
  ▎                     nursingBeds, cumulativeEncounters): number {
  ▎   const escalator = yearlyEscalatorForMonth(p.yearlyEscalators, month);                                                                                                                                    
  ▎   const basePrice = p.unitPrice * escalator;                                                                                                                                                               
  ▎   switch (p.model) {                                                                                                                                                                                       
  ▎     case 'perProvider':    return activeUsers * basePrice;                                                                                                                                                 
  ▎     case 'perStaffedBed':  return (nursingBeds ?? 0) * basePrice;                                                                                                                                          
  ▎     case 'annualFlat':     return basePrice / 12;       
  ▎     case 'perEncounter': {                                                                                                                                                                                 
  ▎       let cost = monthlyEncounters * basePrice;         
  ▎       if (p.contractEncounterLimit && cumulativeEncounters > p.contractEncounterLimit) {                                                                                                                   
  ▎         const overageEncs = Math.min(monthlyEncounters,                                                                                                                                                    
  ▎                                      cumulativeEncounters - p.contractEncounterLimit);                                                                                                                     
  ▎         cost += overageEncs * (p.overageRate ?? basePrice);                                                                                                                                                
  ▎       }                                                                                                                                                                                                    
  ▎       return cost;                                      
  ▎     }                                                                                                                                                                                                      
  ▎     case 'hybrid': {                                    
  ▎       const primary = computeCost({ ...p, model: p.model }, ...);                                                                                                                                          
  ▎       const secondary = computeCost(                                                                                                                                                                       
  ▎         { ...p, model: p.secondaryModel!, unitPrice: p.secondaryUnitPrice! }, ...);                                                                                                                        
  ▎       return primary + secondary;                                                                                                                                                                          
  ▎     }                                                   
  ▎   }                                                                                                                                                                                                        
  ▎ }                                                       
  ▎
  ▎ Onset ramp:                                                                                                                                                                                                
  ▎
  ▎ const ONSET_DELAY_MONTHS = {                                                                                                                                                                               
  ▎   immediate: 2, delayed: 5, phased: 6, longTerm: 15,    
  ▎ };                                                                                                                                                                                                         
  ▎ function rampFactor(onset, month) {                                                                                                                                                                        
  ▎   const delay = ONSET_DELAY_MONTHS[onset];
  ▎   if (month < delay) return 0;                                                                                                                                                                             
  ▎   return Math.min(1, (month - delay) / 3); // linear 3-month ramp                                                                                                                                          
  ▎ }                                                                                                                                                                                                          
  ▎                                                                                                                                                                                                            
  ▎ Scaling basis:                                                                                                                                                                                             
  ▎                                                         
  ▎ function scalingBasis(unit, activeUsers, monthlyEncounters, nursingBeds) {                                                                                                                                 
  ▎   switch (unit) {
  ▎     case 'perEncounter':  return monthlyEncounters;                                                                                                                                                        
  ▎     case 'perActiveUser': return activeUsers;                                                                                                                                                              
  ▎     case 'perBed':        return nursingBeds ?? 0;
  ▎     case 'annualFlat':    return 1 / 12;                                                                                                                                                                   
  ▎   }                                                     
  ▎ }                                                                                                                                                                                                          
  ▎                                                         
  ▎ After the loop, roll up:                                                                                                                                                                                   
  ▎ - fastBreakEvenMonth = first month where cumulativeNetExcludingLongTerm >= 0 (null if not reached)
  ▎ - fullBreakEvenMonth = first month where cumulativeNet >= 0 (null if not reached)                                                                                                                          
  ▎ - runwayMonth (encounter-mode only) = first month where cumulativeEncounters >= contractEncounterLimit                                                                                                     
  ▎ - capacityBreachMonth = first month where cumulativeEncounters >= capacityCeiling                                                                                                                          
  ▎ - projectedOverage = sum of overage costs across all months                                                                                                                                                
  ▎ - totalContractValue = sum of monthly totalValue                                                                                                                                                           
  ▎ - netContractValue = sum of monthly netValue                                                                                                                                                               
  ▎ - roiMultiple = totalContractValue / sum(cost)                                                                                                                                                             
  ▎                                                                                                                                                                                                            
  ▎ Alerts (generate in engine, consumed by UI):                                                                                                                                                               
  ▎ - If runwayMonth within contract term: { type: 'runway', month, message: "You'll hit 500K in Month N" }                                                                                                    
  ▎ - If capacityBreachMonth within term: { type: 'capacity', month, message: "Projected to exceed capacity ceiling in Month N" }                                                                              
  ▎ - If any comparison pricing saves > 10% of current cost: { type: 'pricing-savings', comparisonId, delta }                                                                                                  
  ▎                                                                                                                                                                                                            
  ▎ Pricing comparison section: for each state.comparisonPricing[i], run the same monthly loop but with computeCost called against the alternate PricingConfig — emit parallel ForecastMonthRow[] into         
  ▎ alternateMonthly[comparisonId]. Value curve is identical across comparisons (usage-driven, not pricing-driven).
