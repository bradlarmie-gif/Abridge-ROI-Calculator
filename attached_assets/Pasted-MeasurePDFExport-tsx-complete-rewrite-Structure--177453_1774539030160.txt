                  const label = def?.label ?? m.metricId;
                  const unit = def?.unit ?? "";
                  const lowerIsBetter = def?.lowerIsBetter ?? false;
                  const d = m.after - m.before;
                  const improved = lowerIsBetter ? d < 0 : d > 0;
                  const deltaStr = fmtDelta(m.before, m.after, unit);
                  const suffix = unit === "%" ? "%" : unit ? ` ${unit}` : "";
                  const settingTag = m.setting ? ` (${settingShort(m.setting)})` : "";
                  return (
                    <View key={`${m.metricId}__${m.setting ?? "org"}`} style={[s.metricRow, i % 2 === 1 ? { backgroundColor: C.altRow } : {}]}>
                      <Text style={s.mName}>{label}{settingTag}</Text>
                      <Text style={s.mBefore}>{m.before}{suffix}</Text>
                      <Text style={s.mArrow}>{"\u2192"}</Text>
                      <Text style={s.mAfter}>{m.after}{suffix}</Text>
                      <Text style={[s.mDelta, { color: improved ? C.orange : C.muted }]}>{deltaStr}</Text>
                    </View>
                  );
                })}
                <View style={[s.rule, { marginTop: 4 }]} />
                <Text style={{ fontSize: 8.5, color: C.muted, lineHeight: 1.5 }}>pp = percentage points. Figures reflect {months}-month deployment period. Abridge vs. non-Abridge encounter comparison where
  noted.</Text>
              </>
            )}
            <Footer n={P()} total={TOTAL} org={orgName} />
          </View>
        </Page>

        {/* PAGE 3 (conditional): Financial Impact */}
        {hasFinancials && (
          <Page size="LETTER" style={s.page} wrap={false}>
            <View style={s.wrap}>
              <Conf org={orgName} />
              <Text style={s.eyebrow}>Financial Impact</Text>
              <Text style={s.headline}>What This Could Mean Financially</Text>
              <Text style={s.subline}>Attribution: {fin.attrRange} \u00B7 Realization: {DEFAULT_ASMP.realization}% \u00B7 Based on confirmed deployment data</Text>
              <View style={[s.card, { paddingVertical: 20, alignItems: "center", marginBottom: 14 }]}>
                <Text style={{ fontSize: 9, color: C.muted, textTransform: "uppercase", letterSpacing: 1.5, marginBottom: 8 }}>Confirmed Annual Value Range</Text>
                <Text style={{ fontSize: 28, fontWeight: "bold", color: C.dark }}>{fmtRange(Math.round(fin.totalLo), Math.round(fin.totalHi))}</Text>
                <Text style={{ fontSize: 9, color: C.muted, marginTop: 6 }}>{fmtN(providers)} providers \u00B7 {fmtN(adopted)} adopted encounters</Text>
              </View>
              {fin.hasBill && (
                <View style={s.cardOutline}>
                  <View style={s.finRow}><Text style={s.finLabel}>Billing Capture</Text><Text style={s.finValue}>{fmtRange(Math.round(fin.billLo), Math.round(fin.billHi))}</Text></View>
                  <Text style={s.finCat}>Revenue \u00B7 Better documentation \u2192 more accurate coding</Text>
                  {fin.billDetails.map((d, i) => (
                    <View key={i}><Text style={s.finDetail}>{d.label}</Text><Text style={s.finFormula}>{d.formula}</Text></View>
                  ))}
                </View>
              )}
              {fin.hasRec && (
                <View style={s.cardOutline}>
                  <View style={s.finRow}><Text style={s.finLabel}>Revenue Recovery</Text><Text style={s.finValue}>{fmtRange(Math.round(fin.recLo), Math.round(fin.recHi))}</Text></View>
                  <Text style={s.finCat}>Throughput \u00B7 Recovered visits + reduced claim denials</Text>
                  {fin.recDetails.map((d, i) => (
                    <View key={i}><Text style={s.finDetail}>{d.label}</Text><Text style={s.finFormula}>{d.formula}</Text></View>
                  ))}
                </View>
              )}
              {fin.hasCost && (
                <View style={s.cardOutline}>
                  <View style={s.finRow}><Text style={s.finLabel}>Cost Reduction</Text><Text style={s.finValue}>{fmtRange(Math.round(fin.costLo), Math.round(fin.costHi))}</Text></View>
                  <Text style={s.finCat}>Avoided cost \u00B7 OT premium + physician retention</Text>
                  {fin.costDetails.map((d, i) => (
                    <View key={i}><Text style={s.finDetail}>{d.label}</Text><Text style={s.finFormula}>{d.formula}</Text></View>
                  ))}
                </View>
              )}
              <Footer n={P()} total={TOTAL} org={orgName} />
            </View>
          </Page>
        )}

        {/* PAGE (conditional): Signals Worth Watching */}
        {hasSignals && (
          <Page size="LETTER" style={s.page} wrap>
            <View style={s.wrap}>
              <Conf org={orgName} />
              <Text style={s.eyebrow}>Signals Worth Watching</Text>
              <Text style={s.headline}>Why These Metrics Matter</Text>
              <Text style={s.subline}>Leading indicators and downstream signals \u2014 what each one means for {orgName}</Text>
              <Text style={[s.narrative, { marginBottom: 12 }]}>
                Not every improvement maps to a formula. These metrics are real, measurable, and meaningful. They represent the upstream signals that compound into financial outcomes over time. For any
  metric where improvement is already visible, the next step is connecting it to a downstream dollar-denominated measurement.
              </Text>
              {signalMetrics.map((m) => {
                const def = METRIC_MAP.get(m.metricId);
                if (!def) return null;
                const d = m.after - m.before;
                const improved = def.lowerIsBetter ? d < 0 : d > 0;
                const unit = def.unit;
                const suffix = unit === "%" ? "%" : unit ? ` ${unit}` : "";
                const domainLabel = def.domain.charAt(0).toUpperCase() + def.domain.slice(1);
                const settingTag = m.setting ? ` \u00B7 ${settingFull(m.setting)}` : "";
                return (
                  <View key={`${m.metricId}__${m.setting ?? "org"}`} style={s.signalCard}>
                    <Text style={s.signalDomain}>{domainLabel}{settingTag}</Text>
                    <Text style={s.signalTitle}>{def.label}</Text>
                    <Text style={s.signalChange}>{m.before}{suffix} {"\u2192"} {m.after}{suffix}{"  "}{improved ? "\u2713 improved" : "\u26A0 declined"}</Text>
                    <Text style={s.signalText}>{def.whyItMatters}</Text>
                  </View>
                );
              })}
              <Footer n={P()} total={TOTAL} org={orgName} />
            </View>
          </Page>
        )}

        {/* PAGE: Maturity + What's Next */}
        <Page size="LETTER" style={s.page} wrap={false}>
          <View style={s.wrap}>
            <Conf org={orgName} />
            <Text style={s.eyebrow}>Where You Stand</Text>
            <Text style={s.headline}>Maturity & What\u2019s Next</Text>
            <Text style={s.subline}>{Math.round(utilRate)}% adoption \u00B7 {activeMetrics.length} metrics tracked \u00B7 {activeDomains} domain{activeDomains !== 1 ? "s" : ""} with data</Text>
            <View style={s.card}>
              <View style={s.pipRow}>
                {maturityLabels.map((_, i) => (<View key={i} style={i <= maturityIdx ? s.pipOn : s.pip} />))}
              </View>
              <View style={{ flexDirection: "row", justifyContent: "space-between", marginBottom: 10 }}>
                {maturityLabels.map((label, i) => (
                  <Text key={i} style={{ fontSize: 8, color: i === maturityIdx ? C.orange : C.muted, fontWeight: i === maturityIdx ? "bold" : "normal", flex: 1, textAlign: "center" }}>{label}</Text>
                ))}
              </View>
              <Text style={{ fontSize: 13, fontWeight: "bold", color: C.dark, marginBottom: 4 }}>{maturityLabels[maturityIdx]}</Text>
              <Text style={{ fontSize: 9.5, color: C.mid, lineHeight: 1.55 }}>{maturityDescs[maturityIdx]}</Text>
            </View>
            <View style={s.rule} />
            <Text style={[s.eyebrow, { marginBottom: 8 }]}>Expansion Levers</Text>
            {utilRate < 75 && (
              <View style={s.cardOutline}>
                <Text style={{ fontSize: 10, fontWeight: "bold", color: C.dark, marginBottom: 3 }}>Deepen Adoption</Text>
                <Text style={{ fontSize: 9.5, color: C.mid, lineHeight: 1.5 }}>You\u2019re at {Math.round(utilRate)}% adoption. Reaching 75% with the same {providers} providers would compound your current
  results. Target: {Math.round(providers * 0.75)} active providers.</Text>
              </View>
            )}
            {providers < totalProviders && (
              <View style={s.cardOutline}>
                <Text style={{ fontSize: 10, fontWeight: "bold", color: C.dark, marginBottom: 3 }}>Expand Provider Count</Text>
                <Text style={{ fontSize: 9.5, color: C.mid, lineHeight: 1.5 }}>{totalProviders - providers} providers at {orgName} are not yet on Abridge. Based on current per-provider impact, full
  deployment represents meaningful upside.</Text>
              </View>
            )}
            {activeCareSettings.length <= 1 && (
              <View style={s.cardOutline}>
                <Text style={{ fontSize: 10, fontWeight: "bold", color: C.dark, marginBottom: 3 }}>Add a Care Setting</Text>
                <Text style={{ fontSize: 9.5, color: C.mid, lineHeight: 1.5 }}>You\u2019re currently measuring in {settingLabel}. Each additional care setting multiplies your measurement footprint and
  broadens the story.</Text>
              </View>
            )}
            <View style={s.rule} />
            <Text style={[s.eyebrow, { marginBottom: 8 }]}>Recommended Next Moves</Text>
            {(
              [
                utilRate < 40 ? `Drive adoption to 40% \u2014 schedule department-level Abridge sessions at ${orgName}` : null,
                activeMetrics.length < 4 ? "Add measurement depth \u2014 4+ metrics makes the story defensible for CFO/CMO review" : null,
                hasFinancials ? "Present this EBR to finance and quality leadership \u2014 the data supports an internal ROI conversation" : "Establish quantifiable metrics (wRVU, E/M, LWBS) to generate a
  dollar-denominated ROI story",
                "Name one metric, one target, one owner \u2014 and schedule a 90-day check-in",
              ] as (string | null)[]
            ).filter((x): x is string => x !== null).slice(0, 4).map((move, i) => (
              <View key={i} style={s.bulletRow}><View style={s.bullet} /><Text style={s.bulletText}>{move}</Text></View>
            ))}
            <Footer n={P()} total={TOTAL} org={orgName} />
          </View>
        </Page>

        {/* PAGE: Methodology */}
        <Page size="LETTER" style={s.page} wrap={false}>
          <View style={s.wrap}>
            <Conf org={orgName} />
            <Text style={s.eyebrow}>Methodology</Text>
            <Text style={s.headline}>How We Calculated This</Text>
            <Text style={s.subline}>Attribution model, assumptions, and what we intentionally don\u2019t count</Text>
            <View style={s.card}>
              <Text style={{ fontSize: 9, fontWeight: "bold", textTransform: "uppercase", letterSpacing: 1, color: C.dark, marginBottom: 6 }}>Attribution Model</Text>
              <Text style={{ fontSize: 9.5, color: C.mid, lineHeight: 1.6 }}>Not every observed improvement is attributable to Abridge. Our default attribution range ({fin.attrRange}) accounts for concurrent
   coding education, payer mix shifts, provider behavior changes, and other factors that occur in parallel with any technology deployment. We apply this range to all calculable financial estimates, producing
   low and high bounds rather than a single point estimate.</Text>
            </View>
            <View style={s.card}>
              <Text style={{ fontSize: 9, fontWeight: "bold", textTransform: "uppercase", letterSpacing: 1, color: C.dark, marginBottom: 8 }}>Financial Calculation Assumptions</Text>
              {[
                ["wRVU conversion factor", `$${DEFAULT_ASMP.conversionFactor}/wRVU`],
                ["E/M level value", "~$45/level"],
                ["DRG base rate (IP)", `$${fmtN(DEFAULT_ASMP.drgBaseRate)}`],
                ["ED revenue per recovered visit", `$${DEFAULT_ASMP.edRevenuePerVisit}`],
                ["Denial rework cost", "$350/denial"],
                ["OT premium rate", `$${DEFAULT_ASMP.otPremiumRate}/hr`],
                ["Physician replacement cost", "$50K\u2013$150K (MGMA benchmark)"],
                ["Attribution range", fin.attrRange],
                ["Realization rate", `${DEFAULT_ASMP.realization}%`],
              ].map(([key, val]) => (
                <View key={key} style={s.methRow}><Text style={s.methKey}>{key}</Text><Text style={s.methVal}>{val}</Text></View>
              ))}
            </View>
            <View style={s.callout}>
              <Text style={{ fontSize: 9, fontWeight: "bold", textTransform: "uppercase", letterSpacing: 1, color: C.dark, marginBottom: 5 }}>What We Don\u2019t Count</Text>
              <Text style={{ fontSize: 9.5, color: C.mid, lineHeight: 1.6 }}>This report does not use "provider time \u00D7 hourly rate" as a financial metric. Time savings are real, but how organizations
  deploy recovered time varies \u2014 some reinvest in throughput, some in care quality, some in provider wellbeing. We present time metrics as operational evidence and let your leadership team assign the
  appropriate value for your context.</Text>
            </View>
            {activeMetrics.length > 0 && (
              <Text style={{ fontSize: 8.5, color: C.muted, marginTop: 8, lineHeight: 1.5 }}>
                {"Data sources: "}
                {Array.from(new Set(activeMetrics.map((m) => METRIC_MAP.get(m.metricId)?.sourceLabel).filter((x): x is string => !!x))).join(" \u00B7 ")}
                {`. Observation period: ${months} months.`}
              </Text>
            )}
            <Footer n={P()} total={TOTAL} org={orgName} />
          </View>
        </Page>
      </Document>
    );
  };

  export async function generateMeasurePDF(state: MeasureState): Promise<void> {
    const orgName = (state.deployment.organizationName || "Organization")
      .replace(/[^a-zA-Z0-9\s-]/g, "")        
      .replace(/\s+/g, "-");                                                                                                                                                                                   
    const blob = await pdf(<MeasureEBR state={state} />).toBlob();
    savePdfBlob(blob, `${orgName}-Abridge-EBR.pdf`);                                                                                                                                                           
  }
