Add security measures and data hygiene to the ROI Calculator application. This tool handles potentially sensitive business data (provider counts, revenue figures, etc.) and needs to be secure and privacy-conscious.

## 1. AUTO-CLEAR SESSION DATA

Clear all entered data after a period of inactivity or when the user leaves.

### Session Timeout (30 minutes of inactivity)
```javascript
// Add to your main App component or a global context

const SESSION_TIMEOUT = 30 * 60 * 1000; // 30 minutes in milliseconds
let inactivityTimer;

function resetInactivityTimer() {
  clearTimeout(inactivityTimer);
  inactivityTimer = setTimeout(() => {
    clearAllData();
    showSessionExpiredModal();

function clearAllData() {

      document.removeEventListener(event, resetInactivityTimer);

Never use localStorage for sensitive data

  providers: 75,
  encounters: 150000,
  revenue: 500000

If you must persist data temporarily
javascriptCopy// Use sessionStorage with encryption for any sensitive values
import CryptoJS from 'crypto-js';

const SESSION_KEY = 'roi_calc_' + Date.now(); // Unique per session

function saveToSession(key, data) {
  const encrypted = CryptoJS.AES.encrypt(
    JSON.stringify(data), 
    SESSION_KEY
  ).toString();
  sessionStorage.setItem(key, encrypted);
}

function loadFromSession(key) {
  const encrypted = sessionStorage.getItem(key);
  if (!encrypted) return null;
  
  try {
    const bytes = CryptoJS.AES.decrypt(encrypted, SESSION_KEY);
    return JSON.parse(bytes.toString(CryptoJS.enc.Utf8));
  } catch {
    return null;
  }
}
3. INPUT VALIDATION & SANITIZATION
Validate all numeric inputs
javascriptCopyfunction sanitizeNumber(value, min = 0, max = Infinity) {
  const num = parseFloat(value);
  if (isNaN(num)) return null;
  if (num < min) return min;
  if (num > max) return max;
  return Math.round(num * 100) / 100; // Max 2 decimal places
}

// Usage
const handleProvidersChange = (e) => {
  const sanitized = sanitizeNumber(e.target.value, 1, 10000);
  if (sanitized !== null) {
    setProviders(sanitized);
  }
};
Validate percentage inputs
javascriptCopyfunction sanitizePercentage(value) {
  const num = parseFloat(value);
  if (isNaN(num)) return null;
  return Math.min(100, Math.max(0, num));
}
Prevent script injection in any text inputs
javascriptCopyfunction sanitizeText(value) {
  if (!value) return '';
  return value
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#x27;')
    .trim()
    .slice(0, 500); // Max length
}
4. RATE LIMITING (for any API calls)
If you have any API calls (analytics, PDF export, etc.):
javascriptCopy// Simple client-side rate limiter
const rateLimiter = {
  calls: {},
  
  canMakeCall(endpoint, maxCalls = 10, windowMs = 60000) {
    const now = Date.now();
    const key = endpoint;
    
    if (!this.calls[key]) {
      this.calls[key] = [];
    }
    
    // Remove old calls outside the window
    this.calls[key] = this.calls[key].filter(time => now - time < windowMs);
    
    if (this.calls[key].length >= maxCalls) {
      return false;
    }
    
    this.calls[key].push(now);
    return true;
  }
};

// Usage
async function exportPDF() {
  if (!rateLimiter.canMakeCall('exportPDF', 5, 60000)) {
    showError('Too many export requests. Please wait a minute.');
    return;
  }
  
  // ... proceed with export
}

async function generateShareLink(data) {
  // If you have a backend
  const response = await fetch('/api/share', {
    method: 'POST',
    body: JSON.stringify({
      data: data,
      expiresIn: '24h'
    })
  });

  return `https://roi.abridge.com/shared/${token}`;
}
