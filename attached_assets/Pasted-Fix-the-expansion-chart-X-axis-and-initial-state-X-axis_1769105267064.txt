javascriptCopy// Smart X-axis labels
function generateXAxisLabels(start, end, count = 6) {
  const labels = [];
  const step = (end - start) / (count - 1);
  
  for (let i = 0; i < count; i++) {
    const value = Math.round(start + (step * i));
    labels.push(value);
  }
  
  return labels;
}
