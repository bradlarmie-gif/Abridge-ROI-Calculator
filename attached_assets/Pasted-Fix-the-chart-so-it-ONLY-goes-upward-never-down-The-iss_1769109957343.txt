      providers: Math.round(currentProviders + (targetProviders - currentProviders) * 0.40),
      utilization: Math.min(currentUtilization + 6, targetUtilization),
      value: currentValue * 1.52 // 52% increase (compound growth)
    },
    {
      label: '18 months',
      providers: Math.round(currentProviders + (targetProviders - currentProviders) * 0.70),
      utilization: Math.min(currentUtilization + 9, targetUtilization),
      value: currentValue * 1.95 // 95% increase
    },
    {
      label: 'Full Scale',
      providers: targetProviders,
      utilization: targetUtilization,
      // Calculate based on ratio of providers and utilization improvement
