CALCULATIONS
══════════════════════════════════════════════════════════════════════════════════

// OT Reduction (flows from time saved)
otHoursEliminated = hoursSavedPerYear * (timeToOTConversion / 100)
otSavings = otHoursEliminated * otHourlyRate

// Validation: Cap OT savings at actual OT spend
maxOTHours = nurseFTEs * currentOTHoursPerWeek * 52
otHoursEliminated = Math.min(otHoursEliminated, maxOTHours)

// Retention
nursesLeavingPerYear = nurseFTEs * (turnoverRate / 100)
burnoutDepartures = nursesLeavingPerYear * (burnoutRelated / 100) // default 40%
nursesRetained = burnoutDepartures * (abridgeImpact / 100)
retentionSavings = nursesRetained * replacementCost

// Agency (only if enabled and retention is enabled)
agencySavings = nursesRetained * weeksOfCoverage * weeklyAgencyPremium

// Care Time
timeAfterOT = hoursSavedPerYear - otHoursEliminated
careTimeHours = timeAfterOT * (timeToCare / 100)
absorbedHours = timeAfterOT - careTimeHours

// Total Time Value
totalTimeValue = otSavings + retentionSavings + agencySavings
