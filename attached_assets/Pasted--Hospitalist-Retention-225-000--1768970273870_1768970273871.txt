// Inputs (user-editable)
const hospitalists = 20;

// 3.0 × 0.50 = 1.5 preventable

// Step 3: Abridge Attribution
const departuresAvoided = preventableDepartures * abridgeImpact;
// 1.5 × 0.30 = 0.45 avoided

// Step 4: Cost Savings
const annualRetentionSavings = departuresAvoided * replacementCost;
// 0.45 × $500,000 = $225,000

// Dynamic framing
const yearsToRetainOne = 1 / departuresAvoided;
// 1 / 0.45 = 2.2 years
