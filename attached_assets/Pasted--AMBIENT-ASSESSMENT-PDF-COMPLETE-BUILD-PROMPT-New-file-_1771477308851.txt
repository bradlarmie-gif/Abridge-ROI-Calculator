## OVERVIEW

Create a new file `client/src/lib/ambient-assessment-pdf.tsx`.

This is a standalone PDF generator for the ambient AI assessment path. It uses the same `@react-pdf/renderer` library and the same brand constants as `outpatient-pdf-generator.tsx`. Do not modify any existing PDF files.

**Three pages total:**
- Page 1: Cover (matches existing cover style exactly)
- Page 2: Executive Summary — score, gap, cost of waiting
- Page 3: Four Domains — activation levels, gaps, opportunities

## DATA INTERFACE

```tsx
export interface AmbientAssessmentPDFData {
  // Organization
  organizationName: string;
  preparedBy?: string;
  assessmentDate: string;

  // Baseline
  providers: number;
  annualEncounters: number;
  utilization: number;        // % e.g. 50
  timeSavings: number;        // min e.g. 2.0

  // Score
  documentationScore: number; // 0-100

  // Gap values
  totalAnnualGap: number;     // displayed (post-haircut)
  monthlyGap: number;
  dailyGap: number;

  // Three-year projections
  actNow3yr: number;
  wait6mo3yr: number;
  wait12mo3yr: number;
  permanentlyLost6mo: number;
  permanentlyLost12mo: number;

  // Four domains
  domains: {
    capacity: DomainData;
    revenue: DomainData;
    workforce: DomainData;
    risk: DomainData;
  };
}

export interface DomainData {
  activationLevel: 1 | 2 | 3 | 4;
  activationLabel: string;    // e.g. "Informally tracked"
  score: number;              // 0-100
  gapValue: number;           // $ annual gap
  keyInput?: string;          // e.g. "Denial rate: 7%"
  primaryOpportunity: string; // one sentence
}
```

## HELPER FUNCTIONS

```tsx
const fmt = (value: number): string => {
  if (value >= 1000000) return `$${(value / 1000000).toFixed(1)}M`;
  if (value >= 1000) return `$${Math.round(value / 1000)}K`;
  return `$${value.toLocaleString()}`;
};

const activationColors: Record<1 | 2 | 3 | 4, string> = {
  1: "#EDEDED",   // gray — not activated
  2: "#D4B896",   // tan — partial
  3: "#C17B3E",   // amber — active
  4: "#EA2C00",   // red — fully activated
};

const getScoreVerdict = (score: number): string => {
  if (score < 34) return "Below the industry average. Significant enterprise value is available across all four domains.";
  if (score <= 50) return "At the industry average. The gap to top quartile is not incremental. It is structural.";
  if (score <= 70) return "Above the industry average. The remaining gap is concentrated in specific domains.";
  if (score <= 85) return "Approaching top-quartile performance. The remaining opportunity is in optimization.";
  return "Top-quartile documentation intelligence. Focused on continuous optimization.";
};
```

```tsx
export async function generateAmbientAssessmentPDF(
  data: AmbientAssessmentPDFData
): Promise<void> {
  const blob = await pdf(<AmbientAssessmentDocument data={data} />).toBlob();
  const today = new Date().toISOString().split("T")[0];
  const orgName = (data.organizationName || "Organization").replace(/\s+/g, "_");
  const fileName = `Ambient_Assessment_${orgName}_${today}.pdf`;
  await savePdfBlob(blob, fileName);
}
```
