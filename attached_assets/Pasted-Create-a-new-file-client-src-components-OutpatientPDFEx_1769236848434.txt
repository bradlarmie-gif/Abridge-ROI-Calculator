const formatCurrency = (num: number): string => {
  if (Math.abs(num) >= 1000000) return `$${(num / 1000000).toFixed(1)}M`;
  if (Math.abs(num) >= 1000) return `$${Math.round(num / 1000)}K`;
  return `$${num.toLocaleString()}`;
};

const formatNumber = (num: number): string => num.toLocaleString();

interface DriverContent {
  theory: string;
  getSteps: (inputs: RoiInputs, value: number, eligibleEncounters: number, hoursReturned: number) => {
    label: string;
    description: string;
    math: string;
    result?: string;
  }[];
  benchmarks?: { label: string; value: string }[];
  warning?: { title: string; text: string };
}

const getDriverContent = (driverId: LeverId): DriverContent => {
  switch (driverId) {
    case "patientAccess":
      return {
        theory: "When clinicians spend less time on documentation, they have capacity to see additional patients. Not all saved time converts to visits—scheduling, room availability, and demand limit realization—but even a modest portion creates meaningful revenue.",
        getSteps: (inputs, value, eligibleEncounters, hoursReturned) => {
          const allocatedHours = hoursReturned * (inputs.patientAccess.pctTimeToNewVisits / 100);
          const additionalVisits = Math.round((allocatedHours * 60) / inputs.patientAccess.avgVisitDurationMinutes);
          return [
            {
              label: "Step 1: Time Returned",
              description: "How much time does Abridge give back?",
              math: `${formatNumber(eligibleEncounters)} enc × ${inputs.minutesSavedPerEncounter} min ÷ 60 = ${formatNumber(hoursReturned)} hrs`,
            },
            {
              label: "Step 2: Time Allocated to Access",
              description: `What portion of saved time goes toward seeing more patients? (${inputs.patientAccess.pctTimeToNewVisits}%)`,
              math: `${formatNumber(hoursReturned)} hrs × ${inputs.patientAccess.pctTimeToNewVisits}% = ${formatNumber(Math.round(allocatedHours))} hrs`,
            },
            {
              label: "Step 3: Conversion to Visits",
              description: "How many additional visits is that?",
              math: `${formatNumber(Math.round(allocatedHours))} hrs × 60 ÷ ${inputs.patientAccess.avgVisitDurationMinutes} min/visit = ${formatNumber(additionalVisits)} visits`,
            },
            {
              label: "Step 4: Revenue Impact",
              description: "What's the revenue value?",
              math: `${formatNumber(additionalVisits)} visits × ${formatCurrency(inputs.patientAccess.avgNetRevenuePerVisit)}/visit`,
              result: formatCurrency(value),
            },
          ];
        },
        benchmarks: [
          { label: "Primary Care", value: "$100 - $150" },
          { label: "Specialty", value: "$175 - $300" },
          { label: "Procedural", value: "$300 - $600+" },
        ],
      };

    case "wrvu":
      return {
        theory: "Physicians under time pressure document less than the full clinical picture. AI-assisted documentation captures the complexity that supports accurate coding—not upcoding, just getting credit for work already done.",
        getSteps: (inputs, value, eligibleEncounters) => {
          const baselineWrvus = eligibleEncounters * inputs.baselineWrvuPerEncounter;
          const wrvuGain = baselineWrvus * (inputs.wrvu.pctIncreaseWrvuPerEncounter / 100);
          return [
            {
              label: "Step 1: Baseline wRVUs",
              description: "What's your current productivity?",
              math: `${formatNumber(eligibleEncounters)} enc × ${inputs.baselineWrvuPerEncounter} wRVU/enc = ${formatNumber(Math.round(baselineWrvus))} wRVUs`,
            },
            {
              label: "Step 2: wRVU Improvement",
              description: `How much does Abridge improve capture? (+${inputs.wrvu.pctIncreaseWrvuPerEncounter}%)`,
              math: `${formatNumber(Math.round(baselineWrvus))} wRVUs × ${inputs.wrvu.pctIncreaseWrvuPerEncounter}% = ${formatNumber(Math.round(wrvuGain))} wRVU gain`,
            },
            {
              label: "Step 3: Revenue Impact",
              description: "What's the dollar value?",
              math: `${formatNumber(Math.round(wrvuGain))} wRVU × ${formatCurrency(inputs.wrvu.wrvuConversionFactor)}/wRVU`,
              result: formatCurrency(value),
            },
          ];
        },
        benchmarks: [
          { label: "Medicare (2024)", value: "$33" },
          { label: "Commercial (typical)", value: "$45 - $65" },
        ],
      };

    case "workforce":
      return {
        theory: "Documentation burden is the #1 driver of physician burnout. Reducing this burden improves satisfaction and retention. Replacing a physician costs $400K-$800K+ when you factor in recruiting, lost revenue during vacancy, and onboarding.",
        getSteps: (inputs, value) => {
          const departures = inputs.workforce.providerCount * (inputs.workforce.baselineAttritionRate / 100);
          const burnoutRelated = departures * (inputs.workforce.pctAttritionLinkedToBurnout / 100);
          const avoided = burnoutRelated * (inputs.workforce.pctBurnoutExitsAvoided / 100);
          return [
            {
              label: "Step 1: Baseline Turnover",
              description: "What's the current turnover situation?",
              math: `${inputs.workforce.providerCount} providers × ${inputs.workforce.baselineAttritionRate}% = ${departures.toFixed(1)} departures/year`,
            },
            {
              label: "Step 2: Burnout-Related Departures",
              description: `How much is burnout-driven? (${inputs.workforce.pctAttritionLinkedToBurnout}%)`,
              math: `${departures.toFixed(1)} departures × ${inputs.workforce.pctAttritionLinkedToBurnout}% = ${burnoutRelated.toFixed(2)} preventable`,
            },
            {
              label: "Step 3: Abridge Attribution",
              description: "What can Abridge prevent? (Rule of Thirds: 30%)",
              math: `${burnoutRelated.toFixed(2)} preventable × ${inputs.workforce.pctBurnoutExitsAvoided}% = ${avoided.toFixed(2)}/year avoided`,
            },
            {
              label: "Step 4: Cost Savings",
              description: "What's the dollar value?",
              math: `${avoided.toFixed(2)} avoided × ${formatCurrency(inputs.workforce.costPerDeparture)} replacement cost`,
              result: formatCurrency(value),
            },
          ];
        },
        warning: {
          title: "Long-term Investment Metric",
          text: "With fewer than 50 providers, retention savings are probabilistic over multi-year periods. Consider this a long-term investment metric rather than a near-term ROI driver.",
        },
      };

    case "denials":
      return {
        theory: "Most denials are recoverable—you appeal, you win, it just costs time. But a portion of documentation-related denials are written off without appeal, either because the MDM can't support it or the rework cost exceeds the claim value. Abridge captures the clinical reasoning that saves these.",
        getSteps: (inputs, value, eligibleEncounters) => {
          const totalDenials = eligibleEncounters * (inputs.denials.baselineDenialRate / 100);
          const docDenials = totalDenials * (inputs.denials.pctDenialsFromDocumentation / 100);
          const writtenOff = docDenials * ((100 - inputs.denials.pctDenialsRecoveredAfterRework) / 100);
          const recovered = writtenOff * (inputs.denials.pctDocDenialsRecovered / 100);
          return [
            {
              label: "Step 1: Total Denials",
              description: "How many claims are denied today?",
              math: `${formatNumber(eligibleEncounters)} enc × ${inputs.denials.baselineDenialRate}% = ${formatNumber(Math.round(totalDenials))} denials`,
            },
            {
              label: "Step 2: Documentation-Related",
              description: `How many are caused by documentation gaps? (${inputs.denials.pctDenialsFromDocumentation}%)`,
              math: `${formatNumber(Math.round(totalDenials))} denials × ${inputs.denials.pctDenialsFromDocumentation}% = ${formatNumber(Math.round(docDenials))} doc denials`,
            },
            {
              label: "Step 3: Written Off",
              description: "How many are lost without appeal?",
              math: `${formatNumber(Math.round(docDenials))} doc denials × ${100 - inputs.denials.pctDenialsRecoveredAfterRework}% = ${formatNumber(Math.round(writtenOff))} written off`,
            },
            {
              label: "Step 4: Abridge Recovery",
              description: `How many can Abridge save? (${inputs.denials.pctDocDenialsRecovered}%)`,
              math: `${formatNumber(Math.round(writtenOff))} written off × ${inputs.denials.pctDocDenialsRecovered}% = ${formatNumber(Math.round(recovered))} recovered`,
            },
            {
              label: "Step 5: Value",
              description: "What's the dollar impact?",
              math: `${formatNumber(Math.round(recovered))} claims × ${formatCurrency(inputs.denials.avgRevenuePerEncounter)} avg claim value`,
              result: formatCurrency(value),
            },
          ];
        },
        benchmarks: [
          { label: "Primary Care E/M", value: "$125 - $175" },
          { label: "Specialty E/M", value: "$200 - $350" },
          { label: "Blended Outpatient", value: "~$250" },
        ],
      };

    case "overtime":
      return {
        theory: "Overtime commonly reflects workflow spillover rather than true staffing gaps. When documentation fits inside the workday, premium labor drops naturally and predictably.",
        getSteps: (inputs, value, eligibleEncounters, hoursReturned) => {
          const afterHoursReclaimed = hoursReturned * (inputs.overtime.pctAfterHours / 100);
          const overtimeReduced = afterHoursReclaimed * (inputs.overtime.pctOvertimeReduced / 100);
          return [
            {
              label: "Step 1: Baseline Overtime Hours",
              description: "How much after-hours documentation exists today?",
              math: `${formatNumber(hoursReturned)} hrs returned × ${inputs.overtime.pctAfterHours}% after-hours = ${formatNumber(Math.round(afterHoursReclaimed))} hrs`,
            },
            {
              label: "Step 2: Documentation Time Eliminated",
              description: `How much can Abridge reduce? (${inputs.overtime.pctOvertimeReduced}%)`,
              math: `${formatNumber(Math.round(afterHoursReclaimed))} hrs × ${inputs.overtime.pctOvertimeReduced}% = ${formatNumber(Math.round(overtimeReduced))} hrs`,
            },
            {
              label: "Step 3: Cost Savings",
              description: "What's the dollar value?",
              math: `${formatNumber(Math.round(overtimeReduced))} hrs × ${formatCurrency(inputs.overtime.blendedOvertimeRate)}/hr`,
              result: formatCurrency(value),
            },
          ];
        },
      };

    case "hcc":
      return {
        theory: "Risk adjustment relies on complete documentation of chronic conditions. Physicians discuss multiple conditions per visit, but time pressure means not all make it to the note. Abridge captures what's said, recovering HCC opportunities that would otherwise be missed.",
        getSteps: (inputs, value, eligibleEncounters) => {
          const riskEncounters = eligibleEncounters * (inputs.hcc.pctMedicareAdvantage / 100);
          const missedPerVisit = inputs.hcc.avgConditionsPerMember * (inputs.hcc.pctConditionsMissed / 100) * 0.35; // HCC-eligible portion
          const missedOpportunities = riskEncounters * missedPerVisit;
          const captured = missedOpportunities * (inputs.hcc.pctMissedConditionsRecaptured / 100);
          return [
            {
              label: "Step 1: Risk-Based Encounters",
              description: `How many encounters are in risk contracts? (${inputs.hcc.pctMedicareAdvantage}%)`,
              math: `${formatNumber(eligibleEncounters)} enc × ${inputs.hcc.pctMedicareAdvantage}% = ${formatNumber(Math.round(riskEncounters))} risk encounters`,
            },
            {
              label: "Step 2: Missed HCC Opportunities",
              description: "How many HCC-eligible conditions are discussed but not documented?",
              math: `${formatNumber(Math.round(riskEncounters))} enc × ${missedPerVisit.toFixed(2)} missed/visit = ${formatNumber(Math.round(missedOpportunities))} HCC opportunities`,
            },
            {
              label: "Step 3: Abridge Capture",
              description: `How many can Abridge recover? (${inputs.hcc.pctMissedConditionsRecaptured}%)`,
              math: `${formatNumber(Math.round(missedOpportunities))} opportunities × ${inputs.hcc.pctMissedConditionsRecaptured}% = ${formatNumber(Math.round(captured))} HCCs captured`,
            },
            {
              label: "Step 4: Revenue Impact",
              description: "What's the risk-adjusted value?",
              math: `${formatNumber(Math.round(captured))} HCCs × $${inputs.hcc.pmpmBenchmark} × ${(1 - inputs.hcc.rafRealizationHaircut / 100) * 100}% audit factor`,
              result: formatCurrency(value),
            },
          ];
        },
        benchmarks: [
          { label: "Low-complexity HCC", value: "$400 - $600" },
          { label: "Medium-complexity HCC", value: "$700 - $1,000" },
          { label: "High-complexity HCC", value: "$1,200 - $2,500+" },
        ],
        warning: {
          title: "Varies by Organization",
          text: "HCC value varies significantly based on payer mix and current capture rates. Organizations with mature risk programs may see lower opportunity; those just starting may see more.",
        },
      };

    default:
      return {
        theory: leverDescriptions[driverId] || "Value driver",
        getSteps: () => [],
      };
  }
};

const getROINarrative = (roi: number, netGain: number, totalValue: number): string => {
  if (roi >= 5) {
    return `With a ${roi.toFixed(1)}x return on investment, this deployment generates exceptional value. For every dollar invested in Abridge, your organization captures $${roi.toFixed(2)} in return—a compelling case for both pilot expansion and enterprise deployment.`;
  } else if (roi >= 3) {
    return `At ${roi.toFixed(1)}x ROI, this deployment delivers strong returns that exceed most healthcare technology investments. The ${formatCurrency(netGain)} in annual net gain represents real operational and financial improvement.`;
  } else if (roi >= 2) {
    return `A ${roi.toFixed(1)}x return represents solid value creation. While there may be opportunity to optimize further through increased utilization or additional value drivers, the investment is clearly justified.`;
  } else {
    return `At ${roi.toFixed(1)}x ROI, the investment returns positive value but may benefit from optimization. Consider whether additional value drivers or higher utilization could improve returns.`;
  }
};

const getDriverMixInsight = (
  laborValue: number,
  revenueValue: number,
  totalValue: number
): string => {
  const laborPct = Math.round((laborValue / totalValue) * 100);
  const revenuePct = Math.round((revenueValue / totalValue) * 100);

  if (laborPct > 60) {
    return `Your value is concentrated in Labor & Efficiency drivers (${laborPct}% of total). This suggests the primary opportunity is time savings and workforce retention. Consider whether Revenue & Quality drivers like wRVU lift or HCC capture could add incremental value.`;
  } else if (revenuePct > 60) {
    return `Your value is concentrated in Revenue & Quality drivers (${revenuePct}% of total). This represents direct financial improvement through better documentation and coding. These gains are highly measurable and defensible.`;
  } else {
    return `Your value is balanced across Labor & Efficiency (${laborPct}%) and Revenue & Quality (${revenuePct}%). This diversification means value comes from multiple sources, reducing risk that any single driver underperforms.`;
  }
};

export async function generateOutpatientPDF(
  inputs: RoiInputs,
  results: ReturnType<typeof calculateRoi>,
  enabledDrivers: LeverId[],
  driverValues: Record<LeverId, number>,
  careSettingLabel: string = "Outpatient",
  organizationName?: string
) {
  const blob = await pdf(
    <OutpatientPDFDocument
      inputs={inputs}
      results={results}
      enabledDrivers={enabledDrivers}
      driverValues={driverValues}
      careSettingLabel={careSettingLabel}
      organizationName={organizationName}
    />
  ).toBlob();
  
  const today = new Date().toISOString().split("T")[0];
  const orgSlug = organizationName ? organizationName.replace(/\s+/g, "-").substring(0, 20) : "";
  const filename = orgSlug 
    ? `abridge-roi-${careSettingLabel.toLowerCase()}-${orgSlug}-${today}.pdf`
    : `abridge-roi-${careSettingLabel.toLowerCase()}-${today}.pdf`;
  
  saveAs(blob, filename);
}

export default OutpatientPDFDocument;

HOW TO USE THIS:
In your results page, import and call:
tsxCopyimport { generateOutpatientPDF } from "@/components/OutpatientPDFExport";

// When user clicks export:
const handleExportPDF = async () => {
  setIsExporting(true);
  try {
    await generateOutpatientPDF(
      inputs,
      results,
      enabledDrivers,
      driverValues,
      "Outpatient",
      organizationName // optional
    );
    toast({ title: "PDF Generated", description: "Your ROI Assessment has been downloaded." });
  } catch (error) {
    console.error("Failed to generate PDF:", error);
    toast({ title: "Export Failed", variant: "destructive" });
  } finally {
    setIsExporting(false);
  }
};
