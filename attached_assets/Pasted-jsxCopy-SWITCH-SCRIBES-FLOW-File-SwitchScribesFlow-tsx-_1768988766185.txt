import { useState } from 'react';
import SwitchScribesSetup from './SwitchScribesSetup';
import SwitchScribesCoverage from './SwitchScribesCoverage';
import SwitchScribesHiddenCosts from './SwitchScribesHiddenCosts';
import SwitchScribesFullPicture from './SwitchScribesFullPicture';
import SwitchScribesComparison from './SwitchScribesComparison';

export default function SwitchScribesFlow({ onBack }) {
  const [currentStep, setCurrentStep] = useState(1);
  const [scribeData, setScribeData] = useState({
    // Organization
    totalProviders: null,
    annualEncounters: null,
    
    // Scribe program
    providersWithScribes: null,
    hourlyRate: 22,
    hoursPerWeek: 32,
    
    // Hidden costs
    turnoverRate: 35,
    managementHoursPerWeek: 6,
    managementHourlyRate: 60,
    recruitmentCostPerScribe: 4000
  });
  
  const totalSteps = 5;
  
  const goNext = () => setCurrentStep(prev => Math.min(prev + 1, totalSteps));
  const goBack = () => {
    if (currentStep === 1) {
      onBack();
    } else {
      setCurrentStep(prev => prev - 1);
    }
  };
  
  // Calculations
  const calculations = {
    // Coverage
    coveragePercent: scribeData.totalProviders && scribeData.providersWithScribes
      ? Math.round((scribeData.providersWithScribes / scribeData.totalProviders) * 100)
      : 0,
    providersWithoutScribes: scribeData.totalProviders && scribeData.providersWithScribes
      ? scribeData.totalProviders - scribeData.providersWithScribes
      : 0,
    
    // Annual scribe payroll
    annualScribePayroll: scribeData.providersWithScribes && scribeData.hourlyRate && scribeData.hoursPerWeek
      ? scribeData.providersWithScribes * scribeData.hourlyRate * scribeData.hoursPerWeek * 50
      : 0,
    
    // Hidden costs
    turnoverCost: scribeData.providersWithScribes && scribeData.turnoverRate && scribeData.recruitmentCostPerScribe
      ? Math.round(scribeData.providersWithScribes * (scribeData.turnoverRate / 100)) * scribeData.recruitmentCostPerScribe
      : 0,
    managementCost: scribeData.managementHoursPerWeek && scribeData.managementHourlyRate
      ? scribeData.managementHoursPerWeek * scribeData.managementHourlyRate * 50
      : 0,
    
    // Totals
    get totalHiddenCosts() {
      return this.turnoverCost + this.managementCost;
    },
    get totalInvestment() {
      return this.annualScribePayroll + this.totalHiddenCosts;
    },
    get costPerCoveredProvider() {
      return scribeData.providersWithScribes
        ? Math.round(this.totalInvestment / scribeData.providersWithScribes)
        : 0;
    },
    get monthlySpend() {
      return Math.round(this.totalInvestment / 12);
    },
    get fullCoverageCost() {
      return scribeData.totalProviders && scribeData.providersWithScribes
        ? Math.round((this.annualScribePayroll / scribeData.providersWithScribes) * scribeData.totalProviders)
        : 0;
    },
    get threeYearInvestment() {
      return this.totalInvestment * 3;
    }
  };
  
  return (
    <div className="switch-scribes-flow">
      {currentStep === 1 && (
        <SwitchScribesSetup
          data={scribeData}
          setData={setScribeData}
          calculations={calculations}
          currentStep={currentStep}
          totalSteps={totalSteps}
          onNext={goNext}
          onBack={goBack}
        />
      )}
      {currentStep === 2 && (
        <SwitchScribesCoverage
          data={scribeData}
          calculations={calculations}
          currentStep={currentStep}
          totalSteps={totalSteps}
          onNext={goNext}
          onBack={goBack}
        />
      )}
      {currentStep === 3 && (
        <SwitchScribesHiddenCosts
          data={scribeData}
          setData={setScribeData}
          calculations={calculations}
          currentStep={currentStep}
          totalSteps={totalSteps}
          onNext={goNext}
          onBack={goBack}
        />
      )}
      {currentStep === 4 && (
        <SwitchScribesFullPicture
          data={scribeData}
          calculations={calculations}
          currentStep={currentStep}
          totalSteps={totalSteps}
          onNext={goNext}
          onBack={goBack}
        />
      )}
      {currentStep === 5 && (
        <SwitchScribesComparison
          data={scribeData}
          calculations={calculations}
          currentStep={currentStep}
          totalSteps={totalSteps}
          onBack={goBack}
        />
      )}
    </div>
  );
}

import { useState } from 'react';

export const formatNumber = (value) => {
  if (value === null || value === undefined || value === '') return '—';
  return Number(value).toLocaleString();
};

export const NumberInput = ({ 
  value, 
  onChange, 
  placeholder, 
  className = '' 
}) => {
  const [displayValue, setDisplayValue] = useState(
    value ? formatNumber(value) : ''
  );
  
  const handleChange = (e) => {
    const rawValue = e.target.value.replace(/,/g, '');
    setDisplayValue(e.target.value);
    onChange(rawValue ? Number(rawValue) : null);
  };
  
  const handleBlur = () => {
    if (value) {
      setDisplayValue(formatNumber(value));
    }
  };
  
  const handleFocus = () => {
    if (value) {
      setDisplayValue(value.toString());
    }
  };
  
  return (
    <input
      type="text"
      inputMode="numeric"
      value={displayValue}
      onChange={handleChange}
      onBlur={handleBlur}
      onFocus={handleFocus}
      placeholder={placeholder}
      className={className}
    />
  );
};
