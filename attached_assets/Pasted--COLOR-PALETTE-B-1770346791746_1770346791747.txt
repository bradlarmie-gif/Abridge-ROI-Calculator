wrvuScore = (currentWRVULift / 5.5) * 100
satisfactionScore = (currentSatisfaction / 88) * 100
valueRealizationScore = (utilizationScore + timeScore + wrvuScore + satisfactionScore) / 4

// Documentation Burden
utilizationGap = 76 - currentUtilization // percentage points
encountersWithoutAI = annualEncounters * (utilizationGap / 100)
documentationBurdenHours = (encountersWithoutAI * benchmarkTimeSaved) / 60
workWeeksEquivalent = documentationBurdenHours / 40

// Utilization Gap
encountersNotUsingAI = annualEncounters * ((76 - currentUtilization) / 100)

// Pajama Time

CALCULATIONS
═══════════════════════════════════════════════════════════════════════════════════

// Total Annual Gap
utilizationGapValue = (encountersWithoutAI * benchmarkTimeSaved / 60) * hourlyRate * timeConversion
efficiencyGapValue = (encountersWithAI * timeGapPerEncounter / 60) * hourlyRate * timeConversion
qualityGapValue = (baseWRVUs * wrvuLiftGap) * wrvuRate * wrvuRealization
totalAnnualGap = utilizationGapValue + efficiencyGapValue + qualityGapValue

// Compounding Effect
// Assumes: Optimized path has 10% year-over-year improvement
// Assumes: Current path has 10% natural improvement rate
// Year 0 starts at 0 (implementation begins)

optimizedYear1 = totalAnnualGap * 0.87 // 87% capture in year 1 (ramp-up)
optimizedYear2 = optimizedYear1 + (totalAnnualGap * 1.10) // 110% in year 2
optimizedYear3 = optimizedYear2 + (totalAnnualGap * 1.15) // 115% in year 3

currentYear1 = totalAnnualGap * 0.10 // 10% natural improvement
currentYear2 = currentYear1 + (totalAnnualGap * 0.11) // continuing slow improvement
currentYear3 = currentYear2 + (totalAnnualGap * 0.12)

// Cost of Waiting
wait6Months = optimizedYear3 - (totalAnnualGap * 0.5) // lose 6 months of year 1
wait12Months = optimizedYear3 - totalAnnualGap // lose full year 1

monthlyOpportunityCost = totalAnnualGap / 12
