  ## 3. App.tsx — replace applyIntakeToExploreState entirely

  function applyIntakeToExploreState(base: ExploreState, intake: ExploreIntakeResponse): ExploreState {
    const primarySetting = intake.settings[0] ?? null;

    const providers =
      primarySetting === 'outpatient' ? intake.opProviders :
      primarySetting === 'ed'         ? intake.edProviders :
      primarySetting === 'inpatient'  ? intake.ipProviders :
      primarySetting === 'nursing'    ? intake.nursingFTEs : null;

    const encounters =
      primarySetting === 'outpatient' ? intake.opAnnualEncounters :
      primarySetting === 'ed'         ? intake.edAnnualVisits :
      primarySetting === 'inpatient'  ? intake.ipAnnualAdmissions : null;

    return {
      ...base,
      careSetting: primarySetting,
      numberOfProviders: providers ?? base.numberOfProviders,
      annualEncounters: encounters ?? base.annualEncounters,
      nursingStaffedBeds: intake.nursingStaffedBeds ?? base.nursingStaffedBeds,
      nursingOccupancyRate: intake.nursingOccupancyRate ?? base.nursingOccupancyRate,

      docQualityInputs: {
        ...base.docQualityInputs,
        ...(intake.opCurrentWrvu != null        && { currentWrvu: intake.opCurrentWrvu }),
        ...(intake.opConversionFactor != null   && { conversionFactor: intake.opConversionFactor }),
        ...(intake.opDenialRate != null         && { denialRate: intake.opDenialRate }),
        ...(intake.opAvgClaimValue != null      && { avgClaimValue: intake.opAvgClaimValue }),
        ...(intake.opPanelSize != null          && { panelSize: intake.opPanelSize }),
        ...(intake.opMaEnrollmentRate != null   && { maEnrollmentRate: intake.opMaEnrollmentRate }),
        ...(intake.opAnnualPaymentPerRaf != null && { annualPayment: intake.opAnnualPaymentPerRaf }),
        ...(intake.ipDenialRate != null         && { ipObsDefenseDenialRate: intake.ipDenialRate }),
        ...(intake.ipAvgClaimValue != null      && { ipObsDefenseClaimValue: intake.ipAvgClaimValue }),
        ...(intake.hapiRatePer1000 != null      && { nursingHapiRate: intake.hapiRatePer1000 }),
        ...(intake.fallRatePer1000 != null      && { nursingFallsRate: intake.fallRatePer1000 }),
        ...(intake.cautiRatePer1000 != null     && { nursingCautiRate: intake.cautiRatePer1000 }),
        ...(intake.clabsiRatePer1000 != null    && { nursingClabsiRate: intake.clabsiRatePer1000 }),
      },

      timeDriverInputs: {
        ...base.timeDriverInputs,
        ...(intake.opRevenuePerVisit != null    && { revenuePerVisit: intake.opRevenuePerVisit }),
        ...(intake.opTurnoverRate != null       && { annualTurnoverRate: intake.opTurnoverRate }),
        ...(intake.opReplacementCost != null    && { replacementCost: intake.opReplacementCost }),
        ...(intake.edLwbsRate != null           && { edLwbsRate: intake.edLwbsRate }),
        ...(intake.edRevenuePerVisit != null    && { edRevenuePerVisit: intake.edRevenuePerVisit }),
        ...(intake.edAdmissionRate != null      && { edAdmissionRate: intake.edAdmissionRate }),
        ...(intake.edAdmissionRevenue != null   && { edAdmissionRevenue: intake.edAdmissionRevenue }),
        ...(intake.edTurnoverRate != null       && { annualTurnoverRate: intake.edTurnoverRate }),
        ...(intake.edReplacementCost != null    && { replacementCost: intake.edReplacementCost }),
        ...(intake.ipTurnoverRate != null       && { ipAnnualTurnoverRate: intake.ipTurnoverRate }),
        ...(intake.ipReplacementCost != null    && { replacementCost: intake.ipReplacementCost }),
        ...(intake.nursingOtHoursPerWeek != null  && { nursingOtHoursPerNurseWeek: intake.nursingOtHoursPerWeek }),
        ...(intake.nursingOtHourlyRate != null    && { nursingOtHourlyRate: intake.nursingOtHourlyRate }),
        ...(intake.nursingTurnoverRate != null    && { nursingTurnoverRate: intake.nursingTurnoverRate }),
        ...(intake.nursingReplacementCost != null && { nursingReplacementCost: intake.nursingReplacementCost }),
      },
    };
