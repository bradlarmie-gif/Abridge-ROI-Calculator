import { useState } from 'react';
import { formatCurrency, formatNumber } from '../../utils/formatting';

const Step3_RealityCheck = ({ baseline, inputs, onUpdate, onContinue, onBack }) => {
  const activeDrivers = Object.entries(baseline.benefits || {})
    .filter(([key, value]) => value > 0)
    .map(([key]) => key);

  const [validations, setValidations] = useState(
    inputs.driverValidations || 
    activeDrivers.reduce((acc, driver) => ({
      ...acc,
      [driver]: getDefaultValidation(driver, baseline)
    }), {})
  );

  const [currentDriverIndex, setCurrentDriverIndex] = useState(0);
  const currentDriver = activeDrivers[currentDriverIndex];

  function getDefaultValidation(driver, baseline) {
    switch(driver) {
      case 'access':
        return {
          hasCapacity: true,
          sameDemand: true,
          additionalVisitsPerWeek: baseline.benefits.access 
            ? (baseline.benefits.access / baseline.providers / 48 / (baseline.revenuePerVisit || 200))
            : 2.5
        };
      case 'retention':
        return {
          sameTurnoverRisk: true,
          turnoverReduction: 1.6
        };
      case 'los':
        return {
          sameCaseMix: true,
          wrvuUplift: baseline.benefits.los 
            ? (baseline.benefits.los / baseline.providers / (baseline.revenuePerWRVU || 40))
            : 68
        };
      case 'locum':
        return {
          sameStaffingModel: true
        };
      case 'denials':
        return {
          sameDenialRisk: true
        };
      case 'hcc':
        return {
          sameMAPop: true
        };
      default:
        return { inheritsFromBaseline: true };
    }
  }

  const updateValidation = (driver, updates) => {
    setValidations(prev => ({
      ...prev,
      [driver]: { ...prev[driver], ...updates }
    }));
  };

  const handleNext = () => {
    if (currentDriverIndex < activeDrivers.length - 1) {
      setCurrentDriverIndex(currentDriverIndex + 1);
    } else {
      onUpdate({ driverValidations: validations });
      onContinue();
    }
  };

  const handlePrevious = () => {
    if (currentDriverIndex > 0) {
      setCurrentDriverIndex(currentDriverIndex - 1);
    } else {
      onBack();
    }
  };

  const renderDriverValidation = () => {
    const driver = currentDriver;
    const validation = validations[driver];
    const baselineValue = baseline.benefits[driver];
    const valuePerProvider = baselineValue / baseline.providers;

    switch(driver) {
      case 'access':
        return (
          <AccessValidation
            baseline={baseline}
            validation={validation}
            onUpdate={(updates) => updateValidation(driver, updates)}
            valuePerProvider={valuePerProvider}
          />
        );
      case 'retention':
        return (
          <RetentionValidation
            baseline={baseline}
            validation={validation}
            onUpdate={(updates) => updateValidation(driver, updates)}
            valuePerProvider={valuePerProvider}
          />
        );
      case 'los':
        return (
          <LosValidation
            baseline={baseline}
            validation={validation}
            onUpdate={(updates) => updateValidation(driver, updates)}
            valuePerProvider={valuePerProvider}
          />
        );
      default:
        return (
          <DefaultValidation
            driver={driver}
            baseline={baseline}
            validation={validation}
            valuePerProvider={valuePerProvider}
          />
        );
    }
  };

  const progress = ((currentDriverIndex + 1) / activeDrivers.length) * 100;

  return (
    <div className="step-container reality-check">
      <div className="step-header">
        <h1>Reality Check</h1>
        <p className="subtitle">
          Let's make sure our expansion model reflects your actual situation. 
          We'll go through each value driver to validate assumptions.
        </p>
      </div>

      {/* Progress through drivers */}
      <div className="driver-progress">
        <div className="progress-bar-container">
          <div className="progress-bar-fill" style={{ width: `${progress}%` }}></div>
        </div>
        <div className="progress-text">
          Step {currentDriverIndex + 1} of {activeDrivers.length}: {getDriverName(currentDriver)}
        </div>
      </div>

      {/* Driver Validation Content */}
      <div className="validation-content">
        {renderDriverValidation()}
      </div>

      {/* Navigation */}
      <div className="step-actions">
        <button className="btn-secondary" onClick={handlePrevious}>
          ← {currentDriverIndex === 0 ? 'Back' : 'Previous Driver'}
        </button>
        <button className="btn-primary" onClick={handleNext}>
          {currentDriverIndex < activeDrivers.length - 1 
            ? 'Next Driver →' 
            : 'Show My Model →'}
        </button>
      </div>
    </div>
  );
};

// Helper functions
function getDriverName(driver) {
  const names = {
    access: 'Patient Access',
    retention: 'Clinician Retention',
    los: 'Level of Service',
    locum: 'Overtime & Locum',
    denials: 'Denial Prevention',
    hcc: 'HCC Capture'
  };
  return names[driver] || driver;
}

function getDriverScalingExplanation(driver, model, inputs) {
  const explanations = {
    access: `Scales with encounter volume and capacity constraints. Validated that your new providers will have similar patient demand.`,
    retention: `Accounts for 12-18 month lag effect. Year 1: 20%, Year 2: 70%, Year 3: 100% of full value as providers reach retention decision points.`,
    los: `Scales proportionally with encounter volume, assuming similar case mix to current providers.`,
    locum: `Scales with provider count, with slight efficiency gains at scale (90% multiplier).`,
    denials: `Scales with encounter volume, assuming similar denial risk profile.`,
    hcc: `Scales with encounter volume, assuming similar MA patient population percentage.`
  };
  return explanations[driver] || 'Scales proportionally based on proven baseline performance.';
}
