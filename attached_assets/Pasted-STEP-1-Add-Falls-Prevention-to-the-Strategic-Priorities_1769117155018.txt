STEP 3: JavaScript calculation logic
javascriptCopy// Falls Prevention Calculator
function calculateFallsPrevention() {
  // Step 1: Current falls volume
  const annualAdmissions = parseInt(document.getElementById('falls-annual-admissions').value) || 10000;
  const fallsRate = parseFloat(document.getElementById('falls-rate').value) || 3.5;
  
  // Estimate patient days (avg 4-day LOS)
  const avgLOS = 4;
  const patientDays = annualAdmissions * avgLOS;
  const fallsPerYear = Math.round((patientDays / 1000) * fallsRate);
  
  // Step 2: Documentation-preventable
  const preventionRate = parseFloat(document.getElementById('falls-prevention-rate').value) || 8;
  const fallsPrevented = Math.round(fallsPerYear * (preventionRate / 100));
  
  // Step 3: Cost avoidance
  const costPerFall = parseInt(document.getElementById('falls-cost-per-fall').value) || 6500;
  const potentialValue = fallsPrevented * costPerFall;
  
  // Update displays
  document.getElementById('falls-per-year').textContent = fallsPerYear.toLocaleString();
  document.getElementById('falls-current-display').textContent = fallsPerYear;
  document.getElementById('falls-prevented').textContent = fallsPrevented;
  document.getElementById('falls-prevented-display').textContent = fallsPrevented;
  document.getElementById('falls-potential-value').textContent = '$' + potentialValue.toLocaleString();
  document.getElementById('falls-summary-value').textContent = '$' + potentialValue.toLocaleString();
  document.getElementById('falls-total-value').textContent = '$' + Math.round(potentialValue / 1000) + 'K';
  
  // Update live model
  updateLiveModel('falls-prevention', potentialValue);
  
  return {
    fallsPerYear,
    fallsPrevented,
    potentialValue
  };
}

// Add event listeners
document.getElementById('falls-annual-admissions')?.addEventListener('input', calculateFallsPrevention);
document.getElementById('falls-rate')?.addEventListener('input', calculateFallsPrevention);
document.getElementById('falls-prevention-rate')?.addEventListener('input', calculateFallsPrevention);
document.getElementById('falls-cost-per-fall')?.addEventListener('input', calculateFallsPrevention);

function updateLiveModel(driver, value) {
  const liveModelList = document.getElementById('live-model-list');
