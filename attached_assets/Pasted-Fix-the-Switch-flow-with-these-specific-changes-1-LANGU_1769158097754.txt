6. UPDATE GAP CALCULATIONS
javascriptCopyfunction calculateGap(providers, encounters, utilization, timeSaved, wrvuLift) {
  // Utilization gap (now vs 75%)
  const utilizationGapPP = Math.max(0, 75 - utilization);
  const utilizationGapEncounters = Math.round(encounters * (utilizationGapPP / 100));
  const utilizationGapValue = utilizationGapEncounters * 4;
  
  // Efficiency gap
  const efficiencyGapMin = Math.max(0, 4 - timeSaved);
  const encountersAtBenchmark = encounters * (75 / 100);
  const efficiencyGapHours = Math.round((encountersAtBenchmark * efficiencyGapMin) / 60);
  const efficiencyGapValue = Math.round(efficiencyGapHours * 150 * 0.20);
  
  // wRVU gap (NEW)
  const wrvuGapPercent = Math.max(0, 5 - wrvuLift);
  const avgWRVU = 1.5; // Assume 1.5 wRVU/encounter baseline
  const wrvuGapPerEncounter = avgWRVU * (wrvuGapPercent / 100);
  const wrvuGapValue = Math.round(wrvuGapPerEncounter * encountersAtBenchmark * 33 * 0.5);
  
  // Capture rates
  const utilizationCapture = Math.min(100, Math.round((utilization / 75) * 100));
  const efficiencyCapture = Math.min(100, Math.round((timeSaved / 4) * 100));
  const wrvuCapture = Math.min(100, Math.round((wrvuLift / 5) * 100));
  const captureRate = Math.round((utilizationCapture/100) * (efficiencyCapture/100) * (wrvuCapture/100) * 100);
  
  // Totals
  const annualGap = utilizationGapValue + efficiencyGapValue + wrvuGapValue;
  
  return {
    utilizationGapValue,
    efficiencyGapValue,
    wrvuGapValue,
    utilizationCapture,
    efficiencyCapture,
    wrvuCapture,
    captureRate,
    annualGap,
    // ... rest
  };
}
