  activeNurses       = nurseFTEs × adoptionRate
  hoursPerActiveNurse = Math.round(totalHoursReturned / activeNurses)
  minsPerShift        = minutesSavedPerShift  [direct passthrough]
  NOTE: divide by active (Abridge-using) nurses only, not total FTEs.
  If adoptionRate = 0.5 and nurseFTEs = 300, activeNurses = 150.
