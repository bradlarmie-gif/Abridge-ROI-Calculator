  Add Deployment Profile Section to MeasureDataRequest

  MeasureDataRequest.tsx is the client-facing form customers fill out before an EBR. It currently only collects before/after metric values. It's missing the deployment/partner profile section — org name,
  months on Abridge, provider adoption funnel, encounter coverage — which is essential context for the business review.

  Update decodeDataRequest validation to accept responses where deployment may be absent (for backward compat — treat it as an empty object):

  export function decodeDataRequest(encoded: string): MeasureDataRequestResponse | null {
    try {
      const decompressed = LZString.decompressFromEncodedURIComponent(encoded);
      if (!decompressed) return null;
      const parsed = JSON.parse(decompressed);
      if (!parsed?.setting || !Array.isArray(parsed.metrics)) return null;
      if (!parsed.deployment) parsed.deployment = {
        organizationName: '', monthsOnAbridge: 0, totalProviders: 0,
        liveProviders: 0, mruProviders: 0, totalEncounters: 0, abridgeEncounters: 0,
      };
      return parsed as MeasureDataRequestResponse;
    } catch {
      return null;
    }
  }

  Update generateDataResponseText to include a deployment header block before the metrics:

  export function generateDataResponseText(data: MeasureDataRequestResponse, metricLabels: Record<string, string>): string {
    const dep = data.deployment;
    const lines: string[] = [
      `ABRIDGE — MEASURE DATA REQUEST (${SETTING_LABELS[data.setting]})`,
      '',
    ];

    if (dep.organizationName) lines.push(`  Organization: ${dep.organizationName}`);
    if (dep.monthsOnAbridge) lines.push(`  Months on Abridge: ${dep.monthsOnAbridge}`);
    if (dep.totalProviders) lines.push(`  Total providers: ${dep.totalProviders}`);
    if (dep.liveProviders) lines.push(`  Live on Abridge: ${dep.liveProviders}`);
    if (dep.mruProviders) lines.push(`  Monthly recording users: ${dep.mruProviders}`);
    if (dep.totalEncounters) lines.push(`  Total encounters: ${dep.totalEncounters.toLocaleString('en-US')}`);
    if (dep.abridgeEncounters) lines.push(`  Abridge encounters: ${dep.abridgeEncounters.toLocaleString('en-US')}`);
    lines.push('');

    // ... rest of metrics loop unchanged
  }

  Add a updateDeployment helper:
  function updateDeployment(updates: Partial<DeploymentSnapshot>) {
    setDeployment(prev => {
      const next = { ...prev, ...updates };
      // Auto-calculate months from go-live date
      if (updates.goLiveDate !== undefined) {
        if (updates.goLiveDate) {
          const goLive = new Date(updates.goLiveDate);
          const now = new Date();
          const diff = (now.getFullYear() - goLive.getFullYear()) * 12 + (now.getMonth() - goLive.getMonth());
          next.monthsOnAbridge = Math.max(0, diff);
        }
      }
      return next;
    });
  }
