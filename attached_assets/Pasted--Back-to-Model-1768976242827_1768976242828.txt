  setting,
  currentUnits, // providers or beds
  unitName, // "providers" or "staffed beds"
  totalAnnualValue,
  annualInvestment,
  valueBreakdown, // array of { category, name, value }
  pricePerUnit,
  contractTerm
}) {
  
  // Expansion modeling state
  const [expandedUnits, setExpandedUnits] = useState(currentUnits * 5);
  
  // Calculations
  const netValue = totalAnnualValue - annualInvestment;
  const roiMultiple = (totalAnnualValue / annualInvestment).toFixed(1);
  const paybackMonths = Math.round((annualInvestment / totalAnnualValue) * 12);
  
  // Value per unit (for scaling)
  const valuePerUnit = totalAnnualValue / currentUnits;
  const investmentPerUnit = annualInvestment / currentUnits;
  
  // Expanded calculations
  const expandedValue = valuePerUnit * expandedUnits;
  const expandedInvestment = investmentPerUnit * expandedUnits;
  const expandedNet = expandedValue - expandedInvestment;
  const expansionDelta = expandedNet - netValue;
  
  // Chart data
  const scaleData = [
    { units: currentUnits, value: totalAnnualValue, isCurrent: true },
    { units: Math.round(currentUnits * 2), value: valuePerUnit * currentUnits * 2, isCurrent: false },
    { units: Math.round(currentUnits * 3), value: valuePerUnit * currentUnits * 3, isCurrent: false },
    { units: Math.round(currentUnits * 4), value: valuePerUnit * currentUnits * 4, isCurrent: false },
    { units: Math.round(currentUnits * 5), value: valuePerUnit * currentUnits * 5, isCurrent: false },
    { units: expandedUnits, value: expandedValue, isExpanded: true },
  ].sort((a, b) => a.units - b.units);
  
  // Multi-year projection (10% growth)
  const yearlyGrowth = 1.10;
  const year1Value = totalAnnualValue;
  const year2Value = totalAnnualValue * yearlyGrowth;
  const year3Value = totalAnnualValue * yearlyGrowth * yearlyGrowth;
  
  // Group value breakdown by category
  const laborValue = valueBreakdown
    .filter(d => d.category === 'labor')
    .reduce((sum, d) => sum + d.value, 0);
  const revenueValue = valueBreakdown
    .filter(d => d.category === 'revenue')
    .reduce((sum, d) => sum + d.value, 0);
  
  const laborPercent = Math.round((laborValue / totalAnnualValue) * 100);
  const revenuePercent = 100 - laborPercent;
  
  return (
    <div className="summary-page">

            <span className="divider">│</span>
            <span className="highlight">${netValue.toLocaleString()} net value</span>
            <span className="divider">│</span>
            <span className="highlight">{roiMultiple}x ROI</span>
          </div>
        </div>
        <div className="header-actions">
          <button className="secondary-button">✏️ Edit Model</button>
          <button className="primary-button">📤 Export & Share</button>
        </div>
      </div>
      
      {/* Hero Metrics */}
      <div className="hero-section">
        <h2>YOUR ROI AT A GLANCE</h2>
        <div className="hero-metrics">
          <div className="hero-metric">
            <div className="metric-value green">${netValue.toLocaleString()}</div>
            <div className="metric-label">NET ANNUAL VALUE</div>
          </div>
          <div className="hero-metric">
            <div className="metric-value">{roiMultiple}x</div>
            <div className="metric-label">RETURN ON INVESTMENT</div>
          </div>
          <div className="hero-metric">
            <div className="metric-value">{paybackMonths < 1 ? '<1' : paybackMonths} mo</div>
            <div className="metric-label">PAYBACK PERIOD</div>
          </div>
        </div>
        <div className="hero-subtext">
          <span>📈 Total Annual Value: ${totalAnnualValue.toLocaleString()}</span>
          <span className="divider">│</span>
          <span>💰 Annual Investment: ${annualInvestment.toLocaleString()}</span>
        </div>
      </div>

      <div className="scale-section">

        </div>
        
        <div className="scale-content">
          {/* Slider */}
          <div className="expansion-slider">
            <div className="slider-labels">
              <span>Current: {currentUnits} {unitName}</span>
              <span>Expanded: {expandedUnits} {unitName}</span>
            </div>
            <input
