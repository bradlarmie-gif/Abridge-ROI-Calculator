NAVIGATION JAVASCRIPT
javascriptCopy// Add this to your main app.js

// Navigate home when logo clicked
function navigateHome(event) {
  event.preventDefault();
  
  // If you're using a single-page app
  window.location.hash = '';
  showLandingPage();
  
  // OR if you have separate pages
  // window.location.href = '/';
}

// Make logo clickable on page load
document.addEventListener('DOMContentLoaded', function() {
  const logoLink = document.querySelector('.logo-link');
  if (logoLink) {
    logoLink.addEventListener('click', navigateHome);
  }
});
