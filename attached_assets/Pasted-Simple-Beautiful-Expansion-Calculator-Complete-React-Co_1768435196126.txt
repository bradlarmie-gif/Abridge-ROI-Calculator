import { useState, useEffect } from 'react';
import './ExpandProviders.css';

const ExpandProviders = ({ baselineData, onBack, onSaveScenario }) => {
  // Baseline from existing deployment
  const [baseline] = useState({
    providers: baselineData.providers || 50,
    encounters: baselineData.encounters || 100000,
    utilization: baselineData.utilization || 0.65,
    costPerProvider: baselineData.costPerProvider || 120,
    totalBenefit: baselineData.totalBenefit || 444167,
    benefits: baselineData.benefits || {
      access: 216667,
      retention: 100000,
      los: 227500,
      locum: 29060,
      denials: 97500,
      hcc: 438000
    }
  });

  // Expansion inputs
  const [targetProviders, setTargetProviders] = useState(baseline.providers);
  const [pricingModel, setPricingModel] = useState('per-provider'); // 'per-provider' or 'enterprise'
  const [enterpriseAnnualCost, setEnterpriseAnnualCost] = useState(null);
  const [utilizationAssumption, setUtilizationAssumption] = useState(baseline.utilization);
  
  // Auto-calculate based on user's target
  const [expansion, setExpansion] = useState(null);

  // Calculate expansion whenever inputs change
  useEffect(() => {
    if (targetProviders <= baseline.providers) {
      setExpansion(null);
      return;
    }

    const result = calculateExpansion();
    setExpansion(result);
  }, [targetProviders, pricingModel, enterpriseAnnualCost, utilizationAssumption]);

  const calculateExpansion = () => {
    // Scaling ratios
    const providerRatio = targetProviders / baseline.providers;
    const encounterRatio = providerRatio; // Assumes same encounters per provider
    
    // SCENARIO CALCULATIONS
    const scenarioEncounters = Math.round(baseline.encounters * encounterRatio);
    
    // Cost calculation based on pricing model
    let scenarioCost;
    let effectiveCostPerProvider;
    let volumeDiscount = 0;
    
    if (pricingModel === 'per-provider') {
      scenarioCost = targetProviders * baseline.costPerProvider * 12;
      effectiveCostPerProvider = baseline.costPerProvider;
    } else {
      // Enterprise pricing
      if (enterpriseAnnualCost && enterpriseAnnualCost > 0) {
        scenarioCost = enterpriseAnnualCost;
        effectiveCostPerProvider = scenarioCost / (targetProviders * 12);
      } else {
        // Auto-calculate with volume discount
        volumeDiscount = getVolumeDiscount(targetProviders);
        effectiveCostPerProvider = baseline.costPerProvider * (1 - volumeDiscount);
        scenarioCost = targetProviders * effectiveCostPerProvider * 12;
      }
    }
    
    // Benefits scale proportionally (conservative approach)
    const benefitScaling = {
      access: encounterRatio,
      retention: providerRatio,
      los: encounterRatio,
      locum: providerRatio,
      denials: encounterRatio,
      hcc: encounterRatio
    };
    
    const scenarioBenefits = {
      access: Math.round(baseline.benefits.access * benefitScaling.access),
      retention: Math.round(baseline.benefits.retention * benefitScaling.retention),
      los: Math.round(baseline.benefits.los * benefitScaling.los),
      locum: Math.round(baseline.benefits.locum * benefitScaling.locum),
      denials: Math.round(baseline.benefits.denials * benefitScaling.denials),
      hcc: Math.round(baseline.benefits.hcc * benefitScaling.hcc)
    };
    
    const scenarioTotalBenefit = Object.values(scenarioBenefits).reduce((a, b) => a + b, 0);
    const scenarioNetGain = scenarioTotalBenefit - scenarioCost;
    const scenarioROI = scenarioTotalBenefit / scenarioCost;
    
    // BASELINE CALCULATIONS (for comparison)
    const baselineCost = baseline.providers * baseline.costPerProvider * 12;
    const baselineNetGain = baseline.totalBenefit - baselineCost;
    const baselineROI = baseline.totalBenefit / baselineCost;
    
    // INCREMENTAL ANALYSIS (the key!)
    const incrementalProviders = targetProviders - baseline.providers;
    const incrementalCost = scenarioCost - baselineCost;
    const incrementalBenefit = scenarioTotalBenefit - baseline.totalBenefit;
    const incrementalNetGain = incrementalBenefit - incrementalCost;
    const incrementalROI = incrementalBenefit / incrementalCost;
    
    return {
      baseline: {
        providers: baseline.providers,
        encounters: baseline.encounters,
        cost: baselineCost,
        benefit: baseline.totalBenefit,
        netGain: baselineNetGain,
        roi: baselineROI
      },
      scenario: {
        providers: targetProviders,
        encounters: scenarioEncounters,
        cost: scenarioCost,
        costPerProvider: effectiveCostPerProvider,
        benefit: scenarioTotalBenefit,
        benefits: scenarioBenefits,
        netGain: scenarioNetGain,
        roi: scenarioROI,
        volumeDiscount
      },
      incremental: {
        providers: incrementalProviders,
        encounters: scenarioEncounters - baseline.encounters,
        cost: incrementalCost,
        benefit: incrementalBenefit,
        netGain: incrementalNetGain,
        roi: incrementalROI
      }
    };
  };

  const getVolumeDiscount = (providers) => {
    if (providers >= 250) return 0.30;
    if (providers >= 150) return 0.25;
    if (providers >= 100) return 0.20;
    if (providers >= 75) return 0.15;
    if (providers >= 50) return 0.10;
    return 0;
  };

  const formatCurrency = (value) => {
    return new Intl.NumberFormat('en-US', {
      style: 'currency',
      currency: 'USD',
      minimumFractionDigits: 0,
      maximumFractionDigits: 0,
    }).format(value);
  };

  const formatNumber = (value) => {
    return new Intl.NumberFormat('en-US').format(value);
  };

  return (
    <div className="expand-providers-container">
      
      {/* Header */}
      <div className="expansion-header">
        <button className="back-button" onClick={onBack}>
          ← Back to Scenario Builder
        </button>
        
        <div className="header-content">
          <h1>Expansion Calculator</h1>
          <p className="subtitle">Model the ROI of expanding Abridge to more providers</p>
        </div>
        
        <div className="baseline-summary">
          <span className="label">Your current deployment:</span>
          <div className="baseline-stats">
            <span className="stat">{baseline.providers} providers</span>
            <span className="divider">•</span>
            <span className="stat">{formatNumber(baseline.encounters)} encounters/year</span>
            <span className="divider">•</span>
            <span className="stat">{formatCurrency(baseline.totalBenefit - (baseline.providers * baseline.costPerProvider * 12))} net gain</span>
          </div>
        </div>
      </div>

      {/* Main Content - Two Column Layout */}
      <div className="expansion-content">
        
        {/* LEFT COLUMN: Inputs */}
        <div className="inputs-panel">
          
          {/* Provider Count Slider */}
          <div className="input-section primary-input">
            <label className="section-label">
              <span>How many providers do you want to expand to?</span>
            </label>
            
            <div className="provider-input-control">
              <input
                type="number"
                className="provider-number-input"
                value={targetProviders}
                onChange={(e) => setTargetProviders(parseInt(e.target.value) || baseline.providers)}
                min={baseline.providers}
                max={500}
              />
              <span className="input-suffix">providers</span>
            </div>
            
            <input
              type="range"
              className="provider-slider"
              value={targetProviders}
              onChange={(e) => setTargetProviders(parseInt(e.target.value))}
              min={baseline.providers}
              max={Math.max(baseline.providers * 3, 200)}
              step={5}
            />
            
            <div className="slider-labels">
              <span>{baseline.providers}</span>
              <span className="current-marker">
                {targetProviders} 
                {targetProviders > baseline.providers && (
                  <span className="increment-badge">+{targetProviders - baseline.providers}</span>
                )}
              </span>
              <span>{Math.max(baseline.providers * 3, 200)}</span>
            </div>
          </div>

          {/* Pricing Model Toggle */}
          <div className="input-section">
            <label className="section-label">
              <span>Pricing model</span>
            </label>
            
            <div className="pricing-toggle">
              <button
                className={`toggle-option ${pricingModel === 'per-provider' ? 'active' : ''}`}
                onClick={() => setPricingModel('per-provider')}
              >
                <div className="toggle-option-content">
                  <strong>Per Provider</strong>
                  <span className="toggle-sublabel">${baseline.costPerProvider}/provider/month</span>
                </div>
              </button>
              
              <button
                className={`toggle-option ${pricingModel === 'enterprise' ? 'active' : ''}`}
                onClick={() => setPricingModel('enterprise')}
              >
                <div className="toggle-option-content">
                  <strong>Enterprise Annual</strong>
                  <span className="toggle-sublabel">Volume discounts</span>
                </div>
              </button>
            </div>
          </div>

          {/* Enterprise Pricing Input */}
          {pricingModel === 'enterprise' && (
            <div className="input-section enterprise-section">
              <label className="section-label">
                <span>Custom enterprise pricing (optional)</span>
              </label>
              
              <div className="enterprise-input">
                <div className="input-with-prefix">
                  <span className="prefix">$</span>
                  <input
                    type="number"
                    placeholder="Enter annual contract value"
                    value={enterpriseAnnualCost || ''}
                    onChange={(e) => setEnterpriseAnnualCost(parseFloat(e.target.value) || null)}
                  />
                  <span className="suffix">/year</span>
                </div>
              </div>
              
              {expansion && expansion.scenario.volumeDiscount > 0 && !enterpriseAnnualCost && (
                <div className="discount-info">
                  <span className="discount-badge">
                    {(expansion.scenario.volumeDiscount * 100).toFixed(0)}% volume discount applied
                  </span>
                  <p className="discount-note">
                    Standard pricing: {formatCurrency(targetProviders * baseline.costPerProvider * 12)}/year<br />
                    Your price: {formatCurrency(expansion.scenario.cost)}/year
                  </p>
                </div>
              )}
            </div>
          )}

          {/* Advanced: Utilization adjustment */}
          <details className="advanced-section">
            <summary>Advanced settings</summary>
            <div className="advanced-content">
              <label className="section-label">
                <span>Expected utilization for new providers</span>
              </label>
              <input
                type="range"
                min="40"
                max="90"
                value={utilizationAssumption * 100}
                onChange={(e) => setUtilizationAssumption(e.target.value / 100)}
              />
              <span className="utilization-value">{(utilizationAssumption * 100).toFixed(0)}%</span>
              <p className="field-hint">
                New deployments typically start at 40-50% and mature to 75-90%
              </p>
            </div>
          </details>

        </div>

        {/* RIGHT COLUMN: Results */}
        <div className="results-panel">
          
          {!expansion ? (
            <div className="empty-state">
              <div className="empty-icon">📊</div>
              <p>Adjust the provider count to see expansion ROI</p>
            </div>
          ) : (
            <>
              {/* Hero Metric: Incremental ROI */}
              <div className="hero-metric-card">
                <div className="hero-label">Expansion ROI</div>
                <div className="hero-value">
                  {expansion.incremental.roi.toFixed(1)}x
                </div>
                <div className="hero-subtext">
                  Adding {expansion.incremental.providers} providers generates{' '}
                  <strong>{formatCurrency(expansion.incremental.netGain)}</strong> in additional annual value
                </div>
              </div>

              {/* Quick Comparison Table */}
              <div className="comparison-table">
                <div className="table-header">
                  <div className="cell"></div>
                  <div className="cell column-label">Current</div>
                  <div className="cell column-label highlight">Expanded</div>
                  <div className="cell column-label">Change</div>
                </div>
                
                <div className="table-row">
                  <div className="cell row-label">Providers</div>
                  <div className="cell">{expansion.baseline.providers}</div>
                  <div className="cell highlight">{expansion.scenario.providers}</div>
                  <div className="cell change positive">+{expansion.incremental.providers}</div>
                </div>
                
                <div className="table-row">
                  <div className="cell row-label">Annual Investment</div>
                  <div className="cell">{formatCurrency(expansion.baseline.cost)}</div>
                  <div className="cell highlight">{formatCurrency(expansion.scenario.cost)}</div>
                  <div className="cell change">{formatCurrency(expansion.incremental.cost)}</div>
                </div>
                
                <div className="table-row">
                  <div className="cell row-label">Annual Benefit</div>
                  <div className="cell">{formatCurrency(expansion.baseline.benefit)}</div>
                  <div className="cell highlight">{formatCurrency(expansion.scenario.benefit)}</div>
                  <div className="cell change positive">+{formatCurrency(expansion.incremental.benefit)}</div>
                </div>
                
                <div className="table-row summary">
                  <div className="cell row-label strong">Net Gain</div>
                  <div className="cell strong">{formatCurrency(expansion.baseline.netGain)}</div>
                  <div className="cell highlight strong">{formatCurrency(expansion.scenario.netGain)}</div>
                  <div className="cell change positive strong">+{formatCurrency(expansion.incremental.netGain)}</div>
                </div>
                
                <div className="table-row summary">
                  <div className="cell row-label strong">ROI</div>
                  <div className="cell strong">{expansion.baseline.roi.toFixed(1)}x</div>
                  <div className="cell highlight strong">{expansion.scenario.roi.toFixed(1)}x</div>
                  <div className="cell change">
                    {expansion.scenario.roi >= expansion.baseline.roi ? (
                      <span className="positive">+{(expansion.scenario.roi - expansion.baseline.roi).toFixed(1)}x</span>
                    ) : (
                      <span className="neutral">{(expansion.scenario.roi - expansion.baseline.roi).toFixed(1)}x</span>
                    )}
                  </div>
                </div>
              </div>

              {/* Key Insight */}
              <div className="key-insight">
                <div className="insight-icon">💡</div>
                <div className="insight-content">
                  <strong>Key Insight</strong>
                  <p>
                    {expansion.incremental.roi >= expansion.baseline.roi ? (
                      <>
                        Strong expansion economics: Your incremental ROI ({expansion.incremental.roi.toFixed(1)}x) 
                        {expansion.incremental.roi > expansion.baseline.roi ? ' exceeds ' : ' matches '}
                        your baseline ROI ({expansion.baseline.roi.toFixed(1)}x).
                        {pricingModel === 'enterprise' && expansion.scenario.volumeDiscount > 0 && (
                          <> Volume discounts make this expansion even more attractive, saving {formatCurrency(targetProviders * baseline.costPerProvider * 12 - expansion.scenario.cost)}/year vs. standard pricing.</>
                        )}
                      </>
                    ) : (
                      <>
                        While incremental ROI ({expansion.incremental.roi.toFixed(1)}x) is slightly lower than baseline 
                        ({expansion.baseline.roi.toFixed(1)}x), you still generate <strong>{formatCurrency(expansion.incremental.netGain)}</strong> in 
                        additional annual value with positive returns.
                      </>
                    )}
                  </p>
                </div>
              </div>

              {/* 3-Year Projection */}
              <div className="three-year-projection">
                <h4>3-Year Value</h4>
                <div className="projection-comparison">
                  <div className="projection-item">
                    <span className="projection-label">Current (3-year)</span>
                    <span className="projection-value">{formatCurrency(expansion.baseline.netGain * 3)}</span>
                  </div>
                  <div className="projection-arrow">→</div>
                  <div className="projection-item highlight">
                    <span className="projection-label">Expanded (3-year)</span>
                    <span className="projection-value">{formatCurrency(expansion.scenario.netGain * 3)}</span>
                  </div>
                </div>
                <div className="projection-gain">
                  Additional 3-year value: <strong className="positive">{formatCurrency(expansion.incremental.netGain * 3)}</strong>
                </div>
              </div>

              {/* Action Buttons */}
              <div className="action-buttons">
                <button className="secondary-button" onClick={() => setTargetProviders(baseline.providers)}>
                  Reset
                </button>
                <button 
                  className="primary-button"
                  onClick={() => onSaveScenario({
                    name: `Expand to ${targetProviders} providers`,
                    ...expansion
                  })}
                >
                  Save Scenario
                </button>
              </div>
            </>
          )}
          
        </div>
      </div>
    </div>
  );
};

export default ExpandProviders;

import ExpandProviders from './components/ExpandProviders';

function ScenarioBuilder() {
  const [view, setView] = useState('builder'); // 'builder' or 'expand'
  
  const baselineData = {
    providers: 50,
    encounters: 100000,
    utilization: 0.65,
    costPerProvider: 120,
    totalBenefit: 444167,
    benefits: {
      access: 216667,
      retention: 100000,
      los: 227500,
      locum: 29060,
      denials: 97500,
      hcc: 438000
    }
  };
  
  if (view === 'expand') {
    return (
      <ExpandProviders
        baselineData={baselineData}
        onBack={() => setView('builder')}
        onSaveScenario={(scenario) => {
          console.log('Save scenario:', scenario);
          // Save to your scenarios list
          setView('builder');
        }}
      />
    );
  }
  
  return (
    <div>
      {/* Your scenario builder */}
      <button onClick={() => setView('expand')}>
        Expand Providers →
      </button>
    </div>
  );
}
