// centralCalculations.js

export function calculateROIModel(inputs) {
  const {
    providers,
    annualEncounters,
    utilization,
    selectedDrivers,
    finetuneInputs,
    costPerProvider,
    contractTerm
  } = inputs;
  
  // Base calculations
  const eligibleEncounters = annualEncounters * (utilization / 100);
  const annualInvestment = providers * costPerProvider * 12;
  
  // Calculate each driver
  const driverValues = {};
  
  if (selectedDrivers.includes('patientAccess')) {
    driverValues.patientAccess = calculatePatientAccess(
      eligibleEncounters,
      finetuneInputs.patientAccess
    );
  }
  
  if (selectedDrivers.includes('levelOfService')) {
    driverValues.levelOfService = calculateLevelOfService(
      eligibleEncounters,
      finetuneInputs.levelOfService
    );
  }
  
  if (selectedDrivers.includes('retention')) {
    driverValues.retention = calculateRetention(
      providers,
      finetuneInputs.retention
    );
  }
  
  if (selectedDrivers.includes('hccCapture')) {
    driverValues.hccCapture = calculateHCCCapture(
      eligibleEncounters,
      finetuneInputs.hccCapture
    );
  }
  
  if (selectedDrivers.includes('denialReduction')) {
    driverValues.denialReduction = calculateDenialReduction(
      eligibleEncounters,
      finetuneInputs.denialReduction
    );
  }
  
  if (selectedDrivers.includes('overtimeCost')) {
    driverValues.overtimeCost = calculateOvertimeCost(
      eligibleEncounters,
      finetuneInputs.overtimeCost
    );
  }
  
  // Total annual benefit
  const totalAnnualBenefit = Object.values(driverValues).reduce((sum, val) => sum + val, 0);
  
  // Net annual gain
  const netAnnualGain = totalAnnualBenefit - annualInvestment;
  
  // ROI
  const roi = totalAnnualBenefit / annualInvestment;
  
  // Multi-year projections
  const multiYear = [];
  for (let year = 1; year <= contractTerm; year++) {
    multiYear.push({
      year,
      benefit: totalAnnualBenefit,
      investment: annualInvestment,
      net: netAnnualGain
    });
  }
  
  const totalValue = totalAnnualBenefit * contractTerm;
  const totalInvestment = annualInvestment * contractTerm;
  const totalNet = netAnnualGain * contractTerm;
  const cumulativeROI = totalValue / totalInvestment;
  
  return {
    eligibleEncounters,
    driverValues,
    totalAnnualBenefit,
    annualInvestment,
    netAnnualGain,
    roi,
    multiYear,
    totals: {
      value: totalValue,
      investment: totalInvestment,
      net: totalNet,
      roi: cumulativeROI
    }
  };
}

jsxCopy// ROIContext.jsx
import { createContext, useContext, useState, useEffect } from 'react';
import { calculateROIModel } from './centralCalculations';

const ROIContext = createContext();

export function ROIProvider({ children }) {
  const [inputs, setInputs] = useState({
    providers: null,
    annualEncounters: null,
    utilization: 65,
    selectedDrivers: [],
    finetuneInputs: {},
    costPerProvider: null,
    contractTerm: 2
  });
  
  const [results, setResults] = useState(null);
  
  // Recalculate whenever inputs change
  useEffect(() => {
    if (inputs.providers && inputs.annualEncounters && inputs.costPerProvider) {
      const newResults = calculateROIModel(inputs);
      setResults(newResults);
    }
  }, [inputs]);
  
  return (
    <ROIContext.Provider value={{ inputs, setInputs, results }}>
      {children}
    </ROIContext.Provider>
  );
}

export function useROI() {
  return useContext(ROIContext);
}

export function formatCurrency(value) {

    style: 'currency',
    currency: 'USD',
    minimumFractionDigits: 0,
    maximumFractionDigits: 0

}
