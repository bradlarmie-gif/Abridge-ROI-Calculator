🔄 State Management Logic

javascriptCopy// State structure for comparison
const [comparisonData, setComparisonData] = useState({
  competitor: 'dax',
  deployment: {
    providers: 50,
    costPerProvider: 150,
    utilization: 0.50,
    timeSaved: 1.5 // minutes per encounter
  },
  
  // User inputs per driver (normalized per provider)
  driverInputs: {
    access: {
      visitsPerProvider: 0,
      revenuePerVisit: 200
    },
    los: {
      wrvuUpliftPerProvider: 0,
      wrvuRate: 40
    },
    overtime: {
      hoursPerProvider: 0,
      premiumRate: 145
    }
    // ... other drivers
  },
  
  // Abridge benchmarks (from constants)
  abridgeBenchmarks: {
    access: {
      conservative: 60,
      typical: 120,
      optimistic: 200,
      unit: 'visits/provider/year'
    },
    los: {
      conservative: 50,
      typical: 125,
      optimistic: 200,
      unit: 'wRVUs/provider/year'
    },
    overtime: {
      conservative: 35,
      typical: 65,
      optimistic: 95,
      unit: 'hours/provider/year'
    }
  }
});

// Calculation function
const calculateDriverDelta = (driver) => {
  const userValue = comparisonData.driverInputs[driver];
  const benchmark = comparisonData.abridgeBenchmarks[driver].typical;
  const providers = comparisonData.deployment.providers;
  
  // Per provider delta
  const deltaPerProvider = benchmark - userValue.metric;
  
  // Per provider dollar value
  const userValuePerProvider = userValue.metric * userValue.rate;
  const abridgeValuePerProvider = benchmark * userValue.rate;
  const deltaValuePerProvider = deltaPerProvider * userValue.rate;
  
  // Total across all providers
  const deltaTotal = deltaValuePerProvider * providers;
  
  return {
    deltaMetric: deltaPerProvider,
    deltaValue: deltaValuePerProvider,
    deltaTotal,
    userValuePerProvider,
    abridgeValuePerProvider,
    userTotal: userValuePerProvider * providers,
    abridgeTotal: abridgeValuePerProvider * providers
  };
};

// Calculate all drivers and sum
const totalComparison = useMemo(() => {
  const activeDrivers = Object.keys(comparisonData.driverInputs);
  
  const driverDeltas = activeDrivers.map(driver => ({
    name: driver,
    ...calculateDriverDelta(driver)
  }));
  
  const grandTotal = driverDeltas.reduce((sum, d) => sum + d.deltaTotal, 0);
  
  return { driverDeltas, grandTotal };
}, [comparisonData]);
