2. Specific Insight Generation
Tell Replit to use this function:
javascriptCopyfunction generateInsight(baseline, newSetting, model) {
  const settingNames = { ed: 'ED', nursing: 'Nursing', inpatient: 'Inpatient' };
  const name = settingNames[newSetting.type];
  
  const baselineROI = baseline.roi;
  const combinedROI = model.combined.roi;
  const newROI = model.newSetting.roi;
  const gainIncrease = model.deltas.netGainIncrease;
  
  // Identify what's driving the value
  const topDriver = Object.entries(model.newSetting.benefits)
    .sort(([,a], [,b]) => b - a)[0];
  const driverName = getDriverName(topDriver[0]);
  
  // Build specific insight
  let insight = `Adding ${name} `;
  
  if (newROI > baselineROI) {
    insight += `increases your combined ROI from ${baselineROI.toFixed(1)}x to ${combinedROI.toFixed(1)}x. `;
    insight += `${name}'s ${newROI.toFixed(1)}x ROI outperforms your current deployment, `;
    
    // Explain WHY
    if (newSetting.type === 'ed') {
      const baselineWRVU = baseline.encounters > 0 ? 2.1 : 2.5; // typical outpatient
      insight += `driven by higher wRVU density (2.8 vs ${baselineWRVU.toFixed(1)} per encounter). `;
    }
    
    insight += `${driverName} accounts for the largest value contribution.`;
  } else if (gainIncrease >= 50) {
    insight += `nearly doubles your net gain (+${gainIncrease}%), `;
    insight += `generating an additional $${(model.incremental.netGain / 1000).toFixed(0)}K annually `;
    insight += `while maintaining strong ${combinedROI.toFixed(1)}x combined ROI.`;
  } else {
    insight += `increases your net gain by ${gainIncrease}% `;
    insight += `with solid ${newROI.toFixed(1)}x ROI. `;
    insight += `Combined ROI of ${combinedROI.toFixed(1)}x demonstrates efficient scaling.`;
  }
  
  return insight;
}
