Build a world-class displacement calculator for Abridge that makes prospects feel "holy shit, I need to do something" without ever attacking their current vendor. The frame is: "You vs. Your Potential" not "Them vs. Us."

## THE PSYCHOLOGY

This tool is for someone who:
- Already uses a competitor (DAX, Ambience, Suki, scribes)
- Is defensive about their choice
- Is skeptical of competitor marketing
- Fears the disruption of switching

We win by:
- Being objective and mathematical
- Showing them vs. a ceiling, not vs. Abridge
- Being more conservative than they expect
- Making the gap feel visceral and urgent
- Never attacking their current choice

## THE NARRATIVE STRUCTURE

The tool takes them through 5 psychological stages:

1. NEUTRAL DATA COLLECTION — "Let's see where you are"
2. THE CEILING — "Here's what's possible with best-in-class ambient AI"
3. THE GAP — "Here's what you're leaving on the table"
4. THE COST — "Here's what that gap costs over time"
5. THE PATH — "Here's what closing the gap looks like"

CALCULATIONS
javascriptCopyconst CEILING = {
  utilization: 65,      // 65% is best-in-class
  timeSaved: 4,         // 4 min is best-in-class
  encounterValue: 4,    // $4/encounter for utilization gap
  hourlyRate: 150,      // $150/hr for time
  timeConversion: 0.20, // 20% of time converts to $ (conservative)
};

function calculateGap(providers, encounters, utilization, timeSaved) {
  // Capture rates
  const utilizationCapture = Math.min(100, Math.round((utilization / CEILING.utilization) * 100));
  const efficiencyCapture = Math.min(100, Math.round((timeSaved / CEILING.timeSaved) * 100));
  const captureRate = Math.round((utilizationCapture / 100) * (efficiencyCapture / 100) * 100);
  
  // Utilization gap
  const utilizationGapPP = Math.max(0, CEILING.utilization - utilization);
  const utilizationGapEncounters = Math.round(encounters * (utilizationGapPP / 100));
  const utilizationGapValue = utilizationGapEncounters * CEILING.encounterValue;
  
  // Efficiency gap
  const efficiencyGapMin = Math.max(0, CEILING.timeSaved - timeSaved);
  const encountersAtCeiling = encounters * (CEILING.utilization / 100);
  const efficiencyGapTotalMin = encountersAtCeiling * efficiencyGapMin;
  const efficiencyGapHours = Math.round(efficiencyGapTotalMin / 60);
  const efficiencyGapValue = Math.round(efficiencyGapHours * CEILING.hourlyRate * CEILING.timeConversion);
  
  // Totals
  const annualGap = utilizationGapValue + efficiencyGapValue;
  const monthlyGap = Math.round(annualGap / 12);
  const threeYearGap = Math.round(annualGap * 2.75); // Year 1 partial due to ramp
  
  // Cost of waiting
  const wait6MonthsLoss = monthlyGap * 6;
  const wait12MonthsLoss = annualGap;
  
  return {
    utilizationCapture,
    efficiencyCapture,
    captureRate,
    utilizationGapPP,
    utilizationGapEncounters,
    utilizationGapValue,
    efficiencyGapMin,
    efficiencyGapHours,
    efficiencyGapValue,
    annualGap,
    monthlyGap,
    threeYearGap,
    wait6MonthsLoss,
    wait12MonthsLoss,
    encountersAtCeiling,
  };
}
