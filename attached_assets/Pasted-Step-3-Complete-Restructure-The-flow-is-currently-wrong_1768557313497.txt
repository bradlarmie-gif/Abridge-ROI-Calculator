tsxCopyfunction GapAnalysisStep() {
  const [activeDriver, setActiveDriver] = useState<string | null>(null);
  const [configuredDrivers, setConfiguredDrivers] = useState<Record<string, DriverCalculation>>({});
  
  const handleCalculate = (driverId: string, inputs: DriverInputs) => {
    const calculation = calculateDriver(driverId, inputs, competitorData);
    setConfiguredDrivers(prev => ({
      ...prev,
      [driverId]: calculation
    }));
    setActiveDriver(null); // Close config panel
  };
  
  const handleRemove = (driverId: string) => {
    setConfiguredDrivers(prev => {
      const next = {...prev};
      delete next[driverId];
      return next;
    });
  };
  
  const selectedDrivers = Object.values(configuredDrivers);
  const totalGap = selectedDrivers.reduce((sum, d) => sum + d.gap, 0);
  
  return (
    <div className="space-y-8">
      {/* Gap at a Glance - keep as is */}
      <GapAtAGlance />
      
      {/* Driver Selection Section */}
      <section>
        <h2 className="text-xl font-semibold text-gray-900 mb-2">
          How the Gap Translates to Value
        </h2>
        <p className="text-gray-500 mb-6">
          Select the value drivers that matter to your organization
        </p>
        
        {/* 6 Mini Cards Grid */}
        <div className="grid grid-cols-3 gap-4 mb-6">
          {ALL_DRIVERS.map(driver => (
            <DriverMiniCard
              key={driver.id}
              driver={driver}
              isConfigured={!!configuredDrivers[driver.id]}
              gapValue={configuredDrivers[driver.id]?.gap}
              onClick={() => setActiveDriver(driver.id)}
            />
          ))}
        </div>
        
        {/* Active Configuration Panel */}
        {activeDriver && !configuredDrivers[activeDriver] && (
          <DriverConfigPanel
            driver={ALL_DRIVERS.find(d => d.id === activeDriver)!}
            onCalculate={(inputs) => handleCalculate(activeDriver, inputs)}
            onClose={() => setActiveDriver(null)}
          />
        )}
        
        {/* Configured Driver Results */}
        <div className="space-y-4">
          {selectedDrivers.map(driver => (
            <DriverResultCard
              key={driver.id}
              driver={ALL_DRIVERS.find(d => d.id === driver.id)!}
              calculations={driver}
              onRemove={() => handleRemove(driver.id)}
            />
          ))}
        </div>
      </section>
      
      {/* Summary Section */}
      {selectedDrivers.length > 0 && (
        <section className="bg-white border border-gray-200 rounded-xl p-6">
          <div className="flex items-start justify-between">
            <div>
              <div className="text-sm font-medium text-gray-400 uppercase tracking-wide mb-3">
                Selected Drivers
              </div>
              <div className="space-y-2">
                {selectedDrivers.map(d => (
                  <div key={d.id} className="flex items-center gap-2">
                    <div className="w-2 h-2 rounded-full bg-orange-500" />
                    <span className="text-gray-700">{d.label}</span>
                    <span className="text-orange-600 font-medium ml-auto">
                      +{formatCurrency(d.gap)}
                    </span>
                  </div>
                ))}
              </div>
            </div>
            
            <div className="text-right pl-8 border-l border-gray-100">
              <div className="text-sm text-gray-400 mb-1">Total Annual Gap</div>
              <div className="text-3xl font-bold text-gray-900">
                +{formatCurrency(totalGap)}
              </div>
              <div className="text-sm text-gray-400">per year</div>
            </div>
          </div>
