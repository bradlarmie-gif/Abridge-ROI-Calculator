Step 1: Update the baseline calculation function
javascriptCopyfunction calculateNursingBaseline() {
  // Get inputs from Your Organization page
  const beds = parseInt(document.getElementById('staffed-beds').value) || 200;
  const occupancy = parseFloat(document.getElementById('occupancy-rate').value) || 85;
  const eventsPerDay = parseInt(document.getElementById('events-per-patient-day').value) || 3;
  const utilization = getSelectedUtilization(); // Returns 50, 65, or 80
  
  // Calculate patient days
  const patientDaysPerYear = Math.round(beds * (occupancy / 100) * 365);
  
  // Calculate total documentation events
  const totalEvents = patientDaysPerYear * eventsPerDay;
  
  // Calculate eligible events (with Abridge utilization)
  const eligibleEvents = Math.round(totalEvents * (utilization / 100));
  
  // Calculate implied nurse FTEs for context (1.5 FTEs per bed is standard)
  const impliedFTEs = Math.round(beds * 1.5);
  
  return {
    beds,
    occupancy,
    eventsPerDay,
    utilization,
    patientDaysPerYear,
    totalEvents,
    eligibleEvents,
    impliedFTEs
  };
}
