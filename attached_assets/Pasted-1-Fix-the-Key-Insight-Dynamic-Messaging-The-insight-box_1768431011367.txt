1. Fix the "Key Insight" Dynamic Messaging
The insight box needs to update more intelligently based on what drivers are added:
javascriptCopy// Enhanced Key Insight Generator
const generateKeyInsight = (activeDrivers, baselineROI, scenarioROI, investment) => {
  const roiIncrease = scenarioROI - baselineROI;
  const roiPercentIncrease = ((roiIncrease / baselineROI) * 100).toFixed(0);
  const totalNewValue = activeDrivers.reduce((sum, driver) => sum + driver.value, 0);
  const largestDriver = activeDrivers.reduce((max, driver) => 
    driver.value > max.value ? driver : max, activeDrivers[0]);
  
  // Different messages based on context
  if (activeDrivers.length === 1) {
    return `Adding ${activeDrivers[0].name} increases your ROI from ${baselineROI}x to ${scenarioROI}x without any additional investment. ${activeDrivers[0].name} alone adds $${activeDrivers[0].value.toLocaleString()} (${((activeDrivers[0].value / totalBenefit) * 100).toFixed(0)}% of total value).`;
  }
  
  if (activeDrivers.length === 2) {
    return `Adding these ${activeDrivers.length} drivers increases your ROI from ${baselineROI}x to ${scenarioROI}x (+${roiPercentIncrease}%) without any additional investment. ${largestDriver.name} contributes the most at $${largestDriver.value.toLocaleString()}.`;
  }
  
  if (activeDrivers.length >= 3) {
    return `Adding these ${activeDrivers.length} drivers increases your ROI from ${baselineROI}x to ${scenarioROI}x (+${roiPercentIncrease}%) without any additional investment. ${largestDriver.name} alone adds $${largestDriver.value.toLocaleString()} (${((largestDriver.value / totalNewValue) * 100).toFixed(0)}% of new value).`;
  }
  
  return `Adding drivers increases value without increasing investment. This often dramatically improves ROI.`;
};
