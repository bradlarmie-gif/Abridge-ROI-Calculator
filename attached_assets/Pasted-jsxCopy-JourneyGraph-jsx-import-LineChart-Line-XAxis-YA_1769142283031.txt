import { LineChart, Line, XAxis, YAxis, Tooltip, ResponsiveContainer, Area, ReferenceDot } from 'recharts';

function JourneyGraph({ pilotData, fullScaleData }) {
  // Generate the journey data points
  const journeyData = [
    {
      month: 0,
      label: 'Today',
      beds: pilotData.beds,
      utilization: pilotData.utilization,
      linearValue: pilotData.annualValue,
      actualValue: pilotData.annualValue,
      phase: 'Pilot'
    },
    {
      month: 6,
      label: '6 months',
      beds: Math.round(pilotData.beds * 1.25),
      utilization: Math.min(pilotData.utilization + 5, 85),
      linearValue: pilotData.annualValue * 1.25,
      actualValue: pilotData.annualValue * 1.35, // +utilization boost
      phase: 'Early Expansion'
    },
    {
      month: 12,
      label: '1 year',
      beds: Math.round(pilotData.beds * 1.5),
      utilization: Math.min(pilotData.utilization + 10, 85),
      linearValue: pilotData.annualValue * 1.5,
      actualValue: pilotData.annualValue * 1.75, // +maturity effects
      phase: 'Expansion'
    },
    {
      month: 18,
      label: '18 months',
      beds: Math.round(pilotData.beds * 2),
      utilization: Math.min(pilotData.utilization + 12, 85),
      linearValue: pilotData.annualValue * 2,
      actualValue: pilotData.annualValue * 2.5, // +network effects
      phase: 'Scale'
    },
    {
      month: 24,
      label: '2 years',
      beds: Math.round(pilotData.beds * 2.5),
      utilization: Math.min(pilotData.utilization + 15, 85),
      linearValue: pilotData.annualValue * 2.5,
      actualValue: pilotData.annualValue * 3.3,
      phase: 'Maturity'
    },
    {
      month: 36,
      label: '3 years',
      beds: fullScaleData?.beds || Math.round(pilotData.beds * 3),
      utilization: fullScaleData?.utilization || 80,
      linearValue: pilotData.annualValue * 3,
      actualValue: pilotData.annualValue * 4.2,
      phase: 'Full Scale'
    }
  ];

  const CustomTooltip = ({ active, payload, label }) => {
    if (!active || !payload || !payload.length) return null;
    
    const data = payload[0].payload;
    const compoundingValue = data.actualValue - data.linearValue;
    
    return (
      <div className="journey-tooltip">
        <div className="tooltip-header">
          <span className="tooltip-phase">{data.phase}</span>
          <span className="tooltip-time">{data.label}</span>
        </div>
        
        <div className="tooltip-metrics">
          <div className="tooltip-row">
            <span className="metric-label">Scale</span>
            <span className="metric-value">{data.beds} beds</span>
          </div>
          <div className="tooltip-row">
            <span className="metric-label">Utilization</span>
            <span className="metric-value">{data.utilization}%</span>
          </div>
        </div>
        
        <div className="tooltip-divider" />
        
        <div className="tooltip-values">
          <div className="tooltip-row">
            <span className="metric-label">Linear Value</span>
            <span className="metric-value muted">
              {formatCurrency(data.linearValue)}
            </span>
          </div>
          <div className="tooltip-row">
            <span className="metric-label">Actual Value</span>
            <span className="metric-value highlight">
              {formatCurrency(data.actualValue)}
            </span>
          </div>
          {compoundingValue > 0 && (
            <div className="tooltip-row compounding">
              <span className="metric-label">Compounding Effect</span>
              <span className="metric-value bonus">
                +{formatCurrency(compoundingValue)}
              </span>
            </div>
          )}
        </div>
        
        <div className="tooltip-roi">
          <span className="roi-value">
            {(data.actualValue / pilotData.investment).toFixed(1)}x ROI
          </span>
        </div>
      </div>
    );
  };

  return (
    <div className="journey-graph-container">
      <div className="journey-header">
        <h3>Your Journey with Abridge</h3>
        <p>Start with a pilot. Prove the value. Scale across your organization.</p>
      </div>
      
      <div className="journey-chart">
        <ResponsiveContainer width="100%" height={300}>
          <LineChart data={journeyData} margin={{ top: 20, right: 30, left: 20, bottom: 20 }}>
            <defs>
              <linearGradient id="actualGradient" x1="0" y1="0" x2="0" y2="1">
                <stop offset="0%" stopColor="#10b981" stopOpacity={0.3} />
                <stop offset="100%" stopColor="#10b981" stopOpacity={0.05} />
              </linearGradient>
            </defs>
            
            <XAxis 
              dataKey="label" 
              axisLine={false}
              tickLine={false}
              tick={{ fill: '#64748b', fontSize: 12 }}
            />
            
            <YAxis 
              axisLine={false}
              tickLine={false}
              tick={{ fill: '#64748b', fontSize: 12 }}
              tickFormatter={(value) => formatCurrencyShort(value)}
            />
            
            <Tooltip content={<CustomTooltip />} />
            
            {/* Area under the actual value line */}
            <Area
              type="monotone"
              dataKey="actualValue"
              stroke="none"
              fill="url(#actualGradient)"
            />
            
            {/* Linear projection (dashed) */}
            <Line
              type="monotone"
              dataKey="linearValue"
              stroke="#94a3b8"
              strokeWidth={2}
              strokeDasharray="8 4"
              dot={false}
              name="Linear projection"
            />
            
            {/* Actual value (solid) */}
            <Line
              type="monotone"
              dataKey="actualValue"
              stroke="#10b981"
              strokeWidth={3}
              dot={false}
              name="Actual value"
            />
            
            {/* Pilot marker */}
            <ReferenceDot
              x="Today"
              y={journeyData[0].actualValue}
              r={6}
              fill="#ef4444"
              stroke="white"
              strokeWidth={2}
            />
            
            {/* Full scale marker */}
            <ReferenceDot
              x="3 years"
              y={journeyData[journeyData.length - 1].actualValue}
              r={6}
              fill="#10b981"
              stroke="white"
              strokeWidth={2}
            />
          </LineChart>
        </ResponsiveContainer>
      </div>
      
      <div className="journey-legend">
        <div className="legend-item">
          <span className="legend-dot pilot" />
          <span>Pilot (Today)</span>
        </div>
        <div className="legend-item">
          <span className="legend-line solid" />
          <span>Actual value (with compounding)</span>
        </div>
        <div className="legend-item">
          <span className="legend-line dashed" />
          <span>Linear projection</span>
        </div>
        <div className="legend-item">
          <span className="legend-dot full-scale" />
          <span>Full Scale</span>
        </div>
      </div>
    </div>
  );
}

// Helper functions
function formatCurrency(value) {
  return new Intl.NumberFormat('en-US', {
    style: 'currency',
    currency: 'USD',
    minimumFractionDigits: 0,
    maximumFractionDigits: 0
  }).format(value);
}

function formatCurrencyShort(value) {
  if (value >= 1000000) {
    return `$${(value / 1000000).toFixed(1)}M`;
  }
  if (value >= 1000) {
    return `$${(value / 1000).toFixed(0)}K`;
  }
  return `$${value}`;
}
