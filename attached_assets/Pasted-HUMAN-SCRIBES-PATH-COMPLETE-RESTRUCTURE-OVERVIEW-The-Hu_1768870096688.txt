const realizationFactor = 0.50; // 50%

// Use 65% utilization, not 100%

const encountersWithAbridge = activeProviders * encountersPerProvider;

  const wrvuUplift = 0.035; // 3.5%
  const wrvuValue = 45; // $ per wRVU

// Denial reduction value
if (selectedDrivers.includes('denials')) {

  const denialCostPerClaim = 50; // $ per denied claim
  const denialReduction = 0.30; // 30% reduction
  const denialValue = encountersWithAbridge * denialRate * denialCostPerClaim * denialReduction * realizationFactor;
