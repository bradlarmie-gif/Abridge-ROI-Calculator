staffedBeds = 200
nurseFTEs = 300
occupancyRate = 0.85

// TIME SAVINGS (Per Nurse Per Shift model)
minutesSavedPerNursePerShift = 30 // Conservative: 15, Typical: 30, Aggressive: 45
shiftsPerNursePerYear = 156 // ~3 shifts/week × 52 weeks
utilizationRate = 0.70

// CALCULATE
totalShiftsPerYear = nurseFTEs × shiftsPerNursePerYear
// = 300 × 156 = 46,800 shifts/year

eligibleShifts = totalShiftsPerYear × utilizationRate
// = 46,800 × 0.70 = 32,760 shifts

totalMinutesSaved = eligibleShifts × minutesSavedPerNursePerShift
// = 32,760 × 30 = 982,800 minutes

totalHoursSaved = totalMinutesSaved / 60
// = 16,380 hours/year

hoursPerNursePerYear = totalHoursSaved / nurseFTEs
// = 16,380 / 300 = 54.6 hours/nurse/year
// = ~1 hour/week per nurse
