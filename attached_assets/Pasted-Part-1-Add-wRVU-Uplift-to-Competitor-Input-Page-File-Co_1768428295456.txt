              </div>
            </div>
          </div>
        </div>

      </div>

      {/* What-If CTA Section */}
      <div className="what-if-cta">
        <div className="cta-content">
          <h3>What if you achieved Abridge's typical results?</h3>
          <p>See your potential ROI using proven benchmarks from 200+ health systems</p>
          <button className="primary-button" onClick={() => {/* apply benchmarks */}}>
            Apply Abridge Benchmarks →
          </button>
        </div>
      </div>

    </div>
  );
};
