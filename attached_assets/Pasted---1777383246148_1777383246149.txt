  Replace the {financialDrivers.map(renderDriverCard)} line with a partition-and-render pattern that:                                                                                                          
                                                            
  1. Builds a list of unique subGroup values from financialDrivers (preserve insertion order, drop duplicates).                                                                                                
  2. If there are zero distinct subGroups (i.e., none of the visible financial drivers have a subGroup set), render the flat list as today.                                                                    
  3. Otherwise, for each distinct subGroup, render a small sub-sub-header followed by the drivers that share that subGroup. Drivers without a subGroup field should render last under no header.               
                                                                                                                                                                                                               
  Sketch:                                                                       
                                                                                                                                                                                                               
  const subGroupOrder: string[] = [];                                                                                                                                                                          
  financialDrivers.forEach(d => {                           
    if (d.subGroup && !subGroupOrder.includes(d.subGroup)) subGroupOrder.push(d.subGroup);                                                                                                                     
  });                                                                           
  const ungrouped = financialDrivers.filter(d => !d.subGroup);                                                                                                                                                 
                                                                                                                                                                                                               
  return subGroupOrder.length === 0 ? (                                                                                                                                                                        
    financialDrivers.map(renderDriverCard)                                                                                                                                                                     
  ) : (                                                                                                                                                                                                        
    <>                                                                          
      {subGroupOrder.map(sg => {                            
        const inGroup = financialDrivers.filter(d => d.subGroup === sg);                                                                                                                                       
        return (                           
          <div key={sg} className="space-y-3">                                                                                                                                                                 
            <p className="text-[10px] font-semibold text-[#666666] uppercase tracking-[1.5px] mt-2 mb-1">                                                                                                      
              {sg}                                                                                                                                                                                             
            </p>                                                                                                                                                                                               
            {inGroup.map(renderDriverCard)}                                     
          </div>                                            
        );                                                  
      })}                                                                                                                                                                                                      
      {ungrouped.length > 0 && ungrouped.map(renderDriverCard)}            
    </>                                                                                                                                                                                                        
  );
