Updated Analysis Page — Metric-Based Value
The value breakdown should come from the actual metrics, not arbitrary percentages:
jsxCopy// ============ SWITCH AMBIENT: ANALYSIS (METRIC-BASED) ============

// Instead of:
// Patient Access: +$73,710/yr
// Level of Service Accuracy: +$127,575/yr
// Documentation-Related Denials: +$82,215/yr

// We show:
// Based on THEIR selected metrics:

const calculateValueFromMetrics = (inputs, benchmarks) => {
  const values = [];
  const providers = inputs.providers;
  const encounters = inputs.totalEncounters || providers * 2000;
  
  // 1. Utilization Gap
  if (inputs.utilization < benchmarks.utilization) {
    const additionalEncounters = Math.round(encounters * ((benchmarks.utilization - inputs.utilization) / 100));
    const valuePerEncounter = 2.60; // wRVU lift + quality
    values.push({
      metric: 'Utilization',
      gap: `${inputs.utilization}% → ${benchmarks.utilization}%`,
      impact: `+${formatNumber(additionalEncounters)} encounters`,
      value: Math.round(additionalEncounters * valuePerEncounter),
      calculation: `${formatNumber(additionalEncounters)} encounters × $${valuePerEncounter}/encounter`,
      explanation: 'More encounters documented = more complete clinical picture for every patient'
    });
  }
  
  // 2. Time Savings (if selected)
  const timeSavingsValue = inputs.metricValues?.timeSavings;
  if (timeSavingsValue && timeSavingsValue < 4) {
    const abridgeEncounters = Math.round(encounters * (benchmarks.utilization / 100));
    const additionalMinutes = (4 - timeSavingsValue) * abridgeEncounters;
    const additionalHours = Math.round(additionalMinutes / 60);
    const hourlyValue = 75;
    values.push({
      metric: 'Time Savings',
      gap: `${timeSavingsValue} min → 4 min/encounter`,
      impact: `+${formatNumber(additionalHours)} hours/year`,
      value: Math.round(additionalHours * hourlyValue * 0.5), // 50% conversion
      calculation: `${formatNumber(additionalHours)} hours × $${hourlyValue}/hr × 50% conversion`,
      explanation: 'Time returned to providers for patient care, quality of life, or capacity'
    });
  }
  
  // 3. Work Outside of Work (if selected)
  const wowValue = inputs.metricValues?.workOutsideWork;
  if (wowValue && wowValue > 2) {
    const hoursSavedPerWeek = wowValue - 2;
    const annualHoursSaved = hoursSavedPerWeek * providers * 48;
    const hourlyValue = 50; // Quality of life / retention value
    values.push({
      metric: 'Work Outside of Work',
      gap: `${wowValue} hrs/week → 2 hrs/week`,
      impact: `-${hoursSavedPerWeek} hrs/week per provider`,
      value: Math.round(annualHoursSaved * hourlyValue * 0.33), // Retention proxy
      calculation: `${formatNumber(annualHoursSaved)} hours × $${hourlyValue}/hr × 33% retention factor`,
      explanation: 'Reduced after-hours work correlates with lower burnout and better retention'
    });
  }
  
  // 4. wRVU Lift (if selected)
  const wrvuValue = inputs.metricValues?.wrvuLift;
  if (wrvuValue !== undefined && wrvuValue < 5) {
    const liftDelta = 5 - wrvuValue;
    const abridgeEncounters = Math.round(encounters * (benchmarks.utilization / 100));
    const baseWrvuPerEncounter = 1.5;
    const additionalWrvus = Math.round(abridgeEncounters * baseWrvuPerEncounter * (liftDelta / 100));
    const conversionFactor = 33;
    values.push({
      metric: 'wRVU Capture',
      gap: `${wrvuValue}% lift → 5% lift`,
      impact: `+${formatNumber(additionalWrvus)} wRVUs/year`,
      value: Math.round(additionalWrvus * conversionFactor),
      calculation: `${formatNumber(additionalWrvus)} wRVUs × $${conversionFactor}/wRVU`,
      explanation: 'Better documentation captures clinical complexity that supports accurate coding'
    });
  }
  
  // 5. Level of Service (if selected)
  const losValue = inputs.metricValues?.levelOfService;
  if (losValue && losValue < 4.1) {
    const levelDelta = 4.1 - losValue;
    const abridgeEncounters = Math.round(encounters * (benchmarks.utilization / 100));
    // Each 0.1 increase in avg level ≈ 0.07 wRVU increase
    const additionalWrvus = Math.round(abridgeEncounters * (levelDelta * 0.7));
    const conversionFactor = 33;
    values.push({
      metric: 'Level of Service',
      gap: `${losValue} avg → 4.1 avg`,
      impact: `+${levelDelta.toFixed(1)} average level`,
      value: Math.round(additionalWrvus * conversionFactor),
      calculation: `${formatNumber(additionalWrvus)} additional wRVUs × $${conversionFactor}/wRVU`,
      explanation: 'Higher average E/M level from more complete documentation of complexity'
    });
  }
  
  // 6. Clinician Retention (derived from satisfaction + WOW)
  if ((wowValue && wowValue > 3) || (inputs.metricValues?.satisfaction && inputs.metricValues.satisfaction < 7)) {
    const turnoverReduction = 0.15; // 15% of turnover attributable to doc burden
    const avgTurnover = providers * 0.18; // 18% turnover rate
    const preventedDepartures = Math.round(avgTurnover * turnoverReduction * 10) / 10;
    const replacementCost = 400000;
    values.push({
      metric: 'Clinician Retention',
      gap: 'Improved satisfaction → lower turnover',
      impact: `~${preventedDepartures} departures prevented`,
      value: Math.round(preventedDepartures * replacementCost * 0.5), // Conservative
      calculation: `${preventedDepartures} × $${formatNumber(replacementCost)} × 50% attribution`,
      explanation: 'Better documentation experience reduces burnout-driven turnover',
      isRetention: true // Flag for different treatment
    });
  }
  
  return values;
};
