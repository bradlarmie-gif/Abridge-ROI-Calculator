PAGE GENERATION LOGIC:

javascriptCopyfunction generatePDF(modelData, options) {
  const pages = [];
  
  // Always include
  pages.push(<CoverPage data={modelData} />);
  pages.push(<ExecutiveSummary data={modelData} />);
  pages.push(<ValueBreakdown data={modelData} />);
  
  // Add driver detail pages for each active driver
  modelData.activeDrivers.forEach(driver => {
    pages.push(
      <DriverDetail 
        driver={driver}
        calculations={modelData.calculations[driver]}
        inputs={modelData.inputs}
        showMath={options.includeDetailedCalculations}
        showExplanations={true} // Always true for explanatory style
      />
    );
  });
  
  pages.push(<TotalImpact data={modelData} />);
  pages.push(<Assumptions data={modelData} />);
  
  return pages;
}

FILENAME GENERATION:

javascriptCopyfunction generateFilename(modelData, exportType) {
  const date = new Date().toISOString().split('T')[0]; // YYYY-MM-DD
  const orgName = modelData.organizationName || "Organization";
  const sanitized = orgName.replace(/[^a-zA-Z0-9]/g, '_');
  
  if (exportType === 'baseline') {
    return `Abridge_ROI_Model_${sanitized}_${date}.pdf`;
  } else if (exportType === 'scenario') {
    const scenarioName = modelData.scenarioName.replace(/[^a-zA-Z0-9]/g, '_');
    return `Abridge_ROI_${scenarioName}_${sanitized}_${date}.pdf`;
  } else if (exportType === 'comparison') {
    return `Abridge_ROI_Comparison_${sanitized}_${date}.pdf`;
  }
}
