const documentedAdmissions = 5200;         // From earlier in flow
const gapRate = 0.40;                      // 40% have documentation gaps
const improvementRate = 0.15;              // 15% Abridge can capture

// Step 3: DRG Weight Impact
const grossImpact = admissionsImproved * drgWeightIncrease * baseDrgPayment;
// 312 × 0.4 × $6,000 = $748,800

// Step 4: Reality Check
const annualValue = grossImpact * realizationRate;
// $748,800 × 0.50 = $374,400
