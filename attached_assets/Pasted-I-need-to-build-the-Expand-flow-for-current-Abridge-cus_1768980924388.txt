I need to build the "Expand" flow for current Abridge customers. This is a performance dashboard, not a calculator. It should feel like the same quality UX/UI we've built for the Explore flow.

## OVERVIEW

This flow is for customers already using Abridge who want to:
1. See how their deployment is performing
2. Understand their results with distributions (bell curves), not just averages
3. Calculate their actual ROI from real data
4. Model expansion opportunities

## PAGE STRUCTURE

### Page 1: Setting Selection
- Outpatient is active
- ED, Inpatient, Nursing are blurred/locked with "Coming Soon"
- Link back to Explore if they don't have results yet

### Page 2: Your Deployment + Metrics
- Enter: Providers, Annual Encounters, Utilization Rate, Months on Abridge
- Select metrics to analyze (checkboxes):
  - Time Savings (recommended)
  - Level of Service (recommended)
  - Chart Closure Time (recommended)
  - wRVU Capture (recommended)
  - Work After Hours (optional)

### Page 3: Your Performance (Main Dashboard)
For each selected metric, show:
- Before/After comparison
- Distribution visualization (bell curve showing % of providers benefiting)
- Benchmark comparison
- Trend over time (where applicable)

Key insight: Show "X% of providers seeing meaningful benefit" — not just averages.

### Page 4: Your ROI Story
- Conservative calculation from their actual data
- Full breakdown with math shown
- "Defensible" framing with conversion rates explained

### Page 5: Your Journey
- Timeline chart showing current state → expansion potential
- Editable inputs for full scale opportunity
- Combined potential calculation

## DESIGN SYSTEM

Use the same design system as Explore:
- Clean white/gray backgrounds
- Orange (#f97316) for current/pilot state
- Green (#059669) for targets/success
- Gray (#94a3b8) for intermediate states
- Rounded cards with subtle shadows
- Clear section headers
- Inline benchmarks and insights

## KEY VISUALIZATIONS

1. Bell Curve Distribution:
- Shows how metric is distributed across providers
- Highlights median
- Shows % in each segment (low/medium/high benefit)
- Key stat: "X% of providers seeing meaningful benefit"

2. E/M Level of Service Shift:
- Horizontal bar comparison (before vs after)
- Shows percentage point change for each level
- Insight about what the shift means

3. Trend Over Time:
- Line chart showing progression by month
- Markers for key milestones

4. Journey Timeline:
- Curved area chart showing value over time
- "You are here" marker
- Expansion potential visualization

import { useState } from 'react';
import SettingSelection from './ExpandSettingSelection';
import DeploymentSetup from './ExpandDeploymentSetup';
import PerformanceDashboard from './ExpandPerformanceDashboard';
import ROIStory from './ExpandROIStory';
import JourneyExpansion from './ExpandJourneyExpansion';

export default function ExpandFlow() {
  const [currentStep, setCurrentStep] = useState(1);
  const [deploymentData, setDeploymentData] = useState({
    setting: 'outpatient',
    providers: 150,
    annualEncounters: 195000,
    utilizationRate: 72,
    monthsOnAbridge: 6
  });
  const [selectedMetrics, setSelectedMetrics] = useState([
    'timeSavings',
    'levelOfService',
    'chartClosure',
    'wrvuCapture'
  ]);
  const [metricsData, setMetricsData] = useState({
    timeSavings: { before: 12, after: 4.5 },
    levelOfService: {
      before: { '99215': 35, '99214': 45, '99213': 15, '99212': 4, '99211': 1 },
      after: { '99215': 42, '99214': 44, '99213': 11, '99212': 2, '99211': 1 }
    },
    chartClosure: { before: 36, after: 4.2 },
    wrvuCapture: { before: 1.42, after: 1.49 }
  });

  const steps = [
    { id: 1, name: 'Setting', component: SettingSelection },
    { id: 2, name: 'Setup', component: DeploymentSetup },
    { id: 3, name: 'Performance', component: PerformanceDashboard },
    { id: 4, name: 'ROI', component: ROIStory },
    { id: 5, name: 'Journey', component: JourneyExpansion }
  ];

  const CurrentStepComponent = steps.find(s => s.id === currentStep)?.component;

  return (
    <div className="expand-flow">
      <CurrentStepComponent
        deploymentData={deploymentData}
        setDeploymentData={setDeploymentData}
        selectedMetrics={selectedMetrics}
        setSelectedMetrics={setSelectedMetrics}
        metricsData={metricsData}
        setMetricsData={setMetricsData}
        onNext={() => setCurrentStep(prev => Math.min(prev + 1, 5))}
        onBack={() => setCurrentStep(prev => Math.max(prev - 1, 1))}
        currentStep={currentStep}
      />
    </div>
  );
}

import { useMemo } from 'react';
import BellCurveChart from './BellCurveChart';
import LevelOfServiceChart from './LevelOfServiceChart';
import TrendChart from './TrendChart';

export default function ExpandPerformanceDashboard({
  deploymentData,
  selectedMetrics,
  metricsData,
  setMetricsData,
  onNext,
  onBack
}) {
  
  const abridgeEncounters = Math.round(
    deploymentData.annualEncounters * (deploymentData.utilizationRate / 100)
  );
  
  // Calculate derived values
  const timeSavingsCalc = useMemo(() => {
    if (!selectedMetrics.includes('timeSavings')) return null;
    const { before, after } = metricsData.timeSavings;
    const savings = before - after;
    const totalHours = (savings * abridgeEncounters) / 60;
    return {
      before,
      after,
      savings,
      percentReduction: Math.round((savings / before) * 100),
      totalHours: Math.round(totalHours),
      // Simulated distribution data
      distribution: generateDistribution(savings, 0.3),
      percentBenefiting: 92
    };
  }, [metricsData.timeSavings, abridgeEncounters, selectedMetrics]);
  
  const wrvuCalc = useMemo(() => {
    if (!selectedMetrics.includes('wrvuCapture')) return null;
    const { before, after } = metricsData.wrvuCapture;
    const lift = after - before;
    const liftPercent = (lift / before) * 100;
    const totalWrvuGain = lift * abridgeEncounters;
    return {
      before,
      after,
      lift,
      liftPercent: liftPercent.toFixed(1),
      totalWrvuGain: Math.round(totalWrvuGain),
      distribution: generateDistribution(liftPercent, 0.4),
      percentBenefiting: 85
    };
  }, [metricsData.wrvuCapture, abridgeEncounters, selectedMetrics]);
  
  const chartClosureCalc = useMemo(() => {
    if (!selectedMetrics.includes('chartClosure')) return null;
    const { before, after } = metricsData.chartClosure;
    const reduction = before - after;
    const percentFaster = Math.round((reduction / before) * 100);
    return {
      before,
      after,
      reduction,
      percentFaster,
      // Simulated trend data
      trend: [
        { month: 'M1', value: before },
        { month: 'M2', value: before * 0.6 },
        { month: 'M3', value: before * 0.35 },
        { month: 'M4', value: before * 0.2 },
        { month: 'M5', value: after * 1.2 },
        { month: 'M6', value: after }
      ]
    };
  }, [metricsData.chartClosure, selectedMetrics]);
  
  const losCalc = useMemo(() => {
    if (!selectedMetrics.includes('levelOfService')) return null;
    const { before, after } = metricsData.levelOfService;
    const shifts = {};
    Object.keys(before).forEach(level => {
      shifts[level] = after[level] - before[level];
    });
    return { before, after, shifts };
  }, [metricsData.levelOfService, selectedMetrics]);
  
  return (
    <div className="expand-page performance-dashboard">
      
      <header className="page-header">
        <button className="back-button" onClick={onBack}>← Back</button>
        <div className="step-indicator">Step 2 of 4 · Your Performance</div>
      </header>
      
      <main className="page-content">
        
        <div className="page-title">
          <h1>Your Performance</h1>
          <p>{deploymentData.providers} providers · {deploymentData.monthsOnAbridge} months on Abridge</p>
        </div>
        
        {/* Time Savings */}
        {timeSavingsCalc && (
          <section className="metric-section">
            <div className="metric-header-row">
              <div className="metric-icon">⏱️</div>
              <h2>TIME SAVINGS</h2>
            </div>
            
            {/* Before/After/Change */}
            <div className="before-after-card">
              <div className="ba-item">
                <span className="ba-label">BEFORE</span>
                <div className="ba-value-group">
                  <input
                    type="number"
                    value={metricsData.timeSavings.before}
                    onChange={(e) => setMetricsData({
                      ...metricsData,
                      timeSavings: { ...metricsData.timeSavings, before: Number(e.target.value) }
                    })}
                    className="ba-input"
                  />
                  <span className="ba-unit">min in notes</span>
                </div>
              </div>
              
              <div className="ba-arrow">→</div>
              
              <div className="ba-item">
                <span className="ba-label">AFTER</span>
                <div className="ba-value-group">
                  <input
                    type="number"
                    step="0.1"
                    value={metricsData.timeSavings.after}
                    onChange={(e) => setMetricsData({
                      ...metricsData,
                      timeSavings: { ...metricsData.timeSavings, after: Number(e.target.value) }
                    })}
                    className="ba-input"
                  />
                  <span className="ba-unit">min in notes</span>
                </div>
              </div>
              
              <div className="ba-equals">=</div>
              
              <div className="ba-item result">
                <span className="ba-label">CHANGE</span>
                <div className="ba-result">
                  <span className="ba-result-value">-{timeSavingsCalc.savings} min/enc</span>
                  <span className="ba-result-percent">{timeSavingsCalc.percentReduction}% reduction</span>
                </div>
              </div>
            </div>
            
            {/* Distribution */}
            <div className="distribution-section">
              <h3>PROVIDER DISTRIBUTION</h3>
              <p className="distribution-subtitle">How are savings distributed across your providers?</p>
              
              <BellCurveChart
                data={timeSavingsCalc.distribution}
                median={timeSavingsCalc.savings}
                unit="min"
                segments={[
                  { label: '< 3 min', percent: 8, range: 'low' },

                  { label: '> 10 min', percent: 20, range: 'high' }

// Helper to generate simulated distribution data
function generateDistribution(median, variance) {
  const points = [];
  for (let i = 0; i <= 20; i++) {
    const x = median * (0.2 + (i / 20) * 1.6);
    const y = Math.exp(-Math.pow(x - median, 2) / (2 * Math.pow(median * variance, 2)));
    points.push({ x: x.toFixed(1), y: y * 100 });
  }
  return points;
}
