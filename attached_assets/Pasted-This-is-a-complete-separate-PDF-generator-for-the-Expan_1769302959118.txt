// TYPES & DATA STRUCTURES
// ============================================================================

// Metric types that can be tracked
export type MetricId = 
  | 'wrvuPerEncounter'
  | 'timeInNotes'
  | 'sameDayChartClosure'
  | 'averageEmLevel'
  | 'workOutsideWork'
  | 'clinicianSatisfaction';

export type MetricCategory = 'primary' | 'secondary';

export type DataEntryMode = 'quick' | 'trend';

export type TimeConversionMode = 
  | 'notConverting'      // Show as hours saved, not $
  | 'patientVisits'      // Converting to additional patient visits
  | 'overtimeReduction'; // Converting to OT/locum savings

// Individual metric data
export interface MetricData {
  id: MetricId;
  name: string;
  description: string;
  category: MetricCategory;
  dataSource: string; // "Clarity", "Survey", etc.
  unit: string; // "wRVU/enc", "min", "%", "hrs/week", "pts"
  
  // Data entry mode
  entryMode: DataEntryMode;
  
  // Quick mode (before/after only)
  beforeValue: number | null;
  afterValue: number | null;
  
  // Trend mode (monthly data)
  trendData?: {
    baseline: number;
    months: { month: number; value: number }[];
  };
  
  // Calculated
  change: number; // Absolute change
  changePercent: number; // Percentage change
  changeDirection: 'increase' | 'decrease'; // Which direction is good
  
  // Benchmark comparison
  benchmark: {
    typicalRange: string; // "3-7% lift", "2-5 hours per week"
    status: 'below' | 'within' | 'above';
    statusLabel: string; // "Below typical", "Within expected range", "Above typical"
  };
  
  // For E&M level, optional distribution
  distribution?: {
    before: { level: number; percent: number }[];
    after: { level: number; percent: number }[];
  };
}

// Deployment information
export interface DeploymentData {
  organizationName?: string;
  careSetting: string;
  providers: number;
  totalEncounters: number;
  utilizationRate: number;
  monthsOnAbridge: number;
  documentedEncounters: number; // providers × encounters × utilization
}

// Value tier breakdown
export interface ValueBreakdown {
  // Tier 1: Hard Value (dollarized)
  tier1: {
    total: number;
    items: {
      name: string;
      value: number;
      formula: string;
      isSpeculative?: boolean;
      warning?: string;
    }[];
  };
  
  // Tier 2: Efficiency Gains (natural units)
  tier2: {
    items: {
      name: string;
      value: string; // "8,250 hours/year", "-7 hrs/week", "+11%"
      metric: MetricId;
    }[];
  };
  
  // Tier 3: Leading Indicators (qualitative)
  tier3: {
    items: {
      name: string;
      value: string; // "50 → 55 (+5.0 points)"
      metric: MetricId;
    }[];
  };
}

// Time conversion configuration
export interface TimeConversionConfig {
  mode: TimeConversionMode;
  hoursSaved: number;
  
  // If converting to patient visits
  patientVisits?: {
    conversionPercent: number; // What % of time converts
    revenuePerVisit: number;
    additionalVisits: number;
    value: number;
  };
  
  // If converting to OT reduction
  overtimeReduction?: {
    reductionPercent: number;
    hourlyRate: number;
    hoursReduced: number;
    value: number;
  };
}

// Retention estimation (speculative)
export interface RetentionEstimate {
  enabled: boolean;
  departuresPrevented: number;
  replacementCost: number;
  totalValue: number;
  isSpeculative: boolean;
}

// Expansion projection
export interface ExpansionProjection {
  currentState: {
    providers: number;
    utilization: number;
    annualValue: number;
    valuePerProvider: number;
  };
  
  expansionTarget: {
    totalProviders: number;
    targetUtilization: number;
    investmentPerProvider: number; // $/provider/month
  };
  
  projection: {
    linearValue: number;           // valuePerProvider × targetProviders
    utilizationBoost: number;      // multiplier for utilization increase
    maturityEffects: number;       // multiplier for maturity
    projectedValue: number;        // final projected value
    expansionValue: number;        // projectedValue - currentValue
    annualInvestment: number;      // targetProviders × investmentPerProvider × 12
    netValue: number;              // projectedValue - investment
    projectedRoi: number;          // projectedValue / investment
  };
  
  calculation: {
    steps: string[]; // Step-by-step calculation for transparency
  };
}

// Journey chart data
export interface JourneyChartData {
  points: {
    label: string;          // "Before", "Mo 4", "Today", "Mo 12", "Mo 24", "Full Scale"
    month: number;          // -3, 4, 7, 12, 24, 36 (negative = before Abridge)
    value: number;
    isActual: boolean;      // true = measured, false = projected
    isCurrent: boolean;     // true = "You are here"
    providers?: number;
  }[];
}

// Complete PDF data structure
export interface ExpandPDFData {
  // Deployment info
  deployment: DeploymentData;
  
  // Tracked metrics
  metrics: MetricData[];
  
  // Value breakdown
  valueBreakdown: ValueBreakdown;
  
  // Investment & ROI
  investment: {
    annualCost: number;
    costPerProvider: number;
  };
  currentRoi: number;
  
  // Time conversion config
  timeConversion: TimeConversionConfig;
  
  // Retention estimate (optional)
  retentionEstimate?: RetentionEstimate;
  
  // Journey chart
  journeyChart: JourneyChartData;
  
  // Expansion projection
  expansion: ExpansionProjection;
  
  // Warnings/flags
  warnings: {
    metricId?: MetricId;
    type: 'above_range' | 'below_range' | 'speculative' | 'data_quality';
    message: string;
  }[];
}

const formatCurrency = (value: number): string => {
  if (Math.abs(value) >= 1000000) {
    return `$${(value / 1000000).toFixed(1)}M`;
  }
  if (Math.abs(value) >= 1000) {
    return `$${Math.round(value / 1000)}K`;
  }
  return `$${value.toLocaleString()}`;
};

const formatNumber = (value: number): string => {
  return value.toLocaleString();
};

const formatChange = (value: number, unit: string, direction: 'increase' | 'decrease'): string => {
  const prefix = value >= 0 ? '+' : '';
  
  if (unit === '%') {
    return `${prefix}${value}%`;
  }
  if (unit === 'min') {
    return `${prefix}${value} min`;
  }
  if (unit === 'hrs/week') {
    return `${prefix}${value} hrs/week`;
  }
  if (unit === 'pts') {
    return `${prefix}${value} pts`;
  }
  if (unit === 'wRVU/enc') {
    return `${prefix}${value.toFixed(2)} wRVU/enc`;
  }
  if (unit === 'avg level') {
    return `${prefix}${value.toFixed(2)} avg level`;
  }
  
  return `${prefix}${value} ${unit}`;
};

const formatChangePercent = (beforeValue: number, afterValue: number): string => {
  if (beforeValue === 0) return 'N/A';
  const pctChange = ((afterValue - beforeValue) / beforeValue) * 100;
  const prefix = pctChange >= 0 ? '+' : '';
  return `${prefix}${pctChange.toFixed(0)}%`;
};

const getBenchmarkStatus = (
  metricId: MetricId,
  changePercent: number
): { status: 'below' | 'within' | 'above'; label: string } => {
  // Define typical ranges per metric
  const ranges: Record<MetricId, { min: number; max: number }> = {
    wrvuPerEncounter: { min: 3, max: 7 },
    timeInNotes: { min: 20, max: 40 }, // 20-40% reduction
    sameDayChartClosure: { min: 5, max: 15 }, // 5-15 pp improvement
    averageEmLevel: { min: 5, max: 15 }, // rough approximation
    workOutsideWork: { min: 20, max: 50 }, // 20-50% reduction
    clinicianSatisfaction: { min: 10, max: 30 }, // 10-30% improvement
  };
  
  const range = ranges[metricId];
  const absChange = Math.abs(changePercent);
  
  if (absChange < range.min) {
    return { status: 'below', label: 'Below typical range' };
  } else if (absChange > range.max) {
    return { status: 'above', label: 'Above typical range' };
  } else {
    return { status: 'within', label: 'Within expected range' };
  }
};

const getToday = (): string => {
  return new Date().toLocaleDateString("en-US", {
    year: "numeric",
    month: "long",
    day: "numeric",
  });
};
