Component Code for Replit:
jsxCopy// ============ INVESTMENT PAGE COMPONENT ============

export default function InvestmentPage({ 
  setting,
  units, // providers or beds depending on setting
  totalAnnualValue,
  valueBreakdown // array of { name, value }
}) {
  
  // Setting-specific configuration
  const settingConfig = {
    outpatient: {
      unitName: "provider",
      unitNamePlural: "providers",
      defaultPrice: 150,
      priceRange: "$100-$200",
      pricingLabel: "Cost per provider"
    },
    ed: {
      unitName: "provider",
      unitNamePlural: "providers", 
      defaultPrice: 175,
      priceRange: "$125-$225",
      pricingLabel: "Cost per provider"
    },
    inpatient: {
      unitName: "provider",
      unitNamePlural: "providers",
      defaultPrice: 175,
      priceRange: "$125-$225",
      pricingLabel: "Cost per provider"
    },
    nursing: {
      unitName: "staffed bed",
      unitNamePlural: "staffed beds",
      defaultPrice: 75,
      priceRange: "$50-$125",
      pricingLabel: "Cost per staffed bed"
    }
  };
  
  const config = settingConfig[setting];
  
  // State
  const [pricingModel, setPricingModel] = useState("per_unit_monthly");
  const [costPerUnit, setCostPerUnit] = useState(config.defaultPrice);
  const [contractTerm, setContractTerm] = useState(1);
  const [includeImplementation, setIncludeImplementation] = useState(false);
  const implementationFee = 25000;
  
  // Calculations
  const annualInvestment = units * costPerUnit * 12;
  const totalInvestment = (annualInvestment * contractTerm) + 
                          (includeImplementation ? implementationFee : 0);
  const netGainAnnual = totalAnnualValue - annualInvestment;
  const roiMultiple = (totalAnnualValue / annualInvestment).toFixed(1);
  const monthsToPayback = Math.round((annualInvestment / totalAnnualValue) * 12);
  
  return (
    <div className="investment-page">
      
      {/* Header */}
      <div className="page-header">
        <button className="back-button">← Back</button>
        <div className="brand">ABRIDGE</div>
        <div className="step-indicator">Step 4 of 5 · Your Investment</div>
      </div>
      
      {/* Title */}
      <div className="page-title">
        <h1>Your Investment</h1>
        <p>Align cost with value realization</p>
      </div>
      
      {/* Main Content - Two Columns */}
      <div className="investment-content">
        
        {/* Left Column - Inputs */}
        <div className="investment-inputs">
          
          <h2>CONFIGURE YOUR INVESTMENT</h2>
          
          {/* Pricing Model Toggle */}
          <div className="input-section">
            <label>Pricing Model</label>
            <div className="toggle-group">
              <button 
                className={pricingModel === "per_unit_monthly" ? "active" : ""}
                onClick={() => setPricingModel("per_unit_monthly")}
              >
                Per {config.unitName}/month
              </button>
              <button 
                className={pricingModel === "enterprise_annual" ? "active" : ""}
                onClick={() => setPricingModel("enterprise_annual")}
              >
                Enterprise annual
              </button>
            </div>
          </div>
          
          {/* Cost Input */}
          <div className="input-section">
            <label>{config.pricingLabel}</label>
            <div className="price-input">
              <span className="currency">$</span>
              <input 
                type="number" 
                value={costPerUnit}
                onChange={(e) => setCostPerUnit(Number(e.target.value))}
              />
              <span className="period">/month</span>
            </div>
            <p className="hint">💡 Typical: {config.priceRange}/{config.unitName}/month</p>
          </div>
          
          {/* Contract Term */}
          <div className="input-section">
            <label>Contract Term</label>
            <div className="toggle-group">
              <button 
                className={contractTerm === 1 ? "active" : ""}
                onClick={() => setContractTerm(1)}
              >
                1 yr
              </button>
              <button 
                className={contractTerm === 2 ? "active" : ""}
                onClick={() => setContractTerm(2)}
              >
                2 yrs
              </button>
              <button 
                className={contractTerm === 3 ? "active" : ""}
                onClick={() => setContractTerm(3)}
              >
                3 yrs
              </button>
            </div>
          </div>
          
          {/* Implementation Fee */}
          <div className="input-section">
            <label className="checkbox-label">
              <input 
                type="checkbox"
                checked={includeImplementation}
                onChange={(e) => setIncludeImplementation(e.target.checked)}
              />
              Add implementation fee
            </label>
            {includeImplementation && (
              <p className="implementation-note">One-time: ${implementationFee.toLocaleString()}</p>
            )}
          </div>
          
          {/* Configuration Summary */}
          <div className="config-summary">
            <h3>📊 YOUR CONFIGURATION</h3>
            <div className="config-math">
              <p>{units} {config.unitNamePlural}</p>
              <p>× ${costPerUnit}/{config.unitName}/month</p>
              <p>× 12 months</p>
              <div className="config-line"></div>
              <p className="config-total">= ${annualInvestment.toLocaleString()}/year</p>
              {contractTerm > 1 && (
                <>
                  <p className="config-term">{contractTerm}-year term</p>
                  <p className="config-total">Total: ${totalInvestment.toLocaleString()}</p>
                </>
              )}
            </div>
          </div>
          
        </div>
        
        {/* Right Column - ROI */}
        <div className="roi-display">
          
          <h2>YOUR ROI</h2>
          
          {/* Total Value */}
          <div className="roi-card">
            <label>Total Annual Value</label>
            <div className="roi-value">${totalAnnualValue.toLocaleString()}</div>
            <p className="roi-subtext">From {valueBreakdown.length} value drivers</p>
          </div>
          
          {/* Annual Investment */}
          <div className="roi-card">
            <label>Annual Investment</label>
            <div className="roi-value">${annualInvestment.toLocaleString()}</div>
            <p className="roi-subtext">
              {units} {config.unitNamePlural} × ${costPerUnit} × 12 months
            </p>
          </div>
          
          {/* Net Gain */}
          <div className="roi-card highlight">
            <label>NET GAIN</label>
            <div className="roi-value large green">${netGainAnnual.toLocaleString()}/yr</div>
          </div>
          
          {/* ROI Multiple */}
          <div className="roi-card highlight-large">
            <div className="roi-multiple">{roiMultiple}x</div>
            <label>RETURN ON INVESTMENT</label>
            <p className="payback">~{monthsToPayback} months to payback</p>
          </div>
          
          {/* Value Breakdown */}
          <div className="value-breakdown">
            <h3>Value Breakdown</h3>
            {valueBreakdown.map((item, index) => (
              <div className="breakdown-row" key={index}>
                <span className="breakdown-name">{item.name}</span>
                <span className="breakdown-value">${item.value.toLocaleString()}</span>
              </div>
            ))}
            <div className="breakdown-row total">
              <span className="breakdown-name">Total</span>
              <span className="breakdown-value">${totalAnnualValue.toLocaleString()}</span>
            </div>
          </div>
          
        </div>
        
      </div>
      
      {/* Footer Actions */}
      <div className="page-actions">
        <button className="primary-button">
          View Full Summary →
        </button>
        <button className="text-button">
          ← Edit value drivers
        </button>
      </div>
      
    </div>
  );
}
