The Math Flow Component - The Visual Formula
tsxCopy// MathFlow.tsx - Shows the calculation as a visual waterfall

interface MathStep {
  label: string;
  value: string | number;
  formula?: string;
  isInput?: boolean;
  isResult?: boolean;
}

interface MathFlowProps {
  steps: MathStep[];
  finalValue: number;
  color: 'gray' | 'emerald';
}

export function MathFlow({ steps, finalValue, color }: MathFlowProps) {
  const colorClasses = {
    gray: {
      line: 'bg-gray-300',
      dot: 'bg-gray-400',
      value: 'text-gray-700',
      result: 'bg-gray-100 border-gray-300',
      resultText: 'text-gray-900',
    },
    emerald: {
      line: 'bg-emerald-300',
      dot: 'bg-emerald-500',
      value: 'text-emerald-700',
      result: 'bg-emerald-100 border-emerald-300',
      resultText: 'text-emerald-700',
    },
  }[color];
  
  return (
    <div className="relative">
      {/* Vertical connection line */}
      <div className={`absolute left-3 top-3 bottom-3 w-0.5 ${colorClasses.line}`} />
      
      <div className="space-y-3">
        {steps.map((step, i) => (
          <div key={i} className="flex items-start gap-3 relative">
            {/* Dot */}
            <div className={`w-6 h-6 rounded-full ${colorClasses.dot} flex items-center justify-center z-10`}>
              {step.isInput && <span className="text-white text-xs">●</span>}
              {step.isResult && <span className="text-white text-xs">✓</span>}
              {!step.isInput && !step.isResult && <span className="text-white text-xs">{i + 1}</span>}
            </div>
            
            {/* Content */}
            <div className={`flex-1 ${step.isResult ? 'pb-0' : 'pb-2'}`}>
              <div className="flex items-baseline justify-between">
                <span className="text-sm text-gray-600">{step.label}</span>
                <span className={`font-semibold ${colorClasses.value}`}>
                  {typeof step.value === 'number' 
                    ? step.value.toLocaleString() 
                    : step.value}
                </span>
              </div>
              {step.formula && (
                <div className="text-xs text-gray-400 mt-0.5">{step.formula}</div>
              )}
            </div>
          </div>
        ))}
        
        {/* Final Result */}
        <div className={`ml-9 p-3 rounded-lg border ${colorClasses.result}`}>
          <div className="flex items-baseline justify-between">
            <span className="text-sm font-medium text-gray-600">Annual Value</span>
            <span className={`text-xl font-bold ${colorClasses.resultText}`}>
              ${finalValue.toLocaleString()}
            </span>
          </div>
        </div>
      </div>
    </div>
  );
}

interface DifferenceBarProps {
  competitorValue: number;
  abridgeValue: number;
  gap: number;
}

export function DifferenceBar({ competitorValue, abridgeValue, gap }: DifferenceBarProps) {
  const maxValue = Math.max(competitorValue, abridgeValue);
  const competitorWidth = (competitorValue / maxValue) * 100;
  const abridgeWidth = (abridgeValue / maxValue) * 100;
  
  return (
    <div className="space-y-2">
      {/* Abridge bar */}
      <div className="flex items-center gap-3">
        <div className="w-20 text-sm text-gray-500 text-right">Abridge</div>
        <div className="flex-1 h-8 bg-gray-100 rounded-lg overflow-hidden relative">
          <div 
            className="absolute inset-y-0 left-0 bg-emerald-500 rounded-lg transition-all duration-700 ease-out"
            style={{ width: `${abridgeWidth}%` }}
          />
          {/* The gap portion highlighted */}
          <div 
            className="absolute inset-y-0 bg-emerald-300 rounded-r-lg transition-all duration-700 ease-out opacity-50"
            style={{ 
              left: `${competitorWidth}%`, 
              width: `${abridgeWidth - competitorWidth}%` 
            }}
          />
          <div className="absolute inset-0 flex items-center px-3">
            <span className="text-white font-semibold text-sm drop-shadow">
              ${abridgeValue.toLocaleString()}
            </span>
          </div>
        </div>
      </div>
      
      {/* Competitor bar */}
      <div className="flex items-center gap-3">
        <div className="w-20 text-sm text-gray-400 text-right">Current</div>
        <div className="flex-1 h-8 bg-gray-100 rounded-lg overflow-hidden relative">
          <div 
            className="absolute inset-y-0 left-0 bg-gray-400 rounded-lg transition-all duration-700 ease-out"
            style={{ width: `${competitorWidth}%` }}
          />
          <div className="absolute inset-0 flex items-center px-3">
            <span className="text-white font-semibold text-sm drop-shadow">
              ${competitorValue.toLocaleString()}
            </span>
          </div>
        </div>
      </div>
      
      {/* Gap callout */}
      <div className="flex items-center gap-3">
        <div className="w-20" />
        <div className="flex-1 flex justify-end">
          <div className="inline-flex items-center gap-1 px-3 py-1 bg-emerald-100 rounded-full">
            <ArrowUpIcon className="w-3 h-3 text-emerald-600" />
            <span className="text-sm font-semibold text-emerald-700">
              +${gap.toLocaleString()} gap
            </span>
          </div>
        </div>
      </div>
    </div>
  );
}
