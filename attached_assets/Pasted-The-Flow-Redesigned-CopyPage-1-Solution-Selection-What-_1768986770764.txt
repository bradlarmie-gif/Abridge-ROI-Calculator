export default function SwitchCurrentSetup({

  const INDUSTRY_AVG_UTILIZATION = 45;

export default function SwitchPerformanceBenchmark({
  setupData,
  performanceData,
  setPerformanceData,
  onNext,
  onBack
}) {
  
  // Abridge benchmarks with quartiles
  const benchmarks = {
    utilization: {
      name: 'Utilization',
      unit: '%',
      quartiles: { q1: 45, median: 55, q3: 65, abridge: 65 },
      abridgeLabel: 'Abridge Avg'
    },
    timeSavings: {
      name: 'Time Savings',
      unit: 'min/enc',
      quartiles: { q1: 1.5, median: 2.5, q3: 3.5, abridge: 3.5 },
      abridgeLabel: 'Abridge Avg'
    },
    chartClosure: {
      name: 'Same-Day Chart Closure',
      unit: '%',
      quartiles: { q1: 25, median: 45, q3: 70, abridge: 78 },
      abridgeLabel: 'Abridge Avg'
    }
  };
  
  // Determine quartile position
  const getQuartilePosition = (value, metric) => {
    const q = benchmarks[metric].quartiles;
    if (value <= q.q1) return { label: 'Bottom 25%', position: 'q1', color: '#ef4444' };
    if (value <= q.median) return { label: 'Below Average', position: 'q2', color: '#f97316' };
    if (value <= q.q3) return { label: 'Above Average', position: 'q3', color: '#eab308' };
    return { label: 'Top 25%', position: 'q4', color: '#22c55e' };
  };
  
  return (
    <div className="switch-page">
      
      <header className="page-header">
        <button className="back-button" onClick={onBack}>
          <ArrowLeft size={16} />
          Back
        </button>
        <div className="header-center">
          <span className="brand-small">ABRIDGE</span>
          <span className="flow-badge">SWITCH</span>
        </div>
        <div className="step-dots">
          <span className="dot completed"></span>
          <span className="dot active"></span>
          <span className="dot"></span>
          <span className="dot"></span>
        </div>
      </header>
      
      <div className="switch-content">
        
        <div className="page-hero">
          <h1>How do you compare?</h1>
          <p>Let's see where you stand against industry benchmarks and Abridge performance.</p>
        </div>
        
        {/* Utilization Quartile */}
        <div className="quartile-card">
          <div className="quartile-header">
            <h3>Utilization</h3>
            <span className="your-value">{setupData.utilization}%</span>
          </div>
          
          <div className="quartile-visualization">
            <div className="quartile-bar">
              <div className="quartile-segment q1" style={{ width: '25%' }}>
                <span className="segment-label">Bottom 25%</span>
              </div>
              <div className="quartile-segment q2" style={{ width: '25%' }}>
                <span className="segment-label">Below Avg</span>
              </div>
              <div className="quartile-segment q3" style={{ width: '25%' }}>
                <span className="segment-label">Above Avg</span>
              </div>
              <div className="quartile-segment q4" style={{ width: '25%' }}>
                <span className="segment-label">Top 25%</span>
              </div>
            </div>
            
            {/* Your position marker */}
            <div 
              className="position-marker you"
              style={{ left: `${((setupData.utilization - 10) / 80) * 100}%` }}
            >
              <div className="marker-dot"></div>
              <div className="marker-label">YOU ({setupData.utilization}%)</div>
            </div>
            
            {/* Abridge marker */}
            <div 
              className="position-marker abridge"
              style={{ left: `${((benchmarks.utilization.quartiles.abridge - 10) / 80) * 100}%` }}
            >
              <div className="marker-dot abridge"></div>
              <div className="marker-label abridge">ABRIDGE ({benchmarks.utilization.quartiles.abridge}%)</div>
            </div>
            
            {/* Gap line */}
            {setupData.utilization < benchmarks.utilization.quartiles.abridge && (
              <div 
                className="gap-indicator"
                style={{
                  left: `${((setupData.utilization - 10) / 80) * 100}%`,
                  width: `${((benchmarks.utilization.quartiles.abridge - setupData.utilization) / 80) * 100}%`
                }}
              >
                <span className="gap-value">+{benchmarks.utilization.quartiles.abridge - setupData.utilization}% gap</span>
              </div>
            )}
          </div>
          
          <div className="quartile-insight">
            {setupData.utilization < benchmarks.utilization.quartiles.abridge ? (
              <p>
                You're at <strong>{setupData.utilization}%</strong> utilization. 
                Abridge customers average <strong>{benchmarks.utilization.quartiles.abridge}%</strong>.
                <span className="gap-highlight"> That's a {benchmarks.utilization.quartiles.abridge - setupData.utilization}pp gap.</span>
              </p>
            ) : (
              <p className="success">
                Great! Your utilization is at or above Abridge benchmarks.
              </p>
            )}
          </div>
        </div>
        
        {/* Time Savings Input & Quartile */}
        <div className="quartile-card with-input">
          <div className="quartile-header">
            <h3>Time Savings per Encounter</h3>
            <p className="header-hint">How much time does your current solution save?</p>
          </div>
          
          <div className="metric-input-section compact">
            <div className="input-with-slider">
              <span className="slider-value">{performanceData.timeSavings || 1.5} min</span>
              <input
                type="range"
                min="0.5"
                max="5"
                step="0.5"
                value={performanceData.timeSavings || 1.5}
                onChange={(e) => setPerformanceData({

                  timeSavings: Number(e.target.value)

              <span className="comparison-value highlight">{benchmarks.timeSavings.quartiles.abridge} min</span>
            </div>
          </div>

        {/* Same-Day Chart Closure Input & Quartile */}
        <div className="quartile-card with-input">
          <div className="quartile-header">
            <h3>Same-Day Chart Closure</h3>
            <p className="header-hint">What % of charts are closed within 24 hours?</p>
          </div>
          
          <div className="metric-input-section compact">
            <div className="input-with-slider">
              <span className="slider-value">{performanceData.chartClosure || 40}%</span>
              <input
                type="range"
                min="10"
                max="95"
                step="5"
                value={performanceData.chartClosure || 40}
                onChange={(e) => setPerformanceData({ 
                  ...performanceData, 
                  chartClosure: Number(e.target.value) 
                })}
                className="metric-slider"
              />
              <div className="slider-range">
                <span>10%</span>
                <span>95%</span>
              </div>
            </div>
          </div>

                      style={{ backgroundColor: position.color }}

                      {position.label}
                    </span>

                  {gap > 0 && (
                    <span className="metric-gap">
                      -{gap.toFixed(1)} vs Abridge
                    </span>
                  )}
                </div>
              );
            })}
          </div>
        </div>
        
        {/* Continue */}
        <div className="page-footer">
          <button className="primary-button" onClick={onNext}>
            See What This Costs You
            <ArrowRight size={16} />
          </button>
        </div>
        
      </div>
      
    </div>
  );

export default function SwitchGapAnalysis({
  setupData,
  performanceData,
  onNext,
  onBack
}) {
  
  const [expandedGap, setExpandedGap] = useState(null);
  
  // Calculate gaps
  const utilizationGap = 65 - setupData.utilization;
  const timeSavingsGap = 3.5 - performanceData.timeSavings;
  const chartClosureGap = 78 - performanceData.chartClosure;
  
  // Calculate $ value of gaps
  const encountersDocumented = setupData.encounters * (setupData.utilization / 100);
  const abridgeEncounters = setupData.encounters * 0.65;
  const encounterGap = abridgeEncounters - encountersDocumented;
  
  // Gap calculations (simplified for example)
  const gapValues = {
    utilization: {
      gap: utilizationGap,
      encounters: encounterGap,
      dollarValue: Math.round(encounterGap * 5.50) // $5.50 value per encounter (example)
    },
    timeSavings: {
      gap: timeSavingsGap,
      hoursGap: Math.round((timeSavingsGap * abridgeEncounters) / 60),
      dollarValue: Math.round((timeSavingsGap * abridgeEncounters / 60) * 150 * 0.5)
    },
    chartClosure: {
      gap: chartClosureGap,
      description: 'Revenue cycle efficiency',
      dollarValue: Math.round(chartClosureGap * 1000) // Simplified
    }
  };
  
  const totalGap = gapValues.utilization.dollarValue + gapValues.timeSavings.dollarValue + gapValues.chartClosure.dollarValue;
  
  return (
    <div className="switch-page">
      
      <header className="page-header">
        <button className="back-button" onClick={onBack}>
          <ArrowLeft size={16} />
          Back
        </button>
        <div className="header-center">
          <span className="brand-small">ABRIDGE</span>
          <span className="flow-badge">SWITCH</span>
        </div>
        <div className="step-dots">
          <span className="dot completed"></span>
          <span className="dot completed"></span>
          <span className="dot active"></span>
          <span className="dot"></span>
        </div>
      </header>
      
      <div className="switch-content">
        
        {/* Hero — The Big Number */}
        <div className="gap-hero">
          <p className="hero-pretext">Based on what you've told us, you're leaving</p>
          <div className="gap-value-display">
            <span className="gap-dollar">$</span>
            <span className="gap-amount">{formatNumber(totalGap)}</span>
          </div>
          <p className="hero-posttext">on the table. <strong>Every. Single. Year.</strong></p>
          
          <div className="gap-breakdown-mini">
            <span>${formatNumber(Math.round(totalGap / 12))}/month</span>
            <span className="divider">·</span>
            <span>${formatNumber(Math.round(totalGap / 365))}/day</span>
            <span className="divider">·</span>
            <span>${formatNumber(Math.round(totalGap / 8760))}/hour</span>
          </div>
        </div>
        
        {/* Gaps Identified */}
        <div className="gaps-section">
          <h2>THE GAPS WE IDENTIFIED</h2>
          
          <div className="gaps-summary">
            <div className="gap-summary-row">
              <span className="gap-label">Utilization:</span>
              <span className="gap-change">
                {setupData.utilization}% → 65%
              </span>
              <span className="gap-impact success">= +{formatNumber(encounterGap)} encounters</span>
            </div>
            <div className="gap-summary-row">
              <span className="gap-label">Efficiency:</span>
              <span className="gap-change">
                {performanceData.timeSavings} min → 3.5 min
              </span>
              <span className="gap-impact success">= +{formatNumber(gapValues.timeSavings.hoursGap)} hours</span>
            </div>
          </div>
        </div>
        
        {/* Tab Toggle */}
        <div className="view-toggle">
          <button className="toggle-option active">Value Breakdown</button>
          <button className="toggle-option">Over Time</button>
        </div>
        
        {/* Expandable Gap Cards */}
        <div className="gap-cards">
          
          {/* Utilization Gap */}
          <div 
            className={`gap-card ${expandedGap === 'utilization' ? 'expanded' : ''}`}
            onClick={() => setExpandedGap(expandedGap === 'utilization' ? null : 'utilization')}
          >
            <div className="gap-card-header">
              <div className="gap-card-icon">
                <Users size={20} />
              </div>
              <div className="gap-card-title">
                <h3>Patient Access</h3>
                <p>More encounters documented = more value captured</p>
              </div>
              <div className="gap-card-value">
                +${formatNumber(gapValues.utilization.dollarValue)}/yr
              </div>
              <ChevronDown size={20} className={`expand-icon ${expandedGap === 'utilization' ? 'rotated' : ''}`} />
            </div>
            
            {expandedGap === 'utilization' && (
              <div className="gap-card-body">
                <div className="gap-calculation">
                  <div className="calc-row">
                    <span>Utilization gap</span>
                    <span>{utilizationGap}pp</span>
                  </div>
                  <div className="calc-row">
                    <span>Additional encounters</span>
                    <span>+{formatNumber(encounterGap)}</span>
                  </div>
                  <div className="calc-row">
                    <span>Value per encounter</span>
