. Number Formatting with Commas
jsxCopy// Utility function for formatting numbers with commas
const formatNumber = (value) => {
  if (value === null || value === undefined || value === '') return '';
  return Number(value).toLocaleString();
};

// For input fields, we need to handle both display and storage
const NumberInput = ({ value, onChange, placeholder, ...props }) => {
  const [displayValue, setDisplayValue] = useState(
    value ? formatNumber(value) : ''
  );
  
  const handleChange = (e) => {
    // Remove commas for storage
    const rawValue = e.target.value.replace(/,/g, '');
    setDisplayValue(e.target.value);
    onChange(rawValue ? Number(rawValue) : null);
  };
  
  const handleBlur = () => {
    // Format on blur
    if (value) {
      setDisplayValue(formatNumber(value));
    }
  };
  
  const handleFocus = () => {
    // Show raw number on focus for easier editing
    if (value) {
      setDisplayValue(value.toString());
    }
  };
  
  return (
    <input
      type="text"
      inputMode="numeric"
      value={displayValue}
      onChange={handleChange}
      onBlur={handleBlur}
      onFocus={handleFocus}
      placeholder={placeholder}
      {...props}
    />
  );
};

jsxCopy// ============ LEVEL OF SERVICE INPUT — REWORKED ============

import { useMemo } from 'react';

export default function LevelOfServiceInput({ data, setData }) {
  
  const levels = [
    { id: 'level5', label: 'Level 5', description: 'High complexity' },
    { id: 'level4', label: 'Level 4', description: 'Moderate-high complexity' },
    { id: 'level3', label: 'Level 3', description: 'Moderate complexity' },
    { id: 'level2', label: 'Level 2', description: 'Low complexity' },
    { id: 'level1', label: 'Level 1', description: 'Minimal complexity' }
  ];
  
  const weights = {
    level5: 5,
    level4: 4,
    level3: 3,
    level2: 2,
    level1: 1
  };
  
  // Calculate totals
  const beforeTotal = useMemo(() => {
    return levels.reduce((sum, l) => sum + (data.before?.[l.id] || 0), 0);
  }, [data.before]);
  
  const afterTotal = useMemo(() => {
    return levels.reduce((sum, l) => sum + (data.after?.[l.id] || 0), 0);
  }, [data.after]);
  
  // Calculate weighted averages
  const calculateAverage = (distribution) => {
    if (!distribution) return null;
    let weighted = 0;
    let total = 0;
    levels.forEach(level => {
      const pct = distribution[level.id] || 0;
      weighted += pct * weights[level.id];
      total += pct;
    });
    return total > 0 ? (weighted / total).toFixed(2) : null;
  };
  
  const beforeAvg = calculateAverage(data.before);
  const afterAvg = calculateAverage(data.after);
  const avgChange = beforeAvg && afterAvg 
    ? (((afterAvg - beforeAvg) / beforeAvg) * 100).toFixed(1) 
    : null;
  
  const updateBefore = (levelId, value) => {
    setData({
      ...data,
      before: { ...data.before, [levelId]: value ? Number(value) : 0 }
    });
  };
  
  const updateAfter = (levelId, value) => {
    setData({
      ...data,
      after: { ...data.after, [levelId]: value ? Number(value) : 0 }
    });
  };
  
  return (
    <section className="metric-input-section">
      
      <div className="metric-input-header">
        <div className="metric-icon-container">
          <ClipboardList size={24} className="metric-icon" />
        </div>
        <div className="metric-title-group">
          <h2>Level of Service</h2>
          <p>E/M code distribution</p>
        </div>
      </div>
      
      {/* Blend Note */}
      <div className="blend-note">
        <Info size={14} className="blend-note-icon" />
        <span>
          Enter your blended distribution across new and established patient visits. 
          Level 5 = highest complexity, Level 1 = lowest.
        </span>
      </div>
      
      {/* Averages Summary — Prominent */}
      {beforeAvg && afterAvg && beforeTotal === 100 && afterTotal === 100 && (
        <div className="average-summary-card">
          <div className="avg-summary-row">
            <div className="avg-summary-item">
              <span className="avg-label">AVERAGE BEFORE</span>
              <span className="avg-value muted">{beforeAvg}</span>
            </div>
            <div className="avg-arrow">→</div>
            <div className="avg-summary-item">
              <span className="avg-label">AVERAGE AFTER</span>
              <span className="avg-value">{afterAvg}</span>
            </div>
            <div className="avg-summary-item highlight">
              <span className="avg-label">CHANGE</span>
              <span className="avg-value success">+{avgChange}%</span>
            </div>
          </div>
          <p className="avg-note">
            Weighted average where Level 1 = 1, Level 2 = 2, etc.
          </p>
        </div>
      )}
      
      {/* Input Table */}
      <div className="los-input-table">
        
        <div className="los-table-header">
          <div className="los-col-level">LEVEL</div>
          <div className="los-col-before">BEFORE ABRIDGE</div>
          <div className="los-col-after">AFTER ABRIDGE</div>
          <div className="los-col-change">CHANGE</div>
        </div>
        
        {levels.map(level => {
          const beforeVal = data.before?.[level.id] || 0;
          const afterVal = data.after?.[level.id] || 0;
          const change = afterVal - beforeVal;
          
          return (
            <div key={level.id} className="los-table-row">
              <div className="los-col-level">
                <span className="level-label">{level.label}</span>
                <span className="level-description">{level.description}</span>
              </div>
              
              <div className="los-col-before">
                <div className="los-input-wrapper">
                  <input
                    type="number"
                    min="0"
                    max="100"
                    placeholder="0"
                    value={beforeVal || ''}
                    onChange={(e) => updateBefore(level.id, e.target.value)}
                    className="los-input"
                  />
                  <span className="los-input-suffix">%</span>
                </div>
                <div 
                  className="los-bar before-bar"
                  style={{ width: `${Math.min(beforeVal, 100)}%` }}
                ></div>
              </div>
              
              <div className="los-col-after">
                <div className="los-input-wrapper">
                  <input
                    type="number"
                    min="0"
                    max="100"
                    placeholder="0"
                    value={afterVal || ''}
                    onChange={(e) => updateAfter(level.id, e.target.value)}
                    className="los-input"
                  />
                  <span className="los-input-suffix">%</span>
                </div>
                <div 
                  className="los-bar after-bar"
                  style={{ width: `${Math.min(afterVal, 100)}%` }}
                ></div>
              </div>
              
              <div className="los-col-change">
                {change !== 0 && (
                  <span className={`los-change ${change > 0 ? 'positive' : 'negative'}`}>
                    {change > 0 ? '↑' : '↓'} {Math.abs(change)}pp
                  </span>
                )}
                {change === 0 && <span className="los-change neutral">—</span>}
              </div>
            </div>
          );
        })}
        
        {/* Totals Row */}
        <div className="los-table-row totals-row">
          <div className="los-col-level">
            <span className="level-label">TOTAL</span>
          </div>
          <div className="los-col-before">
            <span className={`los-total ${beforeTotal === 100 ? 'valid' : 'invalid'}`}>
              {beforeTotal}%
              {beforeTotal !== 100 && <span className="total-hint"> (must = 100)</span>}
            </span>
          </div>
          <div className="los-col-after">
            <span className={`los-total ${afterTotal === 100 ? 'valid' : 'invalid'}`}>
              {afterTotal}%
              {afterTotal !== 100 && <span className="total-hint"> (must = 100)</span>}
            </span>
          </div>
          <div className="los-col-change"></div>
        </div>
        
      </div>
      
      {/* Insight */}
      {avgChange && parseFloat(avgChange) > 0 && (
        <div className="insight-callout success">
          <Lightbulb size={18} className="insight-icon" />
          <div className="insight-content">
            <strong>+{avgChange}% shift in average level of service</strong>
            <p>
              This suggests more complete documentation of clinical complexity. 
              Better documentation captures work already being done — not upcoding.
            </p>
          </div>
        </div>
      )}
      
    </section>
  );
}
