JAVASCRIPT FOR INTERACTIVITY
javascriptCopy// Track selections
let selectedPriorities = new Set();

// Update counter and validation
function updateSelections() {
  const count = selectedPriorities.size;
  
  // Update counter in header
  document.querySelector('.counter-number').textContent = count;
  const counterCircle = document.querySelector('.counter-circle');
  if (count > 0) {
    counterCircle.classList.add('has-selections');
  } else {
    counterCircle.classList.remove('has-selections');
  }
  
  // Update sidebar count
  document.getElementById('priority-count').textContent = count;
  
  // Update selected list
  const listEl = document.getElementById('selected-list');
  listEl.innerHTML = '';
  selectedPriorities.forEach(priority => {
    const li = document.createElement('li');
    li.textContent = priority;
    listEl.appendChild(li);
  });
  
  // Enable/disable continue button
  const continueBtn = document.getElementById('continue-btn');
  if (count >= 2 && count <= 6) {
    continueBtn.disabled = false;
  } else {
    continueBtn.disabled = true;
  }
}

// Listen to checkbox changes
document.querySelectorAll('.priority-card-compact input[type="checkbox"]').forEach(checkbox => {
  checkbox.addEventListener('change', function() {
    const label = this.closest('label').querySelector('h4').textContent;
    
    if (this.checked) {
      selectedPriorities.add(label);
    } else {
      selectedPriorities.delete(label);
    }
    
    updateSelections();
  });
});
