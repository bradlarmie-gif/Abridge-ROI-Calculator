6. Calculation Logic
javascriptCopy// calculations.js

// Human Scribe Calculations
export const calculateScribeCost = (numScribes, hourlyRate, hoursPerWeek) => {
  const weeklyPayroll = numScribes * hourlyRate * hoursPerWeek;
  const annualPayroll = weeklyPayroll * 52;
  const fullyLoadedCost = annualPayroll * 1.3; // Benefits, taxes, etc.
  return fullyLoadedCost;
};

export const calculateScribeSavings = (scribeCost, reductionPercent, abridgeCost) => {
  const savings = (scribeCost * (reductionPercent / 100)) - abridgeCost;
  return savings;
};

// Abridge Benchmark Ranges (based on your strategic priorities)
export const abridgeBenchmarks = {
  patientAccess: {
    min: 100000,
    max: 200000,
    typical: 130000,
    description: "Additional revenue from new visit capacity"
  },
  clinicianRetention: {
    min: 50000,
    max: 100000,
    typical: 60000,
    description: "Reduced turnover and replacement costs"
  },
  accurateService: {
    min: 100000,
    max: 180000,
    typical: 136500,
    description: "Improved wRVU capture and coding accuracy"
  },
  overtimeSavings: {
    min: 5000,
    max: 15000,
    typical: 9425,
    description: "Reduced premium labor hours"
  },
  denialReduction: {
    min: 30000,
    max: 80000,
    typical: 58500,
    description: "Fewer documentation-related denials"
  }
};

// Compare competitor to Abridge
export const calculateComparison = (competitorData, abridgeData) => {
  return {
    costDelta: abridgeData.annualCost - competitorData.annualCost,
    benefitDelta: abridgeData.totalBenefit - competitorData.totalBenefit,
    roiDelta: abridgeData.roi - competitorData.roi,
    netValueDelta: (abridgeData.totalBenefit - abridgeData.annualCost) - 
                   (competitorData.totalBenefit - competitorData.annualCost)
  };
};
