3. Math Flow Steps Component (Updated for Orange)
tsxCopyinterface MathStep {
  label: string;
  value: string;
  formula?: string;
  isEditable?: boolean;
  editKey?: string;
}

function MathFlowSteps({ 
  steps, 
  color,
  onEdit 
}: { 
  steps: MathStep[]; 
  color: 'gray' | 'orange';
  onEdit?: (key: string, value: number) => void;
}) {
  const styles = {
    gray: {
      dot: 'bg-gray-300',
      number: 'bg-gray-200 text-gray-600',
      line: 'bg-gray-200',
      value: 'text-gray-700',
      formula: 'text-gray-400',
    },
    orange: {
      dot: 'bg-orange-400',
      number: 'bg-orange-100 text-orange-600',
      line: 'bg-orange-200',
      value: 'text-orange-700',
      formula: 'text-orange-400',
    },
  }[color];
  
  return (
    <div className="relative">
      {/* Vertical line */}
      <div className={`absolute left-3 top-3 bottom-3 w-0.5 ${styles.line}`} />
      
      <div className="space-y-3">
        {steps.map((step, i) => (
          <div key={i} className="flex items-start gap-3 relative">
            {/* Step indicator */}
            <div className={`w-6 h-6 rounded-full ${styles.number} flex items-center justify-center text-xs font-medium z-10`}>
              {step.isEditable ? '●' : i + 1}
            </div>
            
            {/* Content */}
            <div className="flex-1 flex items-center justify-between">
              <div>
                <span className="text-sm text-gray-600">{step.label}</span>
                {step.formula && (
                  <div className={`text-xs ${styles.formula}`}>{step.formula}</div>
                )}
              </div>
              <div className="flex items-center gap-2">
                <span className={`font-semibold ${styles.value}`}>{step.value}</span>
                {step.isEditable && onEdit && (
                  <button 
                    onClick={() => {/* open edit modal */}}
                    className="p-1 hover:bg-gray-100 rounded text-gray-400 hover:text-gray-600"
                  >
                    <PencilIcon className="w-3 h-3" />
                  </button>
                )}
              </div>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}

5. Main Step 3 Component
tsxCopyfunction GapAnalysisStep({ 
  competitor,
  inputs,
  onContinue 
}: Props) {
  // All available drivers
  const allDrivers: Driver[] = [
    { id: 'patient_access', label: 'Patient Access', subtitle: 'Turn documentation efficiency into additional visit capacity', icon: <UserPlusIcon /> },
    { id: 'overtime', label: 'Overtime Savings', subtitle: 'Reduce after-hours documentation burden', icon: <ClockIcon /> },
    { id: 'level_of_service', label: 'Level of Service', subtitle: 'Capture accurate wRVU through complete documentation', icon: <ChartBarIcon /> },
    { id: 'retention', label: 'Clinician Retention', subtitle: 'Reduce burnout-driven turnover', icon: <HeartIcon /> },
    { id: 'hcc', label: 'HCC Capture', subtitle: 'Improve risk adjustment through documentation', icon: <ClipboardIcon /> },
    { id: 'denials', label: 'Documentation-Related Denials', subtitle: 'Reduce preventable claim denials', icon: <ExclamationIcon /> },
  ];
  
  // Track which drivers are selected
  const [selectedDriverIds, setSelectedDriverIds] = useState<string[]>([]);
  
  // Calculate values for each driver
  const driverCalculations = useMemo(() => {
    return allDrivers.map(driver => ({
      ...driver,
      ...calculateDriver(driver.id, inputs, competitor),
    }));
  }, [inputs, competitor]);
  
  // Selected drivers with their calculations
  const selectedDrivers = driverCalculations.filter(d => selectedDriverIds.includes(d.id));
  
  const toggleDriver = (driverId: string) => {
    setSelectedDriverIds(prev => 
      prev.includes(driverId)
        ? prev.filter(id => id !== driverId)
        : [...prev, driverId]
    );
  };
  
  return (
    <div className="space-y-8">
      {/* Gap at a Glance - keep existing */}
      <GapAtAGlance competitor={competitor} inputs={inputs} />
      
      {/* How the Gap Translates to Value */}
      <section>
        <div className="mb-6">
          <h2 className="text-xl font-semibold text-gray-900">
            How the Gap Translates to Value
          </h2>
          <p className="text-gray-500 mt-1">
            Select the value drivers that matter most to your organization
          </p>
        </div>
        
        <div className="space-y-4">
          {driverCalculations.map(driver => (
            <DriverCard
              key={driver.id}
              driver={driver}
              isSelected={selectedDriverIds.includes(driver.id)}
              onToggle={toggleDriver}
              calculations={driver}
              competitor={competitor}
            />
          ))}
        </div>
      </section>
      
      {/* Summary */}
      <DriverSummary 
        selectedDrivers={selectedDrivers}
        onContinue={() => onContinue(selectedDrivers)}
      />
    </div>
  );
}
